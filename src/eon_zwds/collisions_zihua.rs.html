10/// 각 궁의 천간(宮幹)에 의해 자궁 내 주성에 발생하는 자화(自化)를 검출합니다.
11pub fn detect_zi_hua(palaces: &[PalaceData; 12]) -> Vec<ZiHuaInfo> {
12    let mut results = Vec::new();
13
14    for p in palaces {
15        if let Ok(stem) = parse_stem(&p.heavenly_stem) {
16            let sihua_stars = get_sihua_stars(stem);
17            let sihua_types = [
18                SiHuaType::HuaLu,
19                SiHuaType::HuaQuan,
20                SiHuaType::HuaKe,
21                SiHuaType::HuaJi,
22            ];
23
24            for (idx, &s_star) in sihua_stars.iter().enumerate() {
25                if p.stars.iter().any(|st| st.star == s_star) {
26                    let s_type = sihua_types[idx];
27                    let desc = format!(
28                        "{}에서 궁간({}) 자화({}) 발생 — {} 별이 자화됨",
29                        p.name.korean(),
30                        p.heavenly_stem,
31                        s_type.korean(),
32                        s_star.korean()
33                    );
34                    results.push(ZiHuaInfo {
35                        palace_idx: p.index,
36                        palace_name: p.name,
37                        star: s_star,
38                        sihua_type: s_type,
39                        description: desc,
40                    });
41                }
42            }
43        }
44    }
45
46    results
47}

49/// 주성이 없는 공궁(空宮)일 때 대궁(對宮)의 주성을 빌려오는 차성기궁(借星騎宮)을 검출합니다.
50pub fn detect_borrowed_stars(palaces: &[PalaceData; 12]) -> Vec<BorrowedStarsInfo> {
51    let mut results = Vec::new();
52
53    for p in palaces {
54        // 14 주성 존재 여부 체크
55        let has_major_star = p.stars.iter().any(|st| st.star.is_major());
56        if !has_major_star {
57            let opposite_idx = (p.index + 6) % 12;
58            let opp_palace = &palaces[opposite_idx];
59            let borrowed: Vec<ZwdsStar> = opp_palace
60                .stars
61                .iter()
62                .filter(|st| st.star.is_major())
63                .map(|st| st.star)
64                .collect();
65
66            if !borrowed.is_empty() {
67                results.push(BorrowedStarsInfo {
68                    palace_idx: p.index,
69                    palace_name: p.name,
70                    opposite_palace_idx: opposite_idx,
71                    borrowed_stars: borrowed,
72                });
73            }
74        }
75    }
76
77    results
78}

80/// 생년 사화 + 대한 사화 + 유년 사화 간의 3중 충국 및 상충 패턴을 감지합니다.
81pub fn detect_triple_sihua_collisions(
82    palaces: &[PalaceData; 12],
83    daxian_list: &[DaXian],
84    current_liunian: Option<&LiuNian>,
85) -> Vec<TripleSihuaCollision> {
86    let mut collisions = Vec::new();
87
88    if daxian_list.is_empty() {
89        return collisions;
90    }
91
92    // 대표 대한 사화 추출 (첫번째 대한 기준 또는 지정 대한)
93    let active_daxian = &daxian_list[0];
94    let daxian_stem = parse_stem(&active_daxian.stem_hanja).ok();
95
96    for p in palaces {
97        let mut huaji_count = 0;
98        let mut hualu_count = 0;
99        let mut stars_involved = Vec::new();
100
101        // 1. 생년 사화 체크
102        for st in &p.stars {
103            if let Some(s_type) = st.si_hua {
104                match s_type {
105                    SiHuaType::HuaJi => {
106                        huaji_count += 1;
107                        stars_involved.push(st.star);
108                    }
109                    SiHuaType::HuaLu => {
110                        hualu_count += 1;
111                        stars_involved.push(st.star);
112                    }
113                    _ => {}
114                }
115            }
116        }
117
118        // 2. 대한 사화 체크
119        if let Some(d_stem) = daxian_stem {
120            let d_sihua = get_sihua_stars(d_stem);
121            if p.stars.iter().any(|st| st.star == d_sihua[3]) {
122                huaji_count += 1;
123                stars_involved.push(d_sihua[3]);
124            }
125            if p.stars.iter().any(|st| st.star == d_sihua[0]) {
126                hualu_count += 1;
127                stars_involved.push(d_sihua[0]);
128            }
129        }
130
131        // 3. 유년 사화 체크
132        if let Some(ln) = current_liunian {
133            if p.stars.iter().any(|st| st.star == ln.si_hua[3]) {
134                huaji_count += 1;
135                stars_involved.push(ln.si_hua[3]);
136            }
137            if p.stars.iter().any(|st| st.star == ln.si_hua[0]) {
138                hualu_count += 1;
139                stars_involved.push(ln.si_hua[0]);
140            }
141        }
142
143        stars_involved.sort();
144        stars_involved.dedup();
145
146        if huaji_count >= 3 {
147            collisions.push(TripleSihuaCollision {
148                palace_idx: p.index,
149                palace_name: p.name,
150                collision_type: "삼화기 (三化忌 — 3중 화기 중첩)".to_string(),
151                severity: "Crisis".to_string(),
152                description: format!(
153                    "{}에 3중 화기(化忌)가 집결하여 극심한 변화 및 과제가 집중되는 연도/시기입니다.",
154                    p.name.korean()
155                ),
156                stars_involved: stars_involved.clone(),
157            });
158        } else if huaji_count >= 2 {
159            collisions.push(TripleSihuaCollision {
160                palace_idx: p.index,
161                palace_name: p.name,
162                collision_type: "쌍화기 (雙化忌 — 2중 화기 중첩)".to_string(),
163                severity: "Caution".to_string(),
164                description: format!(
165                    "{}에 2중 화기(化忌)가 중첩하여 주의와 세심한 관리가 요구되는 시기입니다.",
166                    p.name.korean()
167                ),
168                stars_involved: stars_involved.clone(),
169            });
170        }
171
172        if hualu_count >= 2 {
173            collisions.push(TripleSihuaCollision {
174                palace_idx: p.index,
175                palace_name: p.name,
176                collision_type: "쌍록 (雙祿 / 祿轉成福 — 2중 화록 중첩)".to_string(),
177                severity: "Opportunity".to_string(),
178                description: format!(
179                    "{}에 2중 화록(化祿)이 중첩하여 대단한 재물과 기회가 함께 도래하는 길시입니다.",
180                    p.name.korean()
181                ),
182                stars_involved: stars_involved.clone(),
183            });
184        } else if hualu_count >= 1 && huaji_count >= 1 {
185            collisions.push(TripleSihuaCollision {
186                palace_idx: p.index,
187                palace_name: p.name,
188                collision_type: "록기상충 (祿忌相沖 — 기회와 위험 교차)".to_string(),
189                severity: "Caution".to_string(),
190                description: format!(
191                    "{}에 화록과 화기가 공존하여 큰 성과 뒤에 리스크 관리가 동시에 요구됩니다.",
192                    p.name.korean()
193                ),
194                stars_involved,
195            });
196        }
197    }
198
199    collisions
200}

202fn parse_stem(s: &str) -> Result<HeavenlyStem, ()> {
203    match s {
204        "甲" | "갑" => Ok(HeavenlyStem::Jia),
205        "乙" | "을" => Ok(HeavenlyStem::Yi),
206        "丙" | "병" => Ok(HeavenlyStem::Bing),
207        "丁" | "정" => Ok(HeavenlyStem::Ding),
208        "戊" | "무" => Ok(HeavenlyStem::Wu),
209        "己" | "기" => Ok(HeavenlyStem::Ji),
210        "庚" | "경" => Ok(HeavenlyStem::Geng),
211        "辛" | "신" => Ok(HeavenlyStem::Xin),
212        "壬" | "임" => Ok(HeavenlyStem::Ren),
213        "癸" | "계" => Ok(HeavenlyStem::Gui),
214        _ => Err(()),
215    }
