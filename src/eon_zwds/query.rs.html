1//! Public query API over a built [`ZwdsChart`]: 飛星四化 (flying transformations), 三方四正, 大限四化,
2//! 小限 and 斗君.
3//!
4//! The natal [`ZwdsChart::flying_sihua`] list is a flat dump of all 48 flights; the methods here
5//! answer targeted questions ("where does the 夫妻宮 stem send 化忌?", "which palaces fly into the
6//! 命宮?", "what is in the 三方四正 of the 財帛宮?") and work for any stem, so the same machinery
7//! serves 大限四化, 小限四化 and the 合盤 cross-chart lookups ([`crate::compat`]).
8

11    fix_index, get_palace_name, get_palace_stem, palace_index_of_name, zwds_idx_to_std_idx,
12};
13use crate::transformations::get_sihua_stars;
14use crate::types::{
15    DaXian, PalaceData, PalaceIndex, PalaceName, SiHuaType, StarInPalace, ZwdsChart, ZwdsStar,

21/// Where one transformation of a stem lands in a chart.
22#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
23pub struct SiHuaLanding {
24    pub sihua_type: SiHuaType,
25    pub star: ZwdsStar,
26    /// Palace (ZWDS index) holding the transformed star.
27    pub palace_idx: PalaceIndex,
28    /// Natal name of that palace.
29    pub palace_name: PalaceName,
30}

32/// The four transformations flown out of one palace's stem.
33#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
34pub struct PalaceSiHua {
35    /// Source palace (ZWDS index).
36    pub from_palace_idx: PalaceIndex,
37    /// Natal name of the source palace.
38    pub from_palace: PalaceName,
39    /// Heavenly stem of the source palace.
40    pub stem_hanja: String,
41    /// 化祿, 化權, 化科, 化忌 landings (in that order).
42    pub landings: Vec<SiHuaLanding>,
43}

45/// 三方四正 of a palace: itself, its opposite (對宮) and the two trine palaces (三合).
46#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
47pub struct SanFangSiZheng {
48    pub palace_idx: PalaceIndex,
49    pub palace: PalaceName,
50    /// Opposite palace (對宮, +6).
51    pub opposite_idx: PalaceIndex,
52    pub opposite: PalaceName,
53    /// The two trine palaces (三合, ±4), in ascending palace-index order.
54    pub trine_idx: [PalaceIndex; 2],
55    pub trine: [PalaceName; 2],
56}
57
58impl SanFangSiZheng {
59    /// The four palace indices: self, opposite, then the two trines.
60    pub fn palace_indices(&self) -> [PalaceIndex; 4] {
61        [
62            self.palace_idx,
63            self.opposite_idx,
64            self.trine_idx[0],
65            self.trine_idx[1],
66        ]
67    }
68}

70/// Stars of a palace inside a 三方四正 group.
71#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
72pub struct SanFangStars {
73    pub palace_idx: PalaceIndex,
74    pub palace: PalaceName,
75    pub stars: Vec<StarInPalace>,
76}

78impl ZwdsChart {
79    /// Natal year stem (from the lunar birth year the chart was built from).
80    pub fn year_stem(&self) -> HeavenlyStem {
81        HeavenlyStem::from_index(self.birth_lunar.year_stem_idx as i32)
82    }
83
84    /// Heavenly stem of a palace (五虎遁 from the natal year stem).
85    pub fn palace_stem(&self, palace_idx: PalaceIndex) -> HeavenlyStem {
86        get_palace_stem(self.year_stem(), palace_idx % 12)
87    }
88
89    /// ZWDS index of the palace carrying `name`.
90    pub fn palace_index_of(&self, name: PalaceName) -> PalaceIndex {
91        palace_index_of_name(self.soul_idx, name)
92    }
93
94    /// The palace carrying `name`.
95    pub fn palace(&self, name: PalaceName) -> &PalaceData {
96        &self.palaces[self.palace_index_of(name)]
97    }
98
99    /// Natal name of the palace at `palace_idx`.
100    pub fn palace_name_at(&self, palace_idx: PalaceIndex) -> PalaceName {
101        get_palace_name(self.soul_idx, palace_idx % 12)
102    }
103
104    /// Palace holding `star`, if it is placed in this chart.
105    pub fn star_palace(&self, star: ZwdsStar) -> Option<PalaceIndex> {
106        self.palaces
107            .iter()
108            .find(|p| p.stars.iter().any(|s| s.star == star))
109            .map(|p| p.index)
110    }
111
112    /// Where the four transformations of `stem` land in this chart.
113    ///
114    /// This is the primitive behind every 飛星 query: pass the stem of a palace of *this* chart
115    /// (飛星), of another person's chart (合盤) or of a luck period (大限/小限/流年).
116    pub fn si_hua_landings_for_stem(&self, stem: HeavenlyStem) -> Vec<SiHuaLanding> {
117        let stars = get_sihua_stars(stem);
118        let types = [
119            SiHuaType::HuaLu,
120            SiHuaType::HuaQuan,
121            SiHuaType::HuaKe,
122            SiHuaType::HuaJi,
123        ];
124        types
125            .into_iter()
126            .zip(stars)
127            .filter_map(|(sihua_type, star)| {
128                self.star_palace(star).map(|palace_idx| SiHuaLanding {
129                    sihua_type,
130                    star,
131                    palace_idx,
132                    palace_name: self.palace_name_at(palace_idx),
133                })
134            })
135            .collect()
136    }
137
138    /// 飛星: the four transformations flown from the stem of the palace at `from_idx`.
139    pub fn flying_from_index(&self, from_idx: PalaceIndex) -> PalaceSiHua {
140        let from_idx = from_idx % 12;
141        let stem = self.palace_stem(from_idx);
142        PalaceSiHua {
143            from_palace_idx: from_idx,
144            from_palace: self.palace_name_at(from_idx),
145            stem_hanja: stem.hanja().to_string(),
146            landings: self.si_hua_landings_for_stem(stem),
147        }
148    }
149
150    /// 飛星: the four transformations flown from palace `from` (by natal name).
151    pub fn flying_from(&self, from: PalaceName) -> PalaceSiHua {
152        self.flying_from_index(self.palace_index_of(from))
153    }
154
155    /// 飛星 from palace `from` that land in palace `to` (empty when none do).
156    pub fn flying_between(&self, from: PalaceName, to: PalaceName) -> Vec<SiHuaLanding> {
157        self.flying_from(from)
158            .landings
159            .into_iter()
160            .filter(|l| l.palace_name == to)
161            .collect()
162    }
163
164    /// Every palace that flies a transformation into `to`: `(source palace, landing)`.
165    pub fn flying_into(&self, to: PalaceName) -> Vec<(PalaceName, SiHuaLanding)> {
166        PalaceName::ALL
167            .into_iter()
168            .flat_map(|from| {
169                self.flying_between(from, to)
170                    .into_iter()
171                    .map(move |l| (from, l))
172            })
173            .collect()
174    }
175
176    /// 三方四正 of palace `name`.
177    pub fn san_fang_si_zheng(&self, name: PalaceName) -> SanFangSiZheng {
178        self.san_fang_si_zheng_at(self.palace_index_of(name))
179    }
180
181    /// 三方四正 of the palace at `palace_idx`.
182    pub fn san_fang_si_zheng_at(&self, palace_idx: PalaceIndex) -> SanFangSiZheng {
183        let p = (palace_idx % 12) as i32;
184        let opposite_idx = fix_index(p + 6);
185        let mut trine_idx = [fix_index(p + 4), fix_index(p + 8)];
186        trine_idx.sort_unstable();
187        SanFangSiZheng {
188            palace_idx: p as usize,
189            palace: self.palace_name_at(p as usize),
190            opposite_idx,
191            opposite: self.palace_name_at(opposite_idx),
192            trine_idx,
193            trine: [
194                self.palace_name_at(trine_idx[0]),
195                self.palace_name_at(trine_idx[1]),
196            ],
197        }
198    }
199
200    /// The stars standing in the 三方四正 of palace `name` (self, opposite, trines, in that order).
201    pub fn san_fang_si_zheng_stars(&self, name: PalaceName) -> Vec<SanFangStars> {
202        let group = self.san_fang_si_zheng(name);
203        group
204            .palace_indices()
205            .into_iter()
206            .map(|idx| SanFangStars {
207                palace_idx: idx,
208                palace: self.palace_name_at(idx),
209                stars: self.palaces[idx].stars.clone(),
210            })
211            .collect()
212    }
213
214    /// 大限四化: the transformations flown from the stem of a 大限 palace.
215    pub fn daxian_si_hua(&self, daxian: &DaXian) -> PalaceSiHua {
216        self.flying_from_index(daxian.palace_idx)
217    }
218
219    /// The 大限 covering a nominal age (虛歲), if any.
220    pub fn daxian_for_age(&self, age: u32) -> Option<&DaXian> {
221        self.daxian
222            .iter()
223            .find(|d| age >= d.age_start && age <= d.age_end)
224    }
225
226    /// Nominal age (虛歲) in `target_year`, counted from the lunar birth year.
227    pub fn nominal_age(&self, target_year: i32) -> u32 {
228        (target_year - self.birth_lunar.year + 1).max(1) as u32
229    }
230
231    /// 小限 for `target_year`.
232    pub fn xiaoxian(&self, target_year: i32) -> XiaoXian {
233        let age = self.nominal_age(target_year);
234        let palace_idx = xiaoxian_palace(self.birth_lunar.year_branch_idx, self.is_male, age);
235        let stem = self.palace_stem(palace_idx);
236        XiaoXian {
237            year: target_year,
238            age,
239            palace_idx,
240            palace_name: self.palace_name_at(palace_idx),
241            stem_hanja: stem.hanja().to_string(),
242            branch_hanja: branch_hanja(palace_idx),
243            si_hua: get_sihua_stars(stem),
244        }
245    }
246
247    /// 斗君 for `target_year`.
248    pub fn dou_jun(&self, target_year: i32) -> DouJun {
249        let target_branch_std = (target_year - 4).rem_euclid(12) as usize;
250        let palace_idx = dou_jun_palace(
251            target_branch_std,
252            self.birth_lunar.month,
253            self.birth_lunar.time_branch_idx,
254        );
255        let stem = self.palace_stem(palace_idx);
256        DouJun {
257            year: target_year,
258            palace_idx,
259            palace_name: self.palace_name_at(palace_idx),
260            stem_hanja: stem.hanja().to_string(),
261            branch_hanja: branch_hanja(palace_idx),
262        }
263    }
264}
265
266fn branch_hanja(palace_idx: PalaceIndex) -> String {
267    EarthlyBranch::from_index(zwds_idx_to_std_idx(palace_idx % 12) as i32)
268        .hanja()
269        .to_string()
270}

286    #[test]
287    fn natal_reference_chart_layout() {
288        let c = chart_1990();
289        assert_eq!(c.soul_idx, 10); // 子
290        assert_eq!(c.body_idx, 8); // 戌
291                                   // Palace names run counter-clockwise from 命宮 (子): 兄弟 亥, 夫妻 戌, 財帛 申, 遷移 午, 官祿 辰.
292        assert_eq!(c.palace_index_of(PalaceName::Xiongdi), 9);
293        assert_eq!(c.palace_index_of(PalaceName::Fuqi), 8);
294        assert_eq!(c.palace_index_of(PalaceName::Caibo), 6);
295        assert_eq!(c.palace_index_of(PalaceName::Qianyi), 4);
296        assert_eq!(c.palace_index_of(PalaceName::Guanlu), 2);
297        assert_eq!(c.palace_index_of(PalaceName::Fumu), 11);
298        // The 身宮 (戌) falls in the 夫妻宮, one of the six palaces it may legally occupy.
299        assert_eq!(c.palaces[c.body_idx].name, PalaceName::Fuqi);
300        // 12 gods are attached to every palace.
301        assert!(c.palaces.iter().all(|p| p.twelve_gods.is_some()));
302        // Fire6 → 長生 in 寅, forward (陽男): 子 is 胎 and 亥 is 絕.
303        let cs = |i: usize| c.palaces[i].twelve_gods.unwrap().chang_sheng;
304        assert_eq!(cs(0), crate::twelve_gods::ChangSheng::ChangSheng);
305        assert_eq!(cs(10), crate::twelve_gods::ChangSheng::Tai);
306        assert_eq!(cs(9), crate::twelve_gods::ChangSheng::Jue);
307        // 博士 starts at 祿存 (申).
308        assert_eq!(
309            c.palaces[c.star_palace(ZwdsStar::LuCun).unwrap()]
310                .twelve_gods
311                .unwrap()
312                .bo_shi,
313            crate::twelve_gods::BoShi::BoShi
314        );
315    }

332    #[test]
333    fn flying_queries_match_natal_list() {
334        let c = chart_1990();
335        // 命宮 stem is 戊 (子 palace): 戊 → 貪狼祿 太陰權 右弼科 天機忌.
336        let ming = c.flying_from(PalaceName::Ming);
337        assert_eq!(ming.stem_hanja, "戊");
338        let star_of = |t: SiHuaType| ming.landings.iter().find(|l| l.sihua_type == t).unwrap();
339        assert_eq!(star_of(SiHuaType::HuaLu).star, ZwdsStar::TanLang);
340        assert_eq!(star_of(SiHuaType::HuaJi).star, ZwdsStar::TianJi);
341        // The flat list and the query agree for every palace and every transformation.
342        for from in PalaceName::ALL {
343            let q = c.flying_from(from);
344            for l in &q.landings {
345                assert!(c.flying_sihua.iter().any(|f| f.from_palace == from
346                    && f.to_palace == l.palace_name
347                    && f.sihua_type == l.sihua_type
348                    && f.star == l.star));
349            }
350            assert_eq!(q.landings.len(), 4);
351        }
352        // flying_into is the transpose of flying_between.
353        let into = c.flying_into(PalaceName::Fuqi);
354        for (from, l) in &into {
355            assert!(c.flying_between(*from, PalaceName::Fuqi).contains(l));
356        }
357    }

375    #[test]
376    fn xiaoxian_and_dou_jun_for_reference_chart() {
377        let c = chart_1990();
378        // 2026: nominal age 37; 午 year male → 小限 age 1 in 辰, 37 − 1 = 36 ≡ 0 → back in 辰 (2).
379        let x = c.xiaoxian(2026);
380        assert_eq!(x.age, 37);
381        assert_eq!(x.palace_idx, 2);
382        assert_eq!(x.branch_hanja, "辰");
383        assert_eq!(x.stem_hanja, "庚"); // 庚午 year: 寅 = 戊, so 辰 = 庚
384        assert_eq!(x.palace_name, PalaceName::Guanlu);
385        // 2026 丙午, lunar month 4, 巳 hour → 斗君 in 申 (6): 午 (4) − 3 + 5.
386        let d = c.dou_jun(2026);
387        assert_eq!(d.palace_idx, 6);
388        assert_eq!(d.branch_hanja, "申");
389        assert_eq!(d.palace_name, PalaceName::Caibo);

392    #[test]
393    fn female_chart_runs_xiaoxian_backwards() {
394        // 甲子 (1984) lunar 6/1, 戌 hour (std idx 10 → hour 20), female.
395        let lunar = lunar_to_birth_info(1984, 6, 1, false, 20);
396        let birth = BirthInfo::solar(1984, 7, 1, 20, 0).with_gender(Gender::Female);
397        let c = build_chart_from_lunar(&lunar, &birth).unwrap();
398        assert!(!c.is_male);
399        // 金四局 with Ming in 酉 (hand-derived in the twelve_gods tests): 陽女 → 逆行.
400        assert_eq!(c.soul_idx, 7);
401        assert_eq!(c.five_elements, crate::types::FiveElementsClass::Metal4);
402        // 申子辰 → 小限 age 1 in 戌 (8); female age 43 (2026) → 辰 (2).
403        assert_eq!(c.xiaoxian(2026).age, 43);
404        assert_eq!(c.xiaoxian(2026).palace_idx, 2);
405        // Direction of 大限: 陽女 逆行 → second decade in 申 (idx 6 = 7 − 1).
406        assert_eq!(c.daxian[1].palace_idx, 6);
407    }
