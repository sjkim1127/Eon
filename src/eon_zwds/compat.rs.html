1//! 合盤 (two-chart comparison) for Zi Wei Dou Shu.
2//!
3//! **This is a heuristic**, not a classical verdict: there is no single canonical 合盤 score. The
4//! module reads two things from a pair of charts:
5//!
6//! 1. 飛星 across charts: the stems of each person's 命宮 and 夫妻宮 are flown into the *other*
7//!    person's chart (each stem gives 化祿·化權·化科·化忌, each landing on the palace that holds the
8//!    transformed star in the other chart). This is the usual 合盤 technique ("A 的命宮化忌入 B 的
9//!    夫妻宮").
10//! 2. Star overlaps: stars that stand in both people's 命宮/夫妻宮 (and across 命宮 ↔ 夫妻宮).
11//!
12//! Scoring (`harmony_score`, 0–100): every flight contributes `type_weight × palace_weight`
13//! to either the *supportive* or the *tense* side, and the score is the balance
14//! `50 + 50 × (supportive − tense) / (supportive + tense)` (50 when nothing lands).
15//!
16//! | Transformation | Side | `type_weight` |
17//! |----------------|------|---------------|
18//! | 化祿 | supportive | 1.0 |
19//! | 化科 | supportive | 0.75 |
20//! | 化權 | supportive | 0.5 |
21//! | 化忌 | tense | 1.5 |
22//!
23//! | Landing palace | `palace_weight` |
24//! |----------------|-----------------|
25//! | 命宮, 夫妻宮 | 3.0 |
26//! | 福德宮 | 2.0 |
27//! | any other | 1.0 |
28//!
29//! Star overlaps are reported but do not change the score.
30
31use crate::types::{PalaceName, SiHuaType, ZwdsChart, ZwdsStar};
32use serde::{Deserialize, Serialize};

34/// Which person of the pair.
35#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
36pub enum Person {
37    A,
38    B,
39}

41/// One transformation flown from a palace stem of one chart into the other chart.
42#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
43pub struct CrossSiHuaHit {
44    /// Person whose stem is flying.
45    pub from_person: Person,
46    /// Palace (of `from_person`'s chart) whose stem flies.
47    pub from_palace: PalaceName,
48    /// Heavenly stem of that palace.
49    pub stem_hanja: String,
50    pub sihua_type: SiHuaType,
51    pub star: ZwdsStar,
52    /// Palace of the *other* person's chart where the transformed star stands.
53    pub to_palace: PalaceName,
54    /// Signed contribution to the score: positive = supportive, negative = tense.
55    pub contribution: f64,
56}
57
58/// Stars standing in both of two compared palaces.
59#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
60pub struct StarOverlap {
61    pub palace_a: PalaceName,
62    pub palace_b: PalaceName,
63    pub shared_stars: Vec<ZwdsStar>,
64}

66/// Result of [`compare_charts`].
67#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
68pub struct ZwdsCompatibility {
69    /// 命宮 / 夫妻宮 stems of A flown into B's chart.
70    pub a_to_b: Vec<CrossSiHuaHit>,
71    /// 命宮 / 夫妻宮 stems of B flown into A's chart.
72    pub b_to_a: Vec<CrossSiHuaHit>,
73    /// Shared stars in 命宮 ↔ 命宮, 夫妻宮 ↔ 夫妻宮, A 命宮 ↔ B 夫妻宮 and A 夫妻宮 ↔ B 命宮.
74    pub overlaps: Vec<StarOverlap>,
75    /// Sum of the supportive contributions.
76    pub supportive: f64,
77    /// Sum of the tense contributions (positive number).
78    pub tense: f64,
79    /// 0–100 balance of supportive versus tense flights (heuristic, see the module docs).
80    pub harmony_score: f64,
81    pub summary: String,
82    /// Reminder that the score is a heuristic.
83    pub method_note: String,
84}
85
86/// Palaces whose stems are flown across the charts.
87pub const SOURCE_PALACES: [PalaceName; 2] = [PalaceName::Ming, PalaceName::Fuqi];
88
89fn type_weight(t: SiHuaType) -> f64 {
90    match t {
91        SiHuaType::HuaLu => 1.0,
92        SiHuaType::HuaKe => 0.75,
93        SiHuaType::HuaQuan => 0.5,
94        SiHuaType::HuaJi => -1.5,
95    }
96}
97
98fn palace_weight(p: PalaceName) -> f64 {
99    match p {
100        PalaceName::Ming | PalaceName::Fuqi => 3.0,
101        PalaceName::Fude => 2.0,
102        _ => 1.0,
103    }
104}

106/// Fly the stems of `SOURCE_PALACES` of `from` into `to`.
107pub fn cross_si_hua(from: &ZwdsChart, person: Person, to: &ZwdsChart) -> Vec<CrossSiHuaHit> {
108    let mut hits = Vec::new();
109    for palace in SOURCE_PALACES {
110        let idx = from.palace_index_of(palace);
111        let stem = from.palace_stem(idx);
112        for landing in to.si_hua_landings_for_stem(stem) {
113            hits.push(CrossSiHuaHit {
114                from_person: person,
115                from_palace: palace,
116                stem_hanja: stem.hanja().to_string(),
117                sihua_type: landing.sihua_type,
118                star: landing.star,
119                to_palace: landing.palace_name,
120                contribution: type_weight(landing.sihua_type) * palace_weight(landing.palace_name),
121            });
122        }
123    }
124    hits
125}

127fn overlap(
128    a: &ZwdsChart,
129    palace_a: PalaceName,
130    b: &ZwdsChart,
131    palace_b: PalaceName,
132) -> StarOverlap {
133    let stars_b: Vec<ZwdsStar> = b.palace(palace_b).stars.iter().map(|s| s.star).collect();
134    let shared_stars = a
135        .palace(palace_a)
136        .stars
137        .iter()
138        .map(|s| s.star)
139        .filter(|s| stars_b.contains(s))
140        .collect();
141    StarOverlap {
142        palace_a,
143        palace_b,
144        shared_stars,
145    }
146}

148/// Compare two charts (see the module docs for the method and its limits).
149pub fn compare_charts(a: &ZwdsChart, b: &ZwdsChart) -> ZwdsCompatibility {
150    let a_to_b = cross_si_hua(a, Person::A, b);
151    let b_to_a = cross_si_hua(b, Person::B, a);
152
153    let overlaps = vec![
154        overlap(a, PalaceName::Ming, b, PalaceName::Ming),
155        overlap(a, PalaceName::Fuqi, b, PalaceName::Fuqi),
156        overlap(a, PalaceName::Ming, b, PalaceName::Fuqi),
157        overlap(a, PalaceName::Fuqi, b, PalaceName::Ming),
158    ];
159
160    let (mut supportive, mut tense) = (0.0, 0.0);
161    for hit in a_to_b.iter().chain(&b_to_a) {
162        if hit.contribution >= 0.0 {
163            supportive += hit.contribution;
164        } else {
165            tense -= hit.contribution;
166        }
167    }
168    let total = supportive + tense;
169    let harmony_score = if total > 0.0 {
170        50.0 + 50.0 * (supportive - tense) / total
171    } else {
172        50.0
173    };
174
175    let mut strongest: Vec<&CrossSiHuaHit> = a_to_b.iter().chain(&b_to_a).collect();
176    strongest.sort_by(|x, y| y.contribution.abs().total_cmp(&x.contribution.abs()));
177    let mut parts: Vec<String> = strongest
178        .iter()
179        .take(3)
180        .map(|h| {
181            format!(
182                "{}'s {} stem sends {} ({}) into {}'s {}",
183                person_label(h.from_person),
184                h.from_palace.hanja(),
185                h.sihua_type.emoji(),
186                h.star.hanja(),
187                person_label(match h.from_person {
188                    Person::A => Person::B,
189                    Person::B => Person::A,
190                }),
191                h.to_palace.hanja(),
192            )
193        })
194        .collect();
195    if parts.is_empty() {
196        parts.push("no cross-chart transformations landed".to_string());
197    }
198    let shared: usize = overlaps.iter().map(|o| o.shared_stars.len()).sum();
199    let summary = format!(
200        "Harmony {:.1}/100 (supportive {:.2} vs tense {:.2}); strongest flights: {}; {} shared star(s) across 命宮/夫妻宮.",
201        harmony_score,
202        supportive,
203        tense,
204        parts.join("; "),
205        shared
206    );
207
208    ZwdsCompatibility {
209        a_to_b,
210        b_to_a,
211        overlaps,
212        supportive,
213        tense,
214        harmony_score,
215        summary,
216        method_note: "Heuristic: balance of weighted cross-chart 四化 flights from the 命宮 and 夫妻宮 stems; star overlaps are informational. Not a classical verdict.".to_string(),
217    }
218}

223        Person::B => "B",

227#[cfg(test)]
228mod tests {
229    use super::*;
230    use crate::calendar::lunar_to_birth_info;
231    use crate::chart::build_chart_from_lunar;
232    use eon_core::birth::{BirthInfo, Gender};
233
234    fn chart(year: i32, month: u32, day: u32, hour: u32, gender: Gender) -> ZwdsChart {
235        let lunar = lunar_to_birth_info(year, month, day, false, hour);
236        let birth = BirthInfo::solar(year, 1, 1, hour, 0).with_gender(gender);
237        build_chart_from_lunar(&lunar, &birth).unwrap()
238    }
239
240    #[test]
241    fn cross_flights_land_where_the_other_charts_stars_stand() {
242        let a = chart(1990, 4, 21, 10, Gender::Male);
243        let b = chart(1984, 6, 1, 20, Gender::Female);
244        let r = compare_charts(&a, &b);
245        // 2 source palaces × 4 transformations, each direction.
246        assert_eq!(r.a_to_b.len(), 8);
247        assert_eq!(r.b_to_a.len(), 8);
248        // A's 命宮 stem is 戊 (子 palace of a 庚 year): 化祿 貪狼. It must land in the palace
249        // where B's chart holds 貪狼.
250        let lu = r
251            .a_to_b
252            .iter()
253            .find(|h| h.from_palace == PalaceName::Ming && h.sihua_type == SiHuaType::HuaLu)
254            .unwrap();
255        assert_eq!(lu.stem_hanja, "戊");
256        assert_eq!(lu.star, ZwdsStar::TanLang);
257        let tan_lang_in_b = b.star_palace(ZwdsStar::TanLang).unwrap();
258        assert_eq!(b.palace_name_at(tan_lang_in_b), lu.to_palace);
259    }
260
261    #[test]
262    fn score_is_a_bounded_balance() {
263        let a = chart(1990, 4, 21, 10, Gender::Male);
264        let b = chart(1984, 6, 1, 20, Gender::Female);
265        let r = compare_charts(&a, &b);
266        assert!((0.0..=100.0).contains(&r.harmony_score));
267        let expected = 50.0 + 50.0 * (r.supportive - r.tense) / (r.supportive + r.tense);
268        assert!((r.harmony_score - expected).abs() < 1e-9);
269        // Deterministic: the same pair yields the identical result.
270        assert_eq!(r, compare_charts(&a, &b));
271        // A chart compared with itself has overlapping stars in 命宮 ↔ 命宮.
272        let same = compare_charts(&a, &a);
273        assert_eq!(
274            same.overlaps[0].shared_stars.len(),
275            a.palace(PalaceName::Ming).stars.len()
276        );
277    }
278
279    #[test]
280    fn contribution_signs_follow_the_transformation() {
281        let a = chart(1990, 4, 21, 10, Gender::Male);
282        let b = chart(2001, 11, 3, 4, Gender::Female);
283        let r = compare_charts(&a, &b);
284        for h in r.a_to_b.iter().chain(&r.b_to_a) {
285            match h.sihua_type {
286                SiHuaType::HuaJi => assert!(h.contribution < 0.0),
287                _ => assert!(h.contribution > 0.0),
288            }
289        }
290    }
