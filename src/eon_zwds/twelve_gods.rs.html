1//! The four "twelve-god" star groups of Zi Wei Dou Shu (十二神煞).
2//!
3//! Each group occupies all twelve palaces, one god per palace:
4//!
5//! | Group | Starts at | Direction |
6//! |-------|-----------|-----------|
7//! | 長生十二神 | by five-element bureau (水二/土五 → 申, 木三 → 亥, 金四 → 巳, 火六 → 寅) | 陽男陰女 順行, 陰男陽女 逆行 |
8//! | 博士十二神 | the palace of 祿存 | 陽男陰女 順行, 陰男陽女 逆行 |
9//! | 將前十二神 | 將星 by year branch (寅午戌 → 午, 申子辰 → 子, 巳酉丑 → 酉, 亥卯未 → 卯) | always 順行 |
10//! | 歲前十二神 | 歲建 in the palace of the year branch | always 順行 |
11//!
12//! 順行 means following the branch order (寅 → 卯 → 辰 ...), i.e. increasing [`PalaceIndex`].
13//!
14//! Sources: 《紫微斗數全書》 安長生十二神訣, 安博士十二神訣, 安將前諸星訣, 安歲前諸星訣; cross-checked
15//! against iztro (`src/star/decorativeStar.ts`, `getchangsheng12`, `getBoShi12`, `getYearly12`).
16//! Unlike the 14 major stars these gods are kept apart from [`crate::PalaceData::stars`]: they are a
17//! per-palace annotation ([`PalaceTwelveGods`]) and never take part in formation or 四化 logic.
18

23macro_rules! twelve_gods_enum {
24    (
25        $(#[$meta:meta])*
26        $name:ident { $( $variant:ident => ($hanja:literal, $english:literal) ),+ $(,)? }
27    ) => {
28        $(#[$meta])*
29        #[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
30        pub enum $name { $( $variant ),+ }
31
32        impl $name {
33            /// The twelve gods in their canonical order (the order they are laid out along the palaces).
34            pub const ALL: [$name; 12] = [ $( $name::$variant ),+ ];
35
36            /// Chinese name.
37            pub fn hanja(&self) -> &'static str {
38                match self { $( Self::$variant => $hanja ),+ }
39            }
40
41            /// Pinyin / English name.
42            pub fn english(&self) -> &'static str {
43                match self { $( Self::$variant => $english ),+ }
44            }
45
46            /// Same as [`Self::english`] (the project convention for the `korean()` accessors).
47            pub fn korean(&self) -> &'static str {
48                self.english()
49            }
50
51            /// Position of this god in the canonical order (0 = first god of the group).
52            pub fn order(&self) -> usize {
53                Self::ALL.iter().position(|g| g == self).unwrap_or(0)
54            }
55        }
56    };
57}

131/// The four twelve-god groups as they fall on one palace.
132#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
133pub struct PalaceTwelveGods {
134    /// 長生十二神
135    pub chang_sheng: ChangSheng,
136    /// 博士十二神
137    pub bo_shi: BoShi,
138    /// 將前十二神
139    pub jiang_qian: JiangQian,
140    /// 歲前十二神
141    pub sui_qian: SuiQian,
142}
143
144/// Palace (ZWDS index, 寅 = 0) where 長生 starts for a five-element bureau.
145///
146/// 水二局·土五局長生在申, 木三局長生在亥, 金四局長生在巳, 火六局長生在寅.
147pub fn chang_sheng_start(bureau: FiveElementsClass) -> PalaceIndex {
148    match bureau {
149        FiveElementsClass::Water2 | FiveElementsClass::Earth5 => 6, // 申
150        FiveElementsClass::Wood3 => 9,                              // 亥
151        FiveElementsClass::Metal4 => 3,                             // 巳
152        FiveElementsClass::Fire6 => 0,                              // 寅
153    }
154}
155
156/// Palace (ZWDS index) where 將星 sits for a year branch (standard index, 子 = 0).
157///
158/// 寅午戌年將星午, 申子辰年將星子, 巳酉丑年將星酉, 亥卯未年將星卯.
159pub fn jiang_xing_start(year_branch_std_idx: usize) -> PalaceIndex {
160    let branch_std = match year_branch_std_idx % 12 {
161        2 | 6 | 10 => 6, // 寅午戌 → 午
162        8 | 0 | 4 => 0,  // 申子辰 → 子
163        5 | 9 | 1 => 9,  // 巳酉丑 → 酉
164        _ => 3,          // 亥卯未 → 卯
165    };
166    std_idx_to_zwds_idx(branch_std)
167}

169/// Lay the four groups over the twelve palaces.
170///
171/// * `bureau` - five-element bureau (decides where 長生 starts)
172/// * `forward` - `true` for 陽男·陰女 (see [`crate::decadal::is_forward_direction`])
173/// * `lucun_idx` - palace of 祿存 (博士 starts there)
174/// * `year_branch_std_idx` - birth-year branch, standard index (子 = 0)
175///
176/// The returned array is indexed by palace (ZWDS index, 寅 = 0).
177pub fn place_twelve_gods(
178    bureau: FiveElementsClass,
179    forward: bool,
180    lucun_idx: PalaceIndex,
181    year_branch_std_idx: usize,
182) -> [PalaceTwelveGods; 12] {
183    let dir: i32 = if forward { 1 } else { -1 };
184    let cs_start = chang_sheng_start(bureau) as i32;
185    let bs_start = lucun_idx as i32;
186    let jq_start = jiang_xing_start(year_branch_std_idx) as i32;
187    let sq_start = std_idx_to_zwds_idx(year_branch_std_idx % 12) as i32;
188
189    std::array::from_fn(|p| {
190        let p = p as i32;
191        // Offset (in palaces, following `dir`) from the starting palace of each directional group.
192        let along = |start: i32| (dir * (p - start)).rem_euclid(12) as usize;
193        // 將前 / 歲前 always follow the branch order.
194        let fixed = |start: i32| fix_index(p - start);
195        PalaceTwelveGods {
196            chang_sheng: ChangSheng::ALL[along(cs_start)],
197            bo_shi: BoShi::ALL[along(bs_start)],
198            jiang_qian: JiangQian::ALL[fixed(jq_start)],
199            sui_qian: SuiQian::ALL[fixed(sq_start)],
200        }
201    })
202}

364    #[test]
365    fn every_group_covers_all_twelve_palaces_once() {
366        for forward in [true, false] {
367            let gods = place_twelve_gods(FiveElementsClass::Wood3, forward, ZI, 5);
368            for group in 0..4 {
369                let mut seen = [false; 12];
370                for g in &gods {
371                    let order = match group {
372                        0 => g.chang_sheng.order(),
373                        1 => g.bo_shi.order(),
374                        2 => g.jiang_qian.order(),
375                        _ => g.sui_qian.order(),
376                    };
377                    assert!(!seen[order], "duplicate god in group {group}");
378                    seen[order] = true;
379                }
380            }
381        }
382    }
