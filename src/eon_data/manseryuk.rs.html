1//! Manseryuk (Ten-Thousand Year Calendar) astronomical data.

3//! Provides daily Ganzi lookup and 24 solar term timing calculations.

5use crate::cache::GlobalCache;
6use chrono::{DateTime, Datelike, NaiveDate, TimeZone, Utc};

8/// Retrieve the 60-Ganzi index of a specific date.
9///
10/// Computed using Julian Day Number (JDN).
11pub fn get_day_ganzi_index(date: NaiveDate) -> u8 {
12    let jdn = gregorian_to_jdn(date.year(), date.month() as i32, date.day() as i32);
13    // 2000-01-01 = JDN 2451545 = Wu-Wu day (index=54)
14    ((jdn + 49) % 60) as u8
15}
16
17/// Retrieve the date and time of a solar term for a given year.
18///
19/// Checks the binary cache first; falls back to real-time astronomical calculation via eon-astro.
20pub fn get_solar_term_datetime(
21    year: i32,
22    term_index: u8,
23) -> Result<DateTime<Utc>, crate::error::DataError> {
24    // 1. Query the binary cache first
25    if let Some(dt) = GlobalCache::get_solar_term(year, term_index) {
26        return Ok(dt);
27    }
28
29    // 2. Fall back to real-time astronomical calculation with eon-astro
30    let engine = eon_astro::AstroEngine::new();
31
32    // Approximate solar term position (Lichun = 315 deg)
33    // 24 solar terms are spaced by ~15 days starting in early February (Lichun).
34    // Start search window around Feb 1.
35    let approx_start = Utc
36        .with_ymd_and_hms(year, 2, 1, 0, 0, 0)
37        .single()
38        .ok_or(crate::error::DataError::InvalidDate)?;
39
40    match engine.find_solar_term_time(approx_start, term_index) {
41        Ok(dt) => Ok(dt),
42        Err(_) => {
43            // Ultimate fallback approximation if astronomical engine fails
44            // Lichun (index 0) is approximately Feb 4
45            // Each solar term/sub-term occurs at roughly 15.22 day intervals
46            let base_date = Utc
47                .with_ymd_and_hms(year, 2, 4, 0, 0, 0)
48                .single()
49                .ok_or(crate::error::DataError::InvalidDate)?;
50            let offset_days = (term_index as f64 * 15.218).round() as i64;
51            Ok(base_date + chrono::Duration::days(offset_days))
52        }
53    }
54}

56/// Convert Gregorian date to Julian Day Number (JDN).
57fn gregorian_to_jdn(year: i32, month: i32, day: i32) -> i64 {
58    let a = (14 - month) / 12;
59    let y = year + 4800 - a;
60    let m = month + 12 * a - 3;
61
62    let jdn = day + (153 * m + 2) / 5 + 365 * y + y / 4 - y / 100 + y / 400 - 32045;
63    jdn as i64
64}

66/// Lunar-Solar calendar conversion engine.
67pub struct LunarCalendar;

69impl LunarCalendar {
70    /// Convert solar date to lunar date (KST-based calculation) with cache lookup.
71    pub fn from_solar(date: NaiveDate) -> Result<(i32, u32, u32, bool), crate::error::DataError> {
72        // 1. Check cache first
73        if let Some(res) = GlobalCache::get_lunar_date(date) {
74            return Ok(res);
75        }
76
77        // 2. Real-time astronomical computation
78        Self::from_solar_internal(date)
79    }
80
81    /// Convert solar date to lunar date using real-time astronomical calculation.
82    pub fn from_solar_internal(
83        date: NaiveDate,
84    ) -> Result<(i32, u32, u32, bool), crate::error::DataError> {
85        let engine = eon_astro::AstroEngine::new();
86        let kst_offset = chrono::Duration::hours(9);
87
88        // 1. Convert end-of-day in KST to UTC
89        let dt_utc = Utc
90            .with_ymd_and_hms(date.year(), date.month(), date.day(), 23, 59, 59)
91            .single()
92            .ok_or(crate::error::DataError::InvalidDate)?
93            - kst_offset;
94
95        // 2. Find preceding New Moon time
96        let nm_t = engine.find_new_moon_before(dt_utc)?;
97        let nm_date_kst = (nm_t + kst_offset).date_naive();
98        let lunar_day = (date - nm_date_kst).num_days() as u32 + 1;
99
100        // 3. Find reference Winter Solstice to determine month and year alignment
101        // Find Winter Solstice before or at the current new moon
102        let ws_t = engine.find_winter_solstice_before(nm_t + chrono::Duration::days(1))?;
103        let ws_nm_t = engine.find_new_moon_before(ws_t + chrono::Duration::minutes(1))?;
104
105        // Determine if this sui (cycle between Winter Solstices) has 13 lunar months (contains a leap month)
106        let next_ws_t = engine.find_winter_solstice_before(ws_t + chrono::Duration::days(380))?;
107        let next_ws_nm_t = engine.find_new_moon_before(next_ws_t + chrono::Duration::minutes(1))?;
108        let sui_months = ((next_ws_nm_t - ws_nm_t).num_days() as f64 / 29.53).round() as i32;
109        let has_leap_year = sui_months >= 13;
110
111        // The month containing the Winter Solstice is always lunar Month 11
112        let mut curr_nm = ws_nm_t;
113        let mut month_num = 11;
114        let mut is_leap = false;
115        let mut leap_placed = false;
116        let mut lunar_year = (ws_t + kst_offset).year();
117
118        // 4. Step forward month by month from Month 11 to current New Moon
119        while (nm_t - curr_nm).num_minutes() > 1440 {
120            let next_nm = engine.find_new_moon_before(curr_nm + chrono::Duration::days(45))?;
121            let next_next_nm = engine.find_new_moon_before(next_nm + chrono::Duration::days(45))?;
122
123            // Evaluate the upcoming month [next_nm, next_next_nm)
124            let mut has_zq = false;
125            let month_start_kst = (next_nm + kst_offset).date_naive();
126            let month_end_kst = (next_next_nm + kst_offset).date_naive();
127
128            if let Some(start_dt) = Utc
129                .with_ymd_and_hms(
130                    month_start_kst.year(),
131                    month_start_kst.month(),
132                    month_start_kst.day(),
133                    0,
134                    0,
135                    0,
136                )
137                .single()
138            {
139                let search_start_utc = start_dt - kst_offset;
140                for i in 1..=23 {
141                    if i % 2 == 1 {
142                        // Zhong-qi odd indices (1, 3, 5, ..., 23)
143                        if let Ok(zq_t) = engine.find_solar_term_time(search_start_utc, i) {
144                            let zq_date = (zq_t + kst_offset).date_naive();
145                            if zq_date >= month_start_kst && zq_date < month_end_kst {
146                                has_zq = true;
147                                break;
148                            }
149                        }
150                    }
151                }
152            }
153
154            if has_leap_year && !leap_placed && !has_zq {
155                // Month without Zhong-qi in a leap sui becomes the leap month
156                is_leap = true;
157                leap_placed = true;
158            } else {
159                month_num += 1;
160                if month_num > 12 {
161                    month_num = 1;
162                    lunar_year += 1;
163                }
164                is_leap = false;
165            }
166            curr_nm = next_nm;
167        }
168
169        Ok((lunar_year, month_num as u32, lunar_day, is_leap))
170    }

172    /// Convert lunar date to solar date.
173    pub fn to_solar(
174        year: i32,
175        month: u32,
176        day: u32,
177        is_leap: bool,
178    ) -> Result<NaiveDate, crate::error::DataError> {
179        let test_date =
180            NaiveDate::from_ymd_opt(year, month, 1).ok_or(crate::error::DataError::InvalidDate)?;
181
182        // Scan across a +/- 6 month window (Lunar month 1 corresponds to ~Feb)
183        for i in -60..360 {
184            let date = test_date + chrono::Duration::days(i);
185            if let Ok((ly, lm, ld, il)) = Self::from_solar(date) {
186                if ly == year && lm == month && ld == day && il == is_leap {
187                    return Ok(date);
188                }
189            }
190        }
191        Err(crate::error::DataError::InvalidDate)
192    }
193}

199    #[test]
200    fn test_get_day_ganzi_index() {
201        // 2000-01-01 = Wu-Wu day (index=54)
202        let date = NaiveDate::from_ymd_opt(2000, 1, 1).unwrap();
203        assert_eq!(get_day_ganzi_index(date), 54);
204
205        // 2000-01-02 = Ji-Wei day (index=55)
206        let date = NaiveDate::from_ymd_opt(2000, 1, 2).unwrap();
207        assert_eq!(get_day_ganzi_index(date), 55);
208    }
209
210    #[test]
211    fn test_sixty_day_cycle() {
212        // Same Ganzi cycle repeats after 60 days
213        let date1 = NaiveDate::from_ymd_opt(2024, 1, 1).unwrap();
214        let date2 = NaiveDate::from_ymd_opt(2024, 3, 1).unwrap(); // 60 days later
215        assert_eq!(get_day_ganzi_index(date1), get_day_ganzi_index(date2));
216    }
217
218    #[test]
219    fn test_lunar_conversion() {
220        // Reference birthdate: Solar 2004-11-27 -> Lunar 2004-10-16
221        let solar = NaiveDate::from_ymd_opt(2004, 11, 27).unwrap();
222        let lunar = LunarCalendar::from_solar(solar).unwrap();
223        println!("DEBUG: Lunar result for 2004-11-27: {:?}", lunar);
224
225        assert_eq!(lunar.0, 2004, "Year should be 2004");
226        assert_eq!(lunar.1, 10, "Month should be 10");
227        assert_eq!(lunar.2, 16, "Day should be 16");
228        assert!(!lunar.3, "Should not be a leap month");
229
230        // Verify inverse conversion
231        let solar_back = LunarCalendar::to_solar(2004, 10, 16, false).unwrap();
232        assert_eq!(solar_back, solar);
233
234        // KASI reference: 2023-03-22 is Lunar 2023 Leap Month 2 Day 1 (윤2월 1일)
235        let leap_date = NaiveDate::from_ymd_opt(2023, 3, 22).unwrap();
236        let lunar_leap = LunarCalendar::from_solar_internal(leap_date).unwrap();
237        println!("DEBUG: Lunar result for 2023-03-22: {:?}", lunar_leap);
238        assert_eq!(lunar_leap, (2023, 2, 1, true));
239    }
