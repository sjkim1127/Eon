3pub fn generate_synthesis(
4    day_master_name: Option<&str>,
5    sun_sign_name: Option<&str>,
6    hd_type: Option<&str>,
7    numerology_lp: Option<u8>,

9    is_seven_killings_or_intense: bool,
10    has_wealth_yoga_or_structure: bool,

12    String,
13    String,
14    Vec<CrossEngineSynthesisGuideline>,

18    let dm_clean = day_master_name.unwrap_or("Unknown");
19    let sun_clean = sun_sign_name.unwrap_or("Unknown");
20    let hd_clean = hd_type.unwrap_or("Unknown");
21    let lp_val = numerology_lp.unwrap_or(1);
22    let py_val = numerology_py.unwrap_or(1);
23
24    // 1. Archetypal Persona Determination
25    let persona_title = if dm_clean.contains("Jia")
26        || dm_clean.contains("Geng")
27        || lp_val == 1
28        || lp_val == 8
29        || hd_clean.to_lowercase().contains("manifestor")
30    {
31        "The Sovereign Architect (주권적 설계자 & 문명 개척자)"

35        || lp_val == 5
36        || sun_clean.contains("Leo")
37        || sun_clean.contains("Aries")
38    {
39        "The Radiant Catalyst (빛의 혁신가 & 패러다임 촉매자)"
40    } else if dm_clean.contains("Ren")
41        || dm_clean.contains("Gui")
42        || lp_val == 7
43        || lp_val == 11
44        || hd_clean.to_lowercase().contains("projector")
45    {
46        "The Intuitive Strategist (직관적 전략가 & 심연의 네비게이터)"

50        "The Harmonizing Alchemist (조화의 연금술사 & 보편적 가치 전달자)"
51    };

53    let archetypal_persona = format!(
54        "{} [Day Master: {} | Sun: {} | Type: {} | LP: {}]",
55        persona_title, dm_clean, sun_clean, hd_clean, lp_val
56    );
57

59    let core_essence_summary = format!(
60        "본 차트는 동양의 음양오행 및 사주명리 격국 체계, 베딕 점성학의 카르마적 요가, 서양 점성학의 행성 원형 각도, 그리고 휴먼디자인 및 피타고라스 수비학이 정밀하게 일치하는 다차원 구조를 지닙니다. 일간(Day Master) '{}'의 기운과 서양 태양 '{}', 휴먼디자인 '{}' 타입 및 수비학 생애여정수 '{}'이 융합되어, 타인의 압박에 종속되지 않고 스스로 거버넌스와 실행 시스템을 장악할 때 폭발적인 현실적 권능을 발휘하는 천부적 리더십 구조를 형성하고 있습니다.",

69            strategic_action: "단기적 잔파도에 일희일비하지 말고, 복리 효과를 창출하는 독점적 플랫폼이나 핵심 IP(지적재산권) 시스템을 선제 구축하십시오. 전문 인력과 자본을 통솔하는 거버넌스 확립에 집중해야 합니다.".to_string(),
70        }
71    } else {

73            domain: "Career & Wealth Architecture (커리어 및 자본 증식)".to_string(),

75            strategic_action: "기술이나 전문 연구 성과를 시장의 유통 채널(재성)과 직접 연결할 수 있는 공식 비즈니스 파이프라인을 구축하십시오. 전문성을 가진 파트너와의 분업이 성공의 열쇠입니다.".to_string(),
