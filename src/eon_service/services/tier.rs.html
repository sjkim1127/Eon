39        score: 50.0,

41    });
42    let pot_res = compute_potential_score(saju, vedic);
43
44    let current_age = transit.map(|t| t.current_age).unwrap_or(30);
45
46    let detailed_components = compute_omni_detailed_components(

48        vedic,

496fn clamp_score(score: f32) -> f32 {
497    score.clamp(0.0, 100.0)

500/// Deterministic Spread Normalization
501/// Expands the score distribution to cover the full 0-100 range more effectively.
502fn spread_normalize(score: f32) -> f32 {
503    let s = clamp_score(score);
504    // Sigmoid-like stretch around 50 to pull clusters away from the middle
505    let normalized = if s < 50.0 {
506        // [0, 50] -> [0, 50] with downward curve
507        50.0 * (s / 50.0).powf(1.4)
508    } else {
509        // [50, 100] -> [50, 100] with upward curve
510        50.0 + 50.0 * ((s - 50.0) / 50.0).powf(0.7) // Squareroot-like to boost high-mid scores
511    };
512    normalized.round()
513}

1041/// Returns a favorability score (0..100) for the Dasha period active at a given age
1042fn compute_dasha_bonus_for_age(
1043    timeline: &[eon_vedic::analysis::dasha::DashaPeriod],
1044    _age: u32,
1045) -> f32 {
1046    // Try to find the Dasha period that covers this age by checking year offsets
1047    for (i, period) in timeline.iter().enumerate() {
1048        let start_year = period
1049            .start_time
1050            .format("%Y")
1051            .to_string()
1052            .parse::<i32>()
1053            .unwrap_or(0);
1054        let end_year = period
1055            .end_time
1056            .format("%Y")
1057            .to_string()
1058            .parse::<i32>()
1059            .unwrap_or(0);
1060        let _duration_years = end_year - start_year;
1061
1062        // Use the period's is_favorable flag if available
1063        if let Some(favorable) = period.is_favorable {
1064            if favorable {
1065                // Stagger scores slightly by period index for variety
1066                return (75.0 + (i as f32 * 2.0) % 20.0).min(95.0);
1067            } else {
1068                return (35.0 + (i as f32 * 3.0) % 15.0).min(50.0);
1069            }
1070        }
1071    }
1072    // Neutral default
1073    60.0
1074}
