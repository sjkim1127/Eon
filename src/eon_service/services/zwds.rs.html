1//! Zi Wei Dou Shu analysis service module
2//!
3//! Performs birth datetime adjustments and calls the ZWDS engine to compute chart, Decadal Luck (Daxian), and Annual Luck (Liunian).
4
5use crate::birth::prepare_birth_context;
6use crate::dto::{AnalysisMeta, BirthTimePrecision, ZwdsAnalysisInput, ZwdsAnalysisOutput};
7use crate::error::ServiceError;
8use chrono::Datelike;
9use eon_core::Gender;
10use eon_zwds::annual::calculate_liunian;
11use eon_zwds::build_chart;
12
13/// Performs Zi Wei Dou Shu analysis.
14pub fn analyze(input: ZwdsAnalysisInput) -> Result<ZwdsAnalysisOutput, ServiceError> {
15    crate::birth::require_known_birth_time(&input.base, "Zi Wei Dou Shu")?;
16    let gender = if input.is_male {
17        Gender::Male
18    } else {
19        Gender::Female
20    };
21
22    // ZWDS typically does not apply longitude True Solar Time correction: apply_tst = false
23    let birth_ctx = prepare_birth_context(&input.base, Some(gender), false)?;
24
25    // ZWDS reads the standard (non-DST) clock: remove the DST hour, but no longitude shift.
26    let mut zwds_birth = birth_ctx.birth_info.clone();
27    let (sy, sm, sd, sh, smin) = birth_ctx.birth_info.standard_clock_datetime();
28    zwds_birth.year = sy;
29    zwds_birth.month = sm;
30    zwds_birth.day = sd;
31    zwds_birth.hour = sh;
32    zwds_birth.minute = smin;
33
34    // 1. Build chart
35    let mut chart = build_chart(&zwds_birth)
36        .map_err(|e| ServiceError::Zwds(format!("Failed to build ZWDS chart: {}", e)))?;
37
38    // 2. Determine target year (defaults to current year)
39    let target_year = input
40        .target_year
41        .unwrap_or_else(|| chrono::Utc::now().year());
42
43    // 3. Compute Annual Luck (Liunian) for target year
44    let current_liu_nian = calculate_liunian(target_year);
45
46    // Update Liunian palace marker in chart
47    for palace in chart.palaces.iter_mut() {
48        if palace.index == current_liu_nian.palace_idx {
49            palace.is_current_liu_nian = true;
50        }
51    }
52
53    // 4. Calculate age and identify current Decadal Luck (Daxian)
54    // ZWDS Daxian starts from Bureau start age rather than Western age.
55    // Map current Daxian by elapsed years.
56    let age =
57        (target_year.saturating_sub(birth_ctx.birth_info.year).max(0) as u32).saturating_add(1);
58
59    let current_daxian = chart
60        .daxian
61        .iter()
62        .find(|d| age >= d.age_start && age <= d.age_end)
63        .cloned()
64        .or_else(|| chart.daxian.first().cloned())
65        .ok_or_else(|| ServiceError::Zwds("Daxian list is empty".to_string()))?;
66
67    Ok(ZwdsAnalysisOutput {
68        meta: AnalysisMeta {
69            precision: BirthTimePrecision::Exact,
70            input_time: birth_ctx.input_time_string,
71            corrected_time: birth_ctx.corrected_time_string,
72            is_dst: birth_ctx.is_dst,
73            dst_offset_hours: birth_ctx.dst_offset_hours,
74            analysis_timezone: input.base.timezone,
75        },
76        chart,
77        current_daxian,
78        current_liu_nian,
79    })
80}

82#[cfg(test)]
83mod tests {
84    use super::*;
85    use crate::dto::AnalysisInput;
86
87    fn chart_key(input: AnalysisInput) -> (usize, usize, String) {
88        let out = analyze(ZwdsAnalysisInput::new(input, true, Some(2024))).unwrap();
89        (
90            out.chart.soul_idx,
91            out.chart.body_idx,
92            format!("{:?}", out.chart.five_elements),
93        )
94    }
95
96    #[test]
97    fn dst_hour_is_removed_before_building_chart() {
98        // Seoul 1988-07-01 01:10 is DST (UTC+10): the standard clock reads 00:10 (Zi hour).
99        // Tokyo has no DST (UTC+9), so its 00:10 is the same standard-clock reading.
100        let seoul_dst = AnalysisInput::new(1988, 7, 1, 1, 10, "Asia/Seoul", 37.5665, 126.978);
101        let tokyo_std = AnalysisInput::new(1988, 7, 1, 0, 10, "Asia/Tokyo", 37.5665, 126.978);
102        assert_eq!(chart_key(seoul_dst), chart_key(tokyo_std));
103    }
