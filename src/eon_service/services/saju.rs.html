22/// Birth context and Four Pillars of one person, shared by the single and two-person analyses.
23struct NatalSaju {
24    gender: Gender,
25    birth_ctx: PreparedBirthContext,
26    pillars: FourPillars,
27}
28
29fn build_natal(input: &SajuAnalysisInput) -> Result<NatalSaju, ServiceError> {
30    let gender = if input.is_male {
31        Gender::Male
32    } else {
33        Gender::Female
34    };
35
36    let birth_ctx = prepare_birth_context(&input.base, Some(gender), true)?;
37    let birth_utc = birth_ctx
38        .birth_info
39        .to_utc()
40        .map_err(|e| ServiceError::InvalidInput(e.to_string()))?;
41
42    let saju_input = SajuInput::new_solar(
43        birth_ctx.corrected_year,
44        birth_ctx.corrected_month,
45        birth_ctx.corrected_day,
46        birth_ctx.corrected_hour,
47        birth_ctx.corrected_minute,
48    )
49    .with_birth_utc(birth_utc)
50    .with_gender(gender)
51    .with_night_rat_hour(input.use_night_rat_hour)
52    .with_unknown_time(matches!(
53        input.precision,
54        BirthTimePrecision::UnknownTimeNoonProxy
55    ));
56
57    let pillars = FourPillars::calculate(&saju_input)
58        .map_err(|e| ServiceError::Saju(format!("Four Pillars calculation failed: {}", e)))?;
59
60    Ok(NatalSaju {
61        gender,
62        birth_ctx,
63        pillars,
64    })
65}

67/// Two-person compatibility audit: stem/branch combinations, clashes and resource contention.
68pub fn analyze_compatibility(
69    a: SajuAnalysisInput,
70    b: SajuAnalysisInput,
71) -> Result<SajuCompatibilityOutput, ServiceError> {
72    let natal_a = build_natal(&a)?;
73    let natal_b = build_natal(&b)?;
74
75    let audit = CompatibilityAuditor::audit(
76        &SajuVM::new(natal_a.pillars.clone()),
77        &SajuVM::new(natal_b.pillars.clone()),
78    );
79
80    let meta = |ctx: PreparedBirthContext, input: &SajuAnalysisInput| AnalysisMeta {
81        precision: input.precision,
82        input_time: ctx.input_time_string,
83        corrected_time: ctx.corrected_time_string,
84        is_dst: ctx.is_dst,
85        dst_offset_hours: ctx.dst_offset_hours,
86        analysis_timezone: input.base.timezone.clone(),
87    };
88
89    Ok(SajuCompatibilityOutput {
90        meta_a: meta(natal_a.birth_ctx, &a),
91        meta_b: meta(natal_b.birth_ctx, &b),
92        audit,
93    })
94}

96pub fn analyze(input: SajuAnalysisInput) -> Result<SajuAnalysisOutput, ServiceError> {
97    let NatalSaju {
98        gender,
99        birth_ctx,
100        pillars,
101    } = build_natal(&input)?;
102
103    // Static engine calculations
104    let lints = DestinyLinter::lint(&pillars);
105    let entropy = DestinyEntropy::analyze(&pillars);
106    let qi_topology = QiTopology::analyze(&pillars);
107    let relationships = pillars.relationships();
108    let void_analysis = pillars.void_analysis();
109
110    let mut report = SajuReport::new(pillars.clone());
111    let mut load_diagnostics = Vec::new();
112    let mut crash_count = 0;
113    let mut vulnerability_report = None;
114    let mut complexity = None;
115
116    if let Ok(major_luck) = MajorLuckAnalysis::calculate_astro(
117        &pillars,
118        gender,
119        birth_ctx.corrected_year,
120        birth_ctx.corrected_month,
121        birth_ctx.corrected_day,
122        birth_ctx.corrected_hour,
123        birth_ctx.corrected_minute,
124    ) {
125        let emulator = LifePathEmulator::new(pillars.clone(), gender, birth_ctx.corrected_year);
126        if let Ok(life_report) = emulator.emulate() {
127            let golden_time = Analyzer::find_golden_time(&life_report.timeline, 10);
128
129            load_diagnostics = KarmaLoadBalancer::diagnose(&life_report.frames);
130            complexity = Some(DestinyComplexity::analyze(&life_report.frames));
131
132            let vm_fuzz = SajuVM::new(pillars.clone());
133            let fuzzer = DestinyFuzzer::new(vm_fuzz);
134            let fuzzer_report = fuzzer.audit(pillars.month);
135            crash_count = fuzzer_report.total_crashes as u32;
136            vulnerability_report = Some(fuzzer_report);
137
138            report = report
139                .with_major_luck(major_luck)
140                .with_timeline(life_report.timeline)
141                .with_vm_simulation(life_report.frames);
142
143            if let Some(gt) = golden_time {
144                report = report.with_golden_time(gt);
145            }
146        }
147    }
148
149    Ok(SajuAnalysisOutput {
150        meta: AnalysisMeta {
151            precision: input.precision,
152            input_time: birth_ctx.input_time_string,
153            corrected_time: birth_ctx.corrected_time_string,
154            is_dst: birth_ctx.is_dst,
155            dst_offset_hours: birth_ctx.dst_offset_hours,
156            analysis_timezone: input.base.timezone,
157        },
158        report,
159        lints,
160        entropy,
161        qi_topology,
162        load_diagnostics,
163        crash_count,
164        vulnerability_report,
165        relationships,
166        void_analysis,
167        complexity,
168    })
169}

171#[cfg(test)]
172mod tests {
173    use super::*;
174    use crate::dto::AnalysisInput;
175
176    #[test]
177    fn test_solar_term_boundary_after_lichun_2024_02_04() {
178        // 2024-02-04 17:45 KST in Seoul. Actual Lichun was at 17:27 KST.
179        // Longitude correction for Seoul is ~ -32 min (17:13).
180        // Year pillar must be Jia-Chen (甲辰) and Month pillar must be Bing-Yin (丙寅).
181        let input = SajuAnalysisInput::new(
182            AnalysisInput::new(2024, 2, 4, 17, 45, "Asia/Seoul", 37.5665, 126.9780)
183                .with_city("Seoul".to_string()),
184            true,
185            false,
186            None,
187        );
188
189        let output = analyze(input).expect("Saju analysis should succeed");
190        let pillars = output.report.pillars;
191
192        assert_eq!(pillars.year.to_string(), "甲辰");
193        assert_eq!(pillars.month.to_string(), "丙寅");
194        assert_eq!(pillars.day.to_string(), "戊戌");
195        assert_eq!(pillars.hour.unwrap().to_string(), "辛酉");
196    }
