9/// House systems accepted by the Western engine (full name or Swiss Ephemeris letter).
10pub const HOUSE_SYSTEMS: &str =
11    "Placidus, Koch, WholeSign, Equal, Porphyry, Regiomontanus, Campanus, Alcabitius, Morinus, Topocentric";
12
13/// Maps a house system name (case-insensitive) or a single Swiss Ephemeris letter to its letter.
14///
15/// Unknown names are rejected instead of silently falling back to another system.
16pub fn parse_house_system(value: &str) -> Result<char, ServiceError> {
17    let v = value.trim();
18    let normalized: String = v
19        .chars()
20        .filter(|c| !matches!(c, ' ' | '-' | '_'))
21        .collect::<String>()
22        .to_ascii_lowercase();
23
24    let letter = match normalized.as_str() {
25        "placidus" | "p" => 'P',
26        "koch" | "k" => 'K',
27        "wholesign" | "whole" | "w" => 'W',
28        "equal" | "e" => 'E',
29        "porphyry" | "porphyrius" | "o" => 'O',
30        "regiomontanus" | "r" => 'R',
31        "campanus" | "c" => 'C',
32        "alcabitius" | "b" => 'B',
33        "morinus" | "m" => 'M',
34        "topocentric" | "polichpage" | "t" => 'T',
35        _ => {
36            return Err(ServiceError::InvalidInput(format!(
37                "Unknown house system '{value}'. Expected one of: {HOUSE_SYSTEMS}"
38            )))
39        }
40    };
41    Ok(letter)
42}

44pub fn analyze(input: WesternAnalysisInput) -> Result<WesternAnalysisOutput, ServiceError> {
45    crate::birth::require_known_birth_time(&input.base, "Western astrology")?;
46    let house_char = parse_house_system(&input.house_system)?;
47    let birth_ctx = prepare_birth_context(&input.base, None, false)?;
48
49    let dt = birth_ctx
50        .birth_info
51        .to_utc()
52        .map_err(|e| ServiceError::BirthInfo(e.to_string()))?;
53
54    let result = eon_western::calculate_western(dt, input.base.lat, input.base.lon, house_char)
55        .map_err(|e| ServiceError::Western(e.to_string()))?;
56
57    Ok(WesternAnalysisOutput {
58        meta: AnalysisMeta {
59            precision: BirthTimePrecision::Exact,
60            input_time: birth_ctx.input_time_string,
61            corrected_time: birth_ctx.corrected_time_string,
62            is_dst: birth_ctx.is_dst,
63            dst_offset_hours: birth_ctx.dst_offset_hours,
64            analysis_timezone: input.base.timezone.clone(),
65        },
66        result,
67    })
68}

70/// Synastry (inter-chart aspects) and composite chart of two people.
71pub fn analyze_synastry(
72    a: WesternAnalysisInput,
73    b: WesternAnalysisInput,
74    name_a: &str,
75    name_b: &str,
76) -> Result<WesternSynastryOutput, ServiceError> {
77    let chart_a = analyze(a)?;
78    let chart_b = analyze(b)?;
79
80    let synastry =
81        eon_western::calculate_synastry(&chart_a.result, &chart_b.result, name_a, name_b);
82    let composite = eon_western::calculate_composite(&chart_a.result, &chart_b.result)
83        .map_err(|e| ServiceError::Western(e.to_string()))?;
84
85    Ok(WesternSynastryOutput {
86        meta_a: chart_a.meta,
87        meta_b: chart_b.meta,
88        synastry,
89        composite,
90    })
91}

93/// Secondary progression (one day = one year) of the natal chart to `target_time`.
94pub fn analyze_progression(
95    input: WesternAnalysisInput,
96    target_time: DateTime<Utc>,
97) -> Result<WesternProgressionOutput, ServiceError> {
98    crate::birth::require_known_birth_time(&input.base, "Western astrology")?;
99    let house_char = parse_house_system(&input.house_system)?;
100    let birth_ctx = prepare_birth_context(&input.base, None, false)?;
101    let dt = birth_ctx
102        .birth_info
103        .to_utc()
104        .map_err(|e| ServiceError::BirthInfo(e.to_string()))?;
105
106    if target_time < dt {
107        return Err(ServiceError::InvalidInput(
108            "Progression target time must not be before the birth time".to_string(),
109        ));
110    }
111
112    let result = eon_western::calculate_secondary_progression(
113        dt,
114        input.base.lat,
115        input.base.lon,
116        target_time,
117        house_char,
118    )
119    .map_err(|e| ServiceError::Western(e.to_string()))?;
120
121    Ok(WesternProgressionOutput {
122        meta: AnalysisMeta {
123            precision: BirthTimePrecision::Exact,
124            input_time: birth_ctx.input_time_string,
125            corrected_time: birth_ctx.corrected_time_string,
126            is_dst: birth_ctx.is_dst,
127            dst_offset_hours: birth_ctx.dst_offset_hours,
128            analysis_timezone: input.base.timezone.clone(),
129        },
130        target_time,
131        result,
132    })
