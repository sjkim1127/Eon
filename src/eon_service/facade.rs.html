34/// QiMen plate for the moment in `input` (birth chart, or a divination chart for any moment)
35/// plus a reading of the palaces that stand for `question`.
36pub fn analyze_qimen_question(
37    input: QimenAnalysisInput,
38    question: eon_qimen::analysis::question::QimenQuestion,
39) -> Result<QimenQuestionOutput, ServiceError> {
40    let is_male = input.is_male;
41    let output = crate::services::qimen::analyze_qimen(input)?;
42    let reading = eon_qimen::analysis::question::read_question(
43        &output.report.pan,
44        &output.report.insights,
45        question,
46        is_male,
47    );
48    Ok(QimenQuestionOutput {
49        meta: output.meta,
50        report: output.report,
51        reading,
52    })
53}

73pub fn analyze_all_synthesized(
74    input: AnalysisInput,
75    is_male: bool,
76    name: Option<String>,
77) -> Result<SynthesizedOmniAnalysis, ServiceError> {
78    let saju_in = SajuAnalysisInput::new(input.clone(), is_male, false, Some(input.unknown_time));
79    let vedic_in = VedicAnalysisInput::new(input.clone(), Some(input.unknown_time), None);
80    let western_in = WesternAnalysisInput::new(input.clone(), "Placidus".to_string());
81    let zwds_in = ZwdsAnalysisInput::new(input.clone(), is_male, None);
82    let hd_in = HumanDesignAnalysisInput::new(input.clone());
83    let qimen_in = QimenAnalysisInput::new(input.clone(), is_male);
84    let num_in = NumerologyAnalysisInput::new(input.clone(), name, None);
85
86    let unknown_time = input.unknown_time;
87
88    let (saju_res, vedic_res, western_res, zwds_res, hd_res, qimen_res, num_res) =
89        std::thread::scope(|s| {
90            let saju_handle = s.spawn(|| analyze_saju(saju_in));
91            let vedic_handle = s.spawn(|| analyze_vedic(vedic_in));
92            let western_handle = s.spawn(|| analyze_western(western_in).ok());
93            let zwds_handle = s.spawn(|| analyze_zwds(zwds_in).ok());
94            let hd_handle = s.spawn(|| analyze_human_design(hd_in).ok());
95            let qimen_handle = s.spawn(|| analyze_qimen(qimen_in).ok());
96            let num_handle = s.spawn(|| analyze_numerology(num_in).ok());
97
98            let saju = saju_handle
99                .join()
100                .unwrap_or_else(|_| Err(ServiceError::Saju("thread panicked".to_string())));
101            let vedic = vedic_handle
102                .join()
103                .unwrap_or_else(|_| Err(ServiceError::Vedic("thread panicked".to_string())));
104            let western = western_handle.join().unwrap_or(None);
105            let zwds = zwds_handle.join().unwrap_or(None);
106            let hd = hd_handle.join().unwrap_or(None);
107            let qimen = qimen_handle.join().unwrap_or(None);
108            let num = num_handle.join().unwrap_or(None);
109
110            (saju, vedic, western, zwds, hd, qimen, num)
111        });
112
113    let saju = saju_res?;
114    // Vedic is mandatory for a known birth time, but intentionally skipped without one.
115    let vedic = match vedic_res {
116        Ok(v) => Some(v),
117        Err(ServiceError::BirthTimeRequired(_)) if unknown_time => None,
118        Err(e) => return Err(e),
119    };
120
121    let skipped_engines = if unknown_time {
122        [
123            ("Vedic astrology", vedic.is_none()),
124            ("Western astrology", western_res.is_none()),
125            ("Zi Wei Dou Shu", zwds_res.is_none()),
126            ("Human Design", hd_res.is_none()),
127            ("QiMen DunJia", qimen_res.is_none()),
128        ]
129        .into_iter()
130        .filter(|(_, skipped)| *skipped)
131        .map(|(engine, _)| SkippedEngine {
132            engine: engine.to_string(),
133            reason: "Birth time of day is unknown; this engine depends on the ascendant, houses or hour-based chart and is not evaluated.".to_string(),
134        })
135        .collect()
136    } else {
137        Vec::new()
138    };
139
140    let omni_input = OmniDestinyTierInput {
141        saju: saju.clone(),
142        vedic: vedic.clone(),
143        western: western_res.clone(),
144        zwds: zwds_res.clone(),
145        human_design: hd_res.clone(),
146        qimen: qimen_res.clone(),
147        numerology: num_res.clone(),
148        transit: None,
149    };
150
151    let tier = analyze_destiny_tier_omni(omni_input)?;
152
153    let handbook = crate::lexicon::build_handbook(
154        &saju,
155        vedic.as_ref(),
156        western_res.as_ref(),
157        zwds_res.as_ref(),
158        hd_res.as_ref(),
159        qimen_res.as_ref(),
160        num_res.as_ref(),
161    );
162
163    Ok(SynthesizedOmniAnalysis {
164        input,
165        tier,
166        saju,
167        vedic,
168        western: western_res,
169        zwds: zwds_res,
170        human_design: hd_res,
171        qimen: qimen_res,
172        numerology: num_res,
173        handbook,
174        skipped_engines,
175    })
176}

178pub fn analyze_iching(input: SajuAnalysisInput) -> Result<IChingAnalysisOutput, ServiceError> {
179    let saju_res = analyze_saju(input.clone())?;
180    let pillars = &saju_res.report.pillars;
181    let birth_year = input.base.year;
182    let hour_pillar = pillars
183        .hour
184        .ok_or_else(|| ServiceError::BirthTimeRequired("I Ching HeLuo".to_string()))?;
185
186    let res = eon_saju::analysis::heluo::calculate_heluo(
187        birth_year,
188        input.is_male,
189        &pillars.year,
190        &pillars.month,
191        &pillars.day,
192        &hour_pillar,
193    );
194
195    Ok(IChingAnalysisOutput {
196        meta: saju_res.meta,
197        result: res,
198    })
199}
