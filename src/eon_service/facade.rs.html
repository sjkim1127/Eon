44pub fn analyze_all_synthesized(
45    input: AnalysisInput,
46    is_male: bool,
47    name: Option<String>,
48) -> Result<SynthesizedOmniAnalysis, ServiceError> {
49    let saju_in = SajuAnalysisInput::new(input.clone(), is_male, false, Some(false));
50    let vedic_in = VedicAnalysisInput::new(input.clone(), Some(false), None);
51    let western_in = WesternAnalysisInput::new(input.clone(), "Placidus".to_string());
52    let zwds_in = ZwdsAnalysisInput::new(input.clone(), is_male, None);
53    let hd_in = HumanDesignAnalysisInput::new(input.clone());
54    let qimen_in = QimenAnalysisInput::new(input.clone(), is_male);
55    let num_in = NumerologyAnalysisInput::new(input.clone(), name, None);
56
57    let (saju_res, vedic_res, western_res, zwds_res, hd_res, qimen_res, num_res) =
58        std::thread::scope(|s| {
59            let saju_handle = s.spawn(|| analyze_saju(saju_in));
60            let vedic_handle = s.spawn(|| analyze_vedic(vedic_in));
61            let western_handle = s.spawn(|| analyze_western(western_in).ok());
62            let zwds_handle = s.spawn(|| analyze_zwds(zwds_in).ok());
63            let hd_handle = s.spawn(|| analyze_human_design(hd_in).ok());
64            let qimen_handle = s.spawn(|| analyze_qimen(qimen_in).ok());
65            let num_handle = s.spawn(|| analyze_numerology(num_in).ok());
66
67            let saju = saju_handle
68                .join()
69                .unwrap_or_else(|_| Err(ServiceError::Saju("thread panicked".to_string())));
70            let vedic = vedic_handle
71                .join()
72                .unwrap_or_else(|_| Err(ServiceError::Vedic("thread panicked".to_string())));
73            let western = western_handle.join().unwrap_or(None);
74            let zwds = zwds_handle.join().unwrap_or(None);
75            let hd = hd_handle.join().unwrap_or(None);
76            let qimen = qimen_handle.join().unwrap_or(None);
77            let num = num_handle.join().unwrap_or(None);
78
79            (saju, vedic, western, zwds, hd, qimen, num)
80        });
81
82    let saju = saju_res?;
83    let vedic = vedic_res?;
84
85    let omni_input = OmniDestinyTierInput {
86        saju: saju.clone(),
87        vedic: vedic.clone(),
88        western: western_res.clone(),
89        zwds: zwds_res.clone(),
90        human_design: hd_res.clone(),
91        qimen: qimen_res.clone(),
92        numerology: num_res.clone(),
93        transit: None,
94    };
95
96    let tier = analyze_destiny_tier_omni(omni_input)?;
97
98    Ok(SynthesizedOmniAnalysis {
99        input,
100        tier,
101        saju,
102        vedic,
103        western: western_res,
104        zwds: zwds_res,
105        human_design: hd_res,
106        qimen: qimen_res,
107        numerology: num_res,
108    })
109}

111pub fn analyze_iching(input: SajuAnalysisInput) -> Result<IChingAnalysisOutput, ServiceError> {
112    let saju_res = analyze_saju(input.clone())?;
113    let pillars = &saju_res.report.pillars;
114    let birth_year = input.base.year;
115
116    let res = eon_saju::analysis::heluo::calculate_heluo(
117        birth_year,
118        input.is_male,
119        &pillars.year,
120        &pillars.month,
121        &pillars.day,
122        &pillars.hour,
123    );
124
125    Ok(IChingAnalysisOutput {
126        meta: saju_res.meta,
127        result: res,
128    })
129}
