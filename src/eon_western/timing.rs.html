13/// Days per tropical year, used for the "one year of life = one step" scalings.
14const YEAR_DAYS: f64 = 365.2425;
15/// Mean daily motion of the Moon (degrees per day), used only to seed the lunar return search.
16const MOON_MEAN_MOTION: f64 = 13.176_358;
17
18const MAJOR_ASPECTS: [AspectType; 5] = [
19    AspectType::Conjunction,
20    AspectType::Sextile,
21    AspectType::Square,
22    AspectType::Trine,
23    AspectType::Opposition,
24];
25
26/// Orb (degrees) for a moving body to a natal point. Faster bodies get tighter orbs; sextiles are
27/// 25% tighter than the other major aspects. This is a school-dependent convention.
28fn moving_orb(body: &str, aspect: AspectType) -> f64 {
29    let base = match body {
30        "Moon" => 1.0,
31        "Sun" | "Mercury" | "Venus" | "Mars" => 2.0,
32        "Jupiter" | "Saturn" | "Uranus" | "Neptune" | "Pluto" => 3.0,
33        _ => 1.5,
34    };
35    if aspect == AspectType::Sextile {
36        base * 0.75
37    } else {
38        base
39    }
40}
41
42/// Orb for solar arc directions, which move about one degree per year.
43const SOLAR_ARC_ORB: f64 = 1.0;
44
45/// Aspects from moving points to natal points. `moving` carries its speed (degrees per day for
46/// transits; the arc rate for directions) so applying/separating can be told.
47fn aspects_to_natal(
48    prefix_moving: &str,
49    moving: &[(String, f64, f64)],
50    natal: &[Point],
51    orb_for: impl Fn(&str, AspectType) -> f64,
52) -> Vec<WesternAspectData> {
53    let mut found = Vec::new();
54    for (name, longitude, speed) in moving {
55        for point in natal {
56            let signed_delta = (longitude - point.longitude + 540.0).rem_euclid(360.0) - 180.0;
57            let angle = signed_delta.abs();
58            for aspect in MAJOR_ASPECTS {
59                let orb = (angle - aspect.angle()).abs();
60                if orb <= orb_for(name, aspect) {
61                    found.push(WesternAspectData {
62                        body_a_name: format!("{prefix_moving} {name}"),
63                        body_b_name: format!("Natal {}", point.name),
64                        aspect_type: aspect,
65                        angle_diff: angle,
66                        orb,
67                        dynamics: aspect_dynamics(angle, aspect.angle(), signed_delta, *speed),
68                        is_major: true,
69                    });
70                }
71            }
72        }
73    }
74    found.sort_by(|a, b| a.orb.total_cmp(&b.orb));
75    found
76}

78fn house_of(longitude: f64, natal: &WesternResult) -> usize {
79    for (i, house) in natal.houses.iter().enumerate() {
80        let next = natal.houses[(i + 1) % natal.houses.len()].cusp_longitude;
81        if is_angle_between(longitude, house.cusp_longitude, next) {
82            return house.house_number;
83        }
84    }
85    1
86}

111/// The sky at `target_time` laid over the natal chart (houses stay natal).
112pub fn calculate_transits(
113    birth_time: DateTime<Utc>,
114    lat: f64,
115    lon: f64,
116    target_time: DateTime<Utc>,
117    house_sys: char,
118) -> Result<TransitResult, WesternError> {
119    let natal = calculate_western(birth_time, lat, lon, house_sys)?;
120    // Houses of the transit chart are not used; only its planet positions matter.
121    let sky = calculate_western(target_time, lat, lon, 'W')?;
122
123    let planets: Vec<TransitPlanetData> = sky
124        .planets
125        .iter()
126        .filter(|p| p.name != "South Node")
127        .map(|p| TransitPlanetData {
128            name: p.name.clone(),
129            longitude: p.longitude,
130            speed: p.speed,
131            is_retrograde: p.is_retrograde,
132            sign_index: p.sign_index,
133            degree_in_sign: p.degree_in_sign,
134            natal_house: house_of(p.longitude, &natal),
135        })
136        .collect();
137
138    let moving: Vec<(String, f64, f64)> = planets
139        .iter()
140        .map(|p| (p.name.clone(), p.longitude, p.speed))
141        .collect();
142    let targets = natal_points(&natal.planets, natal.ascendant, natal.midheaven);
143    let aspects = aspects_to_natal("Transit", &moving, &targets, moving_orb);
144
145    Ok(TransitResult {
146        target_time,
147        natal_chart: natal,
148        planets,
149        aspects,
150    })
151}

173/// Solar return of the calendar `year` (the Sun back on its natal longitude near the birthday).
174/// The return chart is cast for `lat`/`lon`, which may be a relocation.
175pub fn calculate_solar_return(
176    birth_time: DateTime<Utc>,
177    birth_lat: f64,
178    birth_lon: f64,
179    year: i32,
180    return_lat: f64,
181    return_lon: f64,
182    house_sys: char,
183) -> Result<ReturnResult, WesternError> {
184    use chrono::Datelike;
185    let years = year - birth_time.year();
186    if years <= 0 {
187        return Err(WesternError::InvalidInput(
188            "a solar return needs a year after the birth year".to_string(),
189        ));
190    }
191
192    let natal = calculate_western(birth_time, birth_lat, birth_lon, house_sys)?;
193    let natal_sun = natal
194        .planets
195        .iter()
196        .find(|p| p.name == "Sun")
197        .map(|p| p.longitude)
198        .unwrap_or(0.0);
199
200    let engine = AstroEngine::new();
201    let start = birth_time + Duration::seconds((f64::from(years) * YEAR_DAYS * 86_400.0) as i64);
202    let exact_time = engine.find_time_for_planet_longitude(start, natal_sun, 0)?;
203
204    finish_return(
205        ReturnKind::Solar,
206        natal,
207        exact_time,
208        return_lat,
209        return_lon,
210        house_sys,
211    )
212}

214/// First lunar return on or after `after` (the Moon back on its natal longitude).
215pub fn calculate_lunar_return(
216    birth_time: DateTime<Utc>,
217    birth_lat: f64,
218    birth_lon: f64,
219    after: DateTime<Utc>,
220    return_lat: f64,
221    return_lon: f64,
222    house_sys: char,
223) -> Result<ReturnResult, WesternError> {
224    let natal = calculate_western(birth_time, birth_lat, birth_lon, house_sys)?;
225    let natal_moon = natal
226        .planets
227        .iter()
228        .find(|p| p.name == "Moon")
229        .map(|p| p.longitude)
230        .unwrap_or(0.0);
231
232    let engine = AstroEngine::new();
233    let (current, _) = engine.get_planet_full(after, 1, 256)?;
234    let ahead = (natal_moon - current).rem_euclid(360.0);
235    let seed = after + Duration::seconds((ahead / MOON_MEAN_MOTION * 86_400.0) as i64);
236    let mut exact_time = engine.find_time_for_planet_longitude(seed, natal_moon, 1)?;
237
238    // Newton refinement may land on the previous pass when the Moon is almost on the point.
239    if exact_time < after {
240        let next_seed = exact_time + Duration::seconds((27.32 * 86_400.0) as i64);
241        exact_time = engine.find_time_for_planet_longitude(next_seed, natal_moon, 1)?;
242    }
243
244    finish_return(
245        ReturnKind::Lunar,
246        natal,
247        exact_time,
248        return_lat,
249        return_lon,
250        house_sys,
251    )
252}
253
254fn finish_return(
255    kind: ReturnKind,
256    natal: WesternResult,
257    exact_time: DateTime<Utc>,
258    lat: f64,
259    lon: f64,
260    house_sys: char,
261) -> Result<ReturnResult, WesternError> {
262    let return_chart = calculate_western(exact_time, lat, lon, house_sys)?;
263
264    let moving: Vec<(String, f64, f64)> = return_chart
265        .planets
266        .iter()
267        .filter(|p| p.name != "South Node")
268        .map(|p| (p.name.clone(), p.longitude, p.speed))
269        .collect();
270    let targets = natal_points(&natal.planets, natal.ascendant, natal.midheaven);
271    // Charts are compared as static pictures: no applying/separating motion.
272    let static_moving: Vec<(String, f64, f64)> =
273        moving.into_iter().map(|(n, l, _)| (n, l, 0.0)).collect();
274    let aspects = aspects_to_natal("Return", &static_moving, &targets, |_, aspect| {
275        if aspect == AspectType::Sextile {
276            4.0
277        } else {
278            5.0
279        }
280    });
281
282    Ok(ReturnResult {
283        kind,
284        exact_time,
285        natal_chart: natal,
286        return_chart,
287        aspects_to_natal: aspects,
288    })
289}

301#[derive(Debug, Clone, Serialize, Deserialize, PartialEq)]
302pub struct SolarArcResult {
303    pub target_time: DateTime<Utc>,
304    pub age_years: f64,
305    /// Secondary-progressed Sun minus natal Sun: added to every natal point
306    pub arc_degrees: f64,
307    pub natal_chart: WesternResult,
308    pub directed: Vec<DirectedPoint>,
309    /// Directed points to natal planets/angles (orb 1 degree), tightest first
310    pub aspects: Vec<WesternAspectData>,
311}
312
313pub fn calculate_solar_arc(
314    birth_time: DateTime<Utc>,
315    lat: f64,
316    lon: f64,
317    target_time: DateTime<Utc>,
318    house_sys: char,
319) -> Result<SolarArcResult, WesternError> {
320    let natal = calculate_western(birth_time, lat, lon, house_sys)?;
321    let natal_sun = natal
322        .planets
323        .iter()
324        .find(|p| p.name == "Sun")
325        .map(|p| p.longitude)
326        .unwrap_or(0.0);
327
328    let age_years = (target_time - birth_time).num_seconds() as f64 / 86_400.0 / YEAR_DAYS;
329    let progressed_time = birth_time + Duration::seconds((age_years * 86_400.0) as i64);
330    let engine = AstroEngine::new();
331    let (progressed_sun, _) = engine.get_planet_full(progressed_time, 0, 4 | 256)?;
332    let arc_degrees = (progressed_sun - natal_sun).rem_euclid(360.0);
333
334    let points = natal_points(&natal.planets, natal.ascendant, natal.midheaven);
335    let directed: Vec<DirectedPoint> = points
336        .iter()
337        .map(|p| {
338            let longitude = (p.longitude + arc_degrees).rem_euclid(360.0);
339            DirectedPoint {
340                name: p.name.clone(),
341                longitude,
342                sign_index: (longitude / 30.0).floor() as usize,
343                degree_in_sign: longitude % 30.0,
344            }
345        })
346        .collect();
347
348    // A directed point advances about one degree per year; speeds here are degrees per year.
349    let moving: Vec<(String, f64, f64)> = directed
350        .iter()
351        .map(|d| (d.name.clone(), d.longitude, 1.0))
352        .collect();
353    let aspects = aspects_to_natal("Directed", &moving, &points, |_, _| SOLAR_ARC_ORB)
354        .into_iter()
355        // A directed point trivially conjoins its own natal position only when the arc is ~0.
356        .filter(|a| {
357            let own = a.body_a_name.trim_start_matches("Directed ");
358            let target = a.body_b_name.trim_start_matches("Natal ");
359            !(own == target && a.aspect_type == AspectType::Conjunction && arc_degrees < 0.5)
360        })
361        .collect();
362
363    Ok(SolarArcResult {
364        target_time,
365        age_years,
366        arc_degrees,
367        natal_chart: natal,
368        directed,
369        aspects,
370    })
371}

375#[derive(Debug, Clone, Serialize, Deserialize, PartialEq)]
376pub struct ProfectionResult {
377    pub target_time: DateTime<Utc>,
378    /// Completed years of life at the target time
379    pub age: u32,
380    /// House the year profects to (age 0 = 1st house, 12 = 1st again)
381    pub profected_house: usize,
382    pub profected_sign_index: usize,
383    /// Traditional ruler of the profected sign: the lord of the year
384    pub year_lord: String,
385    pub year_lord_natal_sign_index: usize,
386    pub year_lord_natal_house: usize,
387    /// Natal planets in the profected house
388    pub planets_in_profected_house: Vec<String>,
389}
390
391/// Annual profection: the Ascendant advances one sign per year of life.
392pub fn calculate_profection(
393    birth_time: DateTime<Utc>,
394    lat: f64,
395    lon: f64,
396    target_time: DateTime<Utc>,
397    house_sys: char,
398) -> Result<ProfectionResult, WesternError> {
399    if target_time < birth_time {
400        return Err(WesternError::InvalidInput(
401            "profection target time is before the birth time".to_string(),
402        ));
403    }
404    let natal = calculate_western(birth_time, lat, lon, house_sys)?;
405
406    let age =
407        ((target_time - birth_time).num_seconds() as f64 / 86_400.0 / YEAR_DAYS).floor() as u32;
408    let steps = (age % 12) as usize;
409    let asc_sign = (natal.ascendant / 30.0).floor() as usize;
410    let profected_sign_index = (asc_sign + steps) % 12;
411    let profected_house = steps + 1;
412
413    let year_lord = crate::get_traditional_ruler(profected_sign_index).to_string();
414    let lord = natal.planets.iter().find(|p| p.name == year_lord);
415
416    let planets_in_profected_house = natal
417        .planets
418        .iter()
419        .filter(|p| p.name != "South Node" && p.house_number == profected_house)
420        .map(|p| p.name.clone())
421        .collect();
422
423    Ok(ProfectionResult {
424        target_time,
425        age,
426        profected_house,
427        profected_sign_index,
428        year_lord,
429        year_lord_natal_sign_index: lord.map(|p| p.sign_index).unwrap_or(0),
430        year_lord_natal_house: lord.map(|p| p.house_number).unwrap_or(1),
431        planets_in_profected_house,
432    })
433}
