41fn get_nakshatra_lord(nakshatra: u8) -> VedicPlanet {

56fn get_dagdha_rashis(tithi: u8) -> Vec<u8> {
57    let t = if tithi > 15 { tithi - 15 } else { tithi };
58    match t {
59        1 => vec![1, 7],        // Aries, Libra
60        2 => vec![2, 5],        // Taurus, Leo
61        3 => vec![3, 6],        // Gemini, Virgo
62        4 => vec![2, 11],       // Taurus, Aquarius
63        5 => vec![3, 9],        // Gemini, Sagittarius
64        6 => vec![1, 8],        // Aries, Scorpio
65        7 => vec![3, 12],       // Gemini, Pisces
66        8 => vec![3, 6],        // Gemini, Virgo
67        9 => vec![5, 8],        // Leo, Scorpio
68        10 => vec![5, 9],       // Leo, Sagittarius
69        11 => vec![9, 12],      // Sagittarius, Pisces
70        12 => vec![7, 10],      // Libra, Capricorn
71        13 => vec![2, 3],       // Taurus, Gemini
72        14 => vec![2, 3, 6, 9], // Taurus, Gemini, Virgo, Sagittarius
73        _ => vec![],
74    }

79fn normalize_degrees(degrees: f64) -> f64 {
80    if degrees.is_finite() {
81        degrees.rem_euclid(360.0)
82    } else {
83        0.0
84    }
85}

88    pub fn calculate(
89        sun_deg: f64,
90        moon_deg: f64,
91        time: DateTime<Utc>,
92        latitude: f64,
93        longitude: f64,
94    ) -> Panchanga {
95        let sun = normalize_degrees(sun_deg);
96        let moon = normalize_degrees(moon_deg);
97
98        // 1. Fetch sunrise/sunset for the current LOCAL date
99        let local_offset_mins = (longitude * 4.0) as i64;
100        let local_time = time + chrono::Duration::minutes(local_offset_mins);
101
102        let (mut sunrise, mut sunset, mut polar_day) =
103            Self::calculate_sunrise_sunset(local_time, latitude, longitude);
104
105        // If current time is before the sunrise of its local date,
106        // the astrological day started at yesterday's sunrise
107        if polar_day.is_none() && time < sunrise {
108            let (prev_rise, prev_set, prev_polar_day) = Self::calculate_sunrise_sunset(
109                local_time - chrono::Duration::days(1),
110                latitude,
111                longitude,
112            );
113            sunrise = prev_rise;
114            sunset = prev_set;
115            polar_day = prev_polar_day;
116        }
117
118        let (mut next_sunrise, _, _) = Self::calculate_sunrise_sunset(
119            sunrise + chrono::Duration::days(1),
120            latitude,
121            longitude,
122        );
123
124        // No sunrise/sunset exists during polar night/day. Keep the legacy
125        // DateTime fields usable, but use full-day boundaries rather than a
126        // clamped noon event, which would fabricate an 8-part daytime span.
127        if let Some(is_polar_day) = polar_day {
128            let day_start = sunrise.date_naive().and_hms_opt(0, 0, 0).unwrap().and_utc();
129            if is_polar_day {
130                sunrise = day_start;
131                sunset = day_start + chrono::Duration::days(1);
132            } else {
133                sunrise = day_start;
134                sunset = day_start;
135            }
136            next_sunrise = day_start + chrono::Duration::days(1);
137        }
138
139        // 2. Vara (Weekday) - Vedic Day starts at Sunrise
140        let is_day_birth = time >= sunrise && time < sunset;
141        let is_night_birth = !is_day_birth;
142
143        // Use the actual sunrise time to determine the local weekday
144        let local_offset_mins = (longitude * 4.0) as i64;
145        let sunrise_local = sunrise + chrono::Duration::minutes(local_offset_mins);
146        let vara_date = sunrise_local.date_naive();
147        let vara = match vara_date.weekday() {
148            chrono::Weekday::Sun => "Sunday",
149            chrono::Weekday::Mon => "Monday",
150            chrono::Weekday::Tue => "Tuesday",
151            chrono::Weekday::Wed => "Wednesday",
152            chrono::Weekday::Thu => "Thursday",
153            chrono::Weekday::Fri => "Friday",
154            chrono::Weekday::Sat => "Saturday",
155        }
156        .to_string();
157
158        let day_lord = match vara_date.weekday() {
159            chrono::Weekday::Sun => VedicPlanet::Sun,
160            chrono::Weekday::Mon => VedicPlanet::Moon,
161            chrono::Weekday::Tue => VedicPlanet::Mars,
162            chrono::Weekday::Wed => VedicPlanet::Mercury,
163            chrono::Weekday::Thu => VedicPlanet::Jupiter,
164            chrono::Weekday::Fri => VedicPlanet::Venus,
165            chrono::Weekday::Sat => VedicPlanet::Saturn,
166        };
167
168        // 3. Hourly Lord (Hora)
169        let hour_lord = Self::calculate_hora(time, sunrise, sunset, next_sunrise, day_lord);
170        let daily_parts = Self::calculate_tribhaga_lords(day_lord);
171        let tribhaga_lord = Self::calculate_tribhaga_lord(
172            time,
173            sunrise,
174            sunset,
175            next_sunrise,
176            is_day_birth,
177        );
178
179        // 4. Tithi (Moon - Sun) / 12
180        let tithi_deg = normalize_degrees(moon - sun);
181        let tithi = (tithi_deg / 12.0).floor() as u8 + 1;
182        let tithi_name = Self::get_tithi_name(tithi);
183
184        // 5. Nakshatra (Moon / 13.333)
185        let nakshatra = (moon / (360.0 / 27.0)).floor() as u8 + 1;
186
187        // 6. Yoga (Sun + Moon) / 13.333
188        let yoga_deg = normalize_degrees(sun + moon);
189        let yoga = (yoga_deg / (360.0 / 27.0)).floor() as u8 + 1;
190
191        // 7. Karana (Tithi_deg / 6)
192        // Karana is half of a Tithi.
193        let karana_idx = (tithi_deg / 6.0).floor() as u16 + 1;
194        let karana_name = Self::get_karana_name(karana_idx);
195
196        // Yogi, Avayogi & Dagdha Rashi
197        let yogi_point = normalize_degrees(sun + moon + 93.3333333);
198        let yogi_nak = (yogi_point / (360.0 / 27.0)).floor() as u8 + 1;
199        let yogi_planet = get_nakshatra_lord(yogi_nak);
200
201        let avayogi_point = normalize_degrees(yogi_point + 186.6666667);
202        let avayogi_nak = (avayogi_point / (360.0 / 27.0)).floor() as u8 + 1;
203        let avayogi_planet = get_nakshatra_lord(avayogi_nak);
204
205        let dagdha_rashis = get_dagdha_rashis(tithi);
206
207        // Rahu Kalam, Yamaganda, Gulika calculations (daytime divided into 8 parts)
208        let rahu_part = match vara_date.weekday() {
209            chrono::Weekday::Sun => 8,
210            chrono::Weekday::Mon => 2,
211            chrono::Weekday::Tue => 7,
212            chrono::Weekday::Wed => 5,
213            chrono::Weekday::Thu => 6,
214            chrono::Weekday::Fri => 4,
215            chrono::Weekday::Sat => 3,
216        };
217
218        let yama_part = match vara_date.weekday() {
219            chrono::Weekday::Sun => 5,
220            chrono::Weekday::Mon => 4,
221            chrono::Weekday::Tue => 3,
222            chrono::Weekday::Wed => 2,
223            chrono::Weekday::Thu => 1,
224            chrono::Weekday::Fri => 7,
225            chrono::Weekday::Sat => 6,
226        };
227
228        let guli_part = match vara_date.weekday() {
229            chrono::Weekday::Sun => 7,
230            chrono::Weekday::Mon => 6,
231            chrono::Weekday::Tue => 5,
232            chrono::Weekday::Wed => 4,
233            chrono::Weekday::Thu => 3,
234            chrono::Weekday::Fri => 2,
235            chrono::Weekday::Sat => 1,
236        };
237
238        let day_duration_ms = sunset.signed_duration_since(sunrise).num_milliseconds() as f64;
239        let part_ms = day_duration_ms / 8.0;
240
241        let get_time_range = |part_idx: u8| -> (DateTime<Utc>, DateTime<Utc>) {
242            let start_offset_ms = (part_idx - 1) as f64 * part_ms;
243            let end_offset_ms = part_idx as f64 * part_ms;
244
245            let start_time = sunrise + chrono::Duration::milliseconds(start_offset_ms as i64);
246            let end_time = sunrise + chrono::Duration::milliseconds(end_offset_ms as i64);
247            (start_time, end_time)
248        };
249
250        let rahu_kalam = get_time_range(rahu_part);
251        let yamaganda = get_time_range(yama_part);
252        let gulika = get_time_range(guli_part);
253
254        Panchanga {
255            vara,
256            tithi,
257            tithi_name,
258            nakshatra,
259            yoga,
260            karana: Self::get_karana_type(karana_idx),
261            karana_name,
262            current_time: time,
263            sunrise,
264            sunset,
265            next_sunrise,
266            is_day_birth,
267            is_night_birth,
268            day_lord,
269            hour_lord,
270            tribhaga_lord,
271            daily_parts,
272            yogi_point,
273            yogi_planet,
274            avayogi_planet,
275            dagdha_rashis,
276            rahu_kalam,
277            yamaganda,
278            gulika,
279        }
280    }

282    /// Calculate Hora Lord
283    fn calculate_hora(
284        time: DateTime<Utc>,
285        sunrise: DateTime<Utc>,
286        sunset: DateTime<Utc>,
287        next_sunrise: DateTime<Utc>,
288        day_lord: VedicPlanet,
289    ) -> VedicPlanet {
290        // Hora order: Sun, Venus, Mercury, Moon, Saturn, Jupiter, Mars sequence repeats
291        // Standard Sequence starting from Day Lord is:
292        // 1st Hora = Day Lord
293        // 2nd Hora = 6th weekday from current...
294
295        // Correct Sequence for Hora starting from Sunday: Sun, Venus, Mercury, Moon, Saturn, Jupiter, Mars.
296        let hora_seq = [
297            VedicPlanet::Sun,
298            VedicPlanet::Venus,
299            VedicPlanet::Mercury,
300            VedicPlanet::Moon,
301            VedicPlanet::Saturn,
302            VedicPlanet::Jupiter,
303            VedicPlanet::Mars,
304        ];
305
306        // Find index of day lord in the sequence
307        let start_idx = hora_seq.iter().position(|&p| p == day_lord).unwrap_or(0);
308
309        let is_day = time >= sunrise && time < sunset;
310
311        let (duration, start_time) = if is_day {
312            let diff = sunset.signed_duration_since(sunrise).num_seconds() as f64;
313            (diff / 12.0, sunrise)
314        } else {
315            let diff = next_sunrise.signed_duration_since(sunset).num_seconds() as f64;
316            (diff / 12.0, sunset)
317        };
318
319        if duration == 0.0 {
320            return day_lord;
321        } // Safety
322
323        // Calculate offset from start_time
324        let elapsed = if time >= start_time {
325            time.signed_duration_since(start_time).num_seconds() as f64
326        } else {
327            0.0
328        };
329
330        let hora_idx = (elapsed / duration).floor() as usize;
331        let offset = if is_day { hora_idx } else { hora_idx + 12 };
332
333        let final_idx = (start_idx + offset) % 7;
334        hora_seq[final_idx]
335    }

337    /// Calculate Tribhaga Lords (BPHS Standard)
338    /// Day is divided into 3 parts: Mercury (1st), Sun (2nd), Saturn (3rd)
339    /// Night is divided into 3 parts: Moon (1st), Venus (2nd), Mars (3rd)
340    /// Returns array of 8 lords (3 day + 3 night + 2 padding for Kala Bala compatibility)
341    fn calculate_tribhaga_lords(day_lord: VedicPlanet) -> [VedicPlanet; 8] {
342        // BPHS Standard Tribhaga Lords:
343        // Day Parts (from sunrise to sunset): Mercury, Sun, Saturn
344        // Night Parts (from sunset to next sunrise): Moon, Venus, Mars
345
346        // Return standard pattern (first 3 for day, next 3 for night)
347        [
348            // Day lords (used when is_day_birth == true)
349            VedicPlanet::Mercury, // 1st third of day
350            VedicPlanet::Sun,     // 2nd third of day
351            VedicPlanet::Saturn,  // 3rd third of day
352            // Night lords (used when is_day_birth == false)
353            VedicPlanet::Moon,  // 1st third of night
354            VedicPlanet::Venus, // 2nd third of night
355            VedicPlanet::Mars,  // 3rd third of night
356            // Padding for compatibility
357            day_lord,
358            day_lord,
359        ]
360    }

362    fn calculate_tribhaga_lord(
363        time: DateTime<Utc>,
364        sunrise: DateTime<Utc>,
365        sunset: DateTime<Utc>,
366        next_sunrise: DateTime<Utc>,
367        is_day: bool,
368    ) -> VedicPlanet {
369        let (start, end, lords) = if is_day {
370            (
371                sunrise,
372                sunset,
373                [VedicPlanet::Mercury, VedicPlanet::Sun, VedicPlanet::Saturn],
374            )
375        } else {
376            (
377                sunset,
378                next_sunrise,
379                [VedicPlanet::Moon, VedicPlanet::Venus, VedicPlanet::Mars],
380            )
381        };
382        let duration = end.signed_duration_since(start).num_milliseconds();
383        if duration <= 0 {
384            return lords[0];
385        }
386        let elapsed = time
387            .signed_duration_since(start)
388            .num_milliseconds()
389            .clamp(0, duration.saturating_sub(1));
390        lords[((elapsed * 3) / duration) as usize]
391    }

393    /// NOAA Sunrise/Sunset Algorithm (Simplified)
394    fn calculate_sunrise_sunset(
395        date: DateTime<Utc>,
396        lat: f64,
397        lon: f64,
398    ) -> (DateTime<Utc>, DateTime<Utc>, Option<bool>) {
399        // Convert date to Julian Day
400        let timestamp = date.timestamp();
401        let julian_day = (timestamp as f64 / 86400.0) + 2440587.5;
402        let julian_century = (julian_day - 2451545.0) / 36525.0;
403
404        // Geom Mean Long Sun (deg)
405        let geom_mean_long_sun =
406            (280.46646 + julian_century * (36000.76983 + julian_century * 0.0003032)) % 360.0;
407
408        // Geom Mean Anom Sun (deg)
409        let geom_mean_anom_sun =
410            357.52911 + julian_century * (35999.05029 - 0.0001537 * julian_century);
411
412        // Eccent Earth Orbit
413        let eccent_earth_orbit =
414            0.016708634 - julian_century * (0.000042037 + 0.0000001267 * julian_century);
415
416        // Sun Eq of Ctr
417        let sun_eq_of_ctr = geom_mean_anom_sun.to_radians().sin()
418            * (1.914602 - julian_century * (0.004817 + 0.000014 * julian_century))
419            + (2.0 * geom_mean_anom_sun.to_radians()).sin()
420                * (0.019993 - 0.000101 * julian_century)
421            + (3.0 * geom_mean_anom_sun.to_radians()).sin() * 0.000289;
422
423        let sun_true_long = geom_mean_long_sun + sun_eq_of_ctr;
424
425        // Sun App Long
426        let sun_app_long = sun_true_long
427            - 0.00569
428            - 0.00478 * (125.04 - 1934.136 * julian_century).to_radians().sin();
429
430        // Mean Obliq Ecliptic
431        let mean_obliq_ecliptic = 23.0
432            + (26.0
433                + (21.448
434                    - julian_century
435                        * (46.815 + julian_century * (0.00059 - julian_century * 0.001813)))
436                    / 60.0)
437                / 60.0;
438        let obliq_corr =
439            mean_obliq_ecliptic + 0.00256 * (125.04 - 1934.136 * julian_century).to_radians().cos();
440
441        // Sun Declin
442        let sun_declin = (obliq_corr.to_radians().sin() * sun_app_long.to_radians().sin()).asin();
443
444        // Var y
445        let var_y = (obliq_corr.to_radians() / 2.0).tan().powi(2);
446
447        // Eq of Time (minutes)
448        let eq_of_time = 4.0
449            * (var_y * (2.0 * geom_mean_long_sun.to_radians()).sin()
450                - 2.0 * eccent_earth_orbit * geom_mean_anom_sun.to_radians().sin()
451                + 4.0
452                    * eccent_earth_orbit
453                    * var_y
454                    * geom_mean_anom_sun.to_radians().sin()
455                    * (2.0 * geom_mean_long_sun.to_radians()).cos()
456                - 0.5 * var_y * var_y * (4.0 * geom_mean_long_sun.to_radians()).sin()
457                - 1.25
458                    * eccent_earth_orbit
459                    * eccent_earth_orbit
460                    * (2.0 * geom_mean_anom_sun.to_radians()).sin())
461            .to_degrees();
462
463        // HA Sunrise (deg)
464        // cos(HA) = cos(90.833) / (cos(lat)*cos(declin)) - tan(lat)*tan(declin)
465        let zenith = 90.833f64.to_radians(); // Official sunrise zenith
466        let cos_ha = (zenith.cos() / (lat.to_radians().cos() * sun_declin.cos()))
467            - (lat.to_radians().tan() * sun_declin.tan());
468
469        // Check for polar day/night (simplification: clamp)
470        let (ha_deg, polar_day) = if cos_ha > 1.0 {
471            (0.0, Some(false))
472        } else if cos_ha < -1.0 {
473            (180.0, Some(true))
474        } else {
475            (cos_ha.acos().to_degrees(), None)
476        };
477
478        let solar_noon = 720.0 - 4.0 * lon - eq_of_time;
479
480        let sunrise_min = solar_noon - 4.0 * ha_deg;
481        let sunset_min = solar_noon + 4.0 * ha_deg;
482
483        // Use UTC midnight of the given date reference
484        let midnight = date.date_naive().and_hms_opt(0, 0, 0).unwrap().and_utc();
485
486        let rise_secs = (sunrise_min * 60.0) as i64;
487        let set_secs = (sunset_min * 60.0) as i64;
488
489        (
490            midnight + chrono::Duration::seconds(rise_secs),
491            midnight + chrono::Duration::seconds(set_secs),
492            polar_day,
493        )
494    }

535    /// Karana type mapping (BPHS rules)
536    /// Sequential index 1-60 → Type 1-11
537    /// 1: Kimstughna (fixed, only first karana)
538    /// 2-8: Bava, Balava, Kaulava, Taitila, Gara, Vanija, Vishti (rotating, 8 cycles)
539    /// 9-11: Shakuni, Chatushpada, Naga (fixed, last 3 karanas)
540    fn get_karana_type(karana_idx: u16) -> u8 {
541        if karana_idx == 1 {
542            1 // Kimstughna (fixed)
543        } else if karana_idx >= 58 {
544            // Last 3: Shakuni(9), Chatushpada(10), Naga(11)
545            (karana_idx - 58 + 9) as u8
546        } else {
547            // Rotating 7 movable karanas: Bava(2)..Vishti(8)
548            (((karana_idx - 2) % 7) + 2) as u8
549        }
550    }
551
552    fn get_karana_name(idx: u16) -> String {
553        if idx == 1 {
554            return "Kimstughna".to_string();
555        }
556        if idx >= 58 {
557            match idx {
558                58 => return "Shakuni".to_string(),
559                59 => return "Chatushpada".to_string(),
560                60 => return "Naga".to_string(),
561                _ => {}
562            }
563        }
564        let movables = [
565            "Bava", "Balava", "Kaulava", "Taitila", "Gara", "Vanija", "Vishti",
566        ];
567        let m_idx = ((idx.saturating_sub(2)) % 7) as usize;
568        movables[m_idx].to_string()
569    }
