150    pub fn get_aspect_type(house_diff_1_indexed: u8) -> TajikaAspectType {
151        match house_diff_1_indexed {
152            3 | 11 => TajikaAspectType::Mitra(false),
153            5 | 9 => TajikaAspectType::Mitra(true),
154            1 | 7 => TajikaAspectType::Shatru(true),
155            4 | 10 => TajikaAspectType::Shatru(false),
156            _ => TajikaAspectType::Sama,
157        }
158    }

160    /// Ithasala Yoga (Applying Aspect within Orb)
161    pub fn check_ithasala(
162        p1_long: f64,
163        p1_speed: f64,
164        p2_long: f64,
165        p2_speed: f64,
166        orb: f64,
167    ) -> bool {
168        let (faster_long, _faster_speed, slower_long, _slower_speed) = if p1_speed > p2_speed {
169            (p1_long, p1_speed, p2_long, p2_speed)
170        } else {
171            (p2_long, p2_speed, p1_long, p1_speed)
172        };
173
174        let diff = (slower_long - faster_long + 360.0) % 360.0;
175        diff > 0.0 && diff <= orb
176    }

595    /// Pancha-Vargeeya Bala (5-Fold Strength in Tajika)
596    pub fn calculate_pancha_vargeeya_bala(chart: &VedicChart) -> Vec<PanchaVargeeyaBala> {
597        let mut results = Vec::new();
598        let planets = [
599            VedicPlanet::Sun,
600            VedicPlanet::Moon,
601            VedicPlanet::Mars,
602            VedicPlanet::Mercury,
603            VedicPlanet::Jupiter,
604            VedicPlanet::Venus,
605            VedicPlanet::Saturn,
606        ];
607
608        for &planet in &planets {
609            let pos = match chart.planets.iter().find(|p| p.planet == planet) {
610                Some(p) => p,
611                None => continue,
612            };
613
614            // 1. Kshetra Bala (Max 30)
615            let rasi_lord = VedicPlanet::get_ruler_of(pos.rasi);
616            let kshetra = if rasi_lord == planet {
617                30.0
618            } else if pos.rasi == planet.exaltation_rasi() {
619                30.0
620            } else {
621                15.0
622            };
623
624            // 2. Uchcha Bala (Max 20)
625            let deb_long = (planet.exaltation_rasi() as f64 * 30.0 - 15.0 + 180.0) % 360.0;
626            let mut dist = (pos.sidereal_deg - deb_long + 360.0) % 360.0;
627            if dist > 180.0 {
628                dist = 360.0 - dist;
629            }
630            let uchcha = (dist / 180.0) * 20.0;
631
632            // 3. Hadda Bala (Bounds Strength, Max 15)
633            let hadda = match ((pos.sidereal_deg % 30.0) / 6.0).floor() as u8 {
634                0 => 15.0,
635                1 => 11.25,
636                2 => 7.5,
637                3 => 3.75,
638                _ => 3.75,
639            };
640
641            // 4. Drekkana Bala (Max 10)
642            let decan = ((pos.sidereal_deg % 30.0) / 10.0).floor() as u8;
643            let drekkana = match (planet, decan) {
644                (VedicPlanet::Sun | VedicPlanet::Mars | VedicPlanet::Jupiter, 0) => 10.0,
645                (VedicPlanet::Mercury | VedicPlanet::Saturn, 1) => 10.0,
646                (VedicPlanet::Moon | VedicPlanet::Venus, 2) => 10.0,
647                _ => 5.0,
648            };
649
650            // 5. Navamsha Bala (Max 10)
651            let nav_rasi = (pos.sidereal_deg / (360.0 / 108.0)).floor() as u8 % 12 + 1;
652            let nav_lord = VedicPlanet::get_ruler_of(nav_rasi);
653            let navamsha = if nav_lord == planet {
654                10.0
655            } else if nav_rasi == planet.exaltation_rasi() {
656                10.0
657            } else {
658                5.0
659            };
660
661            let total_virupas = (kshetra + uchcha + hadda + drekkana + navamsha) / 4.0;
662            let grade = if total_virupas >= 15.0 {
663                "우수 (Purna Bala)".to_string()
664            } else if total_virupas >= 10.0 {
665                "보통 (Madhya Bala)".to_string()
666            } else {
667                "약함 (Alpa Bala)".to_string()
668            };
669
670            results.push(PanchaVargeeyaBala {
671                planet,
672                kshetra_bala: kshetra,
673                uchcha_bala: uchcha,
674                hadda_bala: hadda,
675                drekkana_bala: drekkana,
676                navamsha_bala: navamsha,
677                total_virupas,
678                grade,
679            });
680        }
681
682        results
683    }

685    /// Mudda Dasha (1-Year Annual Dasha System)
686    pub fn calculate_mudda_dasha(chart: &VedicChart) -> Vec<MuddaDashaPeriod> {
687        let moon_pos = match chart.planets.iter().find(|p| p.planet == VedicPlanet::Moon) {
688            Some(p) => p.sidereal_deg,
689            None => 0.0,
690        };
691
692        // Vimshottari Lord sequence & base proportions (in 120 year scale)
693        let dasha_lords = [
694            (VedicPlanet::Ketu, "케투 (Ketu)", 7.0),
695            (VedicPlanet::Venus, "금성 (Venus)", 20.0),
696            (VedicPlanet::Sun, "태양 (Sun)", 6.0),
697            (VedicPlanet::Moon, "달 (Moon)", 10.0),
698            (VedicPlanet::Mars, "화성 (Mars)", 7.0),
699            (VedicPlanet::Rahu, "라후 (Rahu)", 18.0),
700            (VedicPlanet::Jupiter, "목성 (Jupiter)", 16.0),
701            (VedicPlanet::Saturn, "토성 (Saturn)", 19.0),
702            (VedicPlanet::Mercury, "수성 (Mercury)", 17.0),
703        ];
704
705        let nak_span = 360.0 / 27.0;
706        let nak_idx = (moon_pos / nak_span).floor() as usize;
707        let lord_start_idx = nak_idx % 9;
708
709        let _nak_progress = (moon_pos % nak_span) / nak_span;
710        let total_year_days = 365.25;
711
712        let mut periods = Vec::new();
713        let mut current_offset = 0.0;
714
715        for i in 0..9 {
716            let idx = (lord_start_idx + i) % 9;
717            let (planet, planet_kr, base_years) = dasha_lords[idx];
718            let duration_days = (base_years / 120.0) * total_year_days;
719
720            let start = current_offset;
721            let end = current_offset + duration_days;
722            current_offset = end;
723
724            periods.push(MuddaDashaPeriod {
725                planet,
726                planet_kr: planet_kr.to_string(),
727                duration_days,
728                start_day_offset: start,
729                end_day_offset: end,
730            });
731        }
732
733        periods
734    }

825    }

858    /// Selection of Year Lord (Varsheshwara) - Full Orthodox Tajika Implementation
859    pub fn select_year_lord(
860        chart: &VedicChart,
861        birth_lagna_rasi: u8,
862        age_years: u32,
863    ) -> VedicPlanet {
864        let annual_lagna_rasi = chart.ascendant.rasi;
865        let muntha_rasi = Self::calculate_muntha(birth_lagna_rasi, age_years);
866        let is_day = chart.panchanga.is_day_birth;
867
868        let mut candidates = Vec::new();
869        candidates.push(VedicPlanet::get_ruler_of(muntha_rasi));
870        candidates.push(VedicPlanet::get_ruler_of(birth_lagna_rasi));
871        candidates.push(VedicPlanet::get_ruler_of(annual_lagna_rasi));
872
873        if is_day {
874            if let Some(sun) = chart.planets.iter().find(|p| p.planet == VedicPlanet::Sun) {
875                candidates.push(VedicPlanet::get_ruler_of(sun.rasi));
876            }
877        } else {
878            if let Some(moon) = chart.planets.iter().find(|p| p.planet == VedicPlanet::Moon) {
879                candidates.push(VedicPlanet::get_ruler_of(moon.rasi));
880            }
881        }
882
883        candidates.push(Self::get_tri_rashi_pati(annual_lagna_rasi, is_day));
884
885        let eligible: Vec<VedicPlanet> = candidates
886            .into_iter()
887            .filter(|&p| {
888                if let Some(pos) = chart.planets.iter().find(|pos| pos.planet == p) {
889                    let dist = (pos.rasi as i16 - annual_lagna_rasi as i16 + 12) % 12;
890                    let aspect = Self::get_aspect_type((dist + 1) as u8);
891                    !matches!(aspect, TajikaAspectType::Sama)
892                        || p == VedicPlanet::get_ruler_of(annual_lagna_rasi)
893                } else {
894                    false
895                }
896            })
897            .collect();
898
899        if eligible.is_empty() {
900            return VedicPlanet::get_ruler_of(muntha_rasi);
901        }
902
903        let mut strongest = eligible[0];
904        let mut max_bala = 0;
905
906        for p in eligible {
907            let bala = TajikaBala::calculate_harsha_bala(chart, p);
908            if bala >= max_bala {
909                max_bala = bala;
910                strongest = p;
911            }
912        }
913
914        strongest
915    }

921    /// Harsha Bala (4 factors of Joy)
922    pub fn calculate_harsha_bala(chart: &VedicChart, planet: VedicPlanet) -> u32 {
923        let mut score = 0;
924        let p_pos = chart.planets.iter().find(|p| p.planet == planet);
925
926        if let Some(p) = p_pos {
927            let h = p.house_index;
928            match planet {
929                VedicPlanet::Sun if h == 9 => score += 1,
930                VedicPlanet::Moon if h == 4 => score += 1,
931                VedicPlanet::Mars if h == 6 => score += 1,
932                VedicPlanet::Mercury if h == 1 => score += 1,
933                VedicPlanet::Jupiter if h == 11 => score += 1,
934                VedicPlanet::Venus if h == 5 => score += 1,
935                VedicPlanet::Saturn if h == 12 => score += 1,
936                _ => {}
937            }
938
939            let lord = VedicPlanet::get_ruler_of(p.rasi);
940            if lord == planet || p.rasi == planet.exaltation_rasi() {
941                score += 1;
942            }
943
944            let is_day = chart.panchanga.is_day_birth;
945            match planet {
946                VedicPlanet::Sun | VedicPlanet::Mars | VedicPlanet::Jupiter if is_day => score += 1,
947                VedicPlanet::Moon | VedicPlanet::Venus | VedicPlanet::Saturn if !is_day => {
948                    score += 1
949                }
950                _ => {}
951            }
952
953            if [1, 4, 7, 10].contains(&h) {
954                score += 1;
955            }
956        }
957        score
958    }
