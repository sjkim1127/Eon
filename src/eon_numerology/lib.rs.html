39/// Reduces a number down to a single digit (1..=9) or Master Number (11, 22, 33)
40pub fn reduce_number(mut num: u32, preserve_master: bool) -> u8 {
41    while num > 9 {
42        if preserve_master && (num == 11 || num == 22 || num == 33) {
43            return num as u8;
44        }
45        let sum: u32 = num.to_string().chars().filter_map(|c| c.to_digit(10)).sum();
46        num = sum;
47    }
48    num as u8
49}

51/// Converts a letter to Pythagorean Numerology number (1..=9)
52pub fn letter_value(ch: char) -> u8 {
53    match ch.to_ascii_uppercase() {
54        'A' | 'J' | 'S' => 1,
55        'B' | 'K' | 'T' => 2,
56        'C' | 'L' | 'U' => 3,
57        'D' | 'M' | 'V' => 4,
58        'E' | 'N' | 'W' => 5,
59        'F' | 'O' | 'X' => 6,
60        'G' | 'P' | 'Y' => 7,
61        'H' | 'Q' | 'Z' => 8,
62        'I' | 'R' => 9,
63        _ => 0,
64    }
65}
66
67pub fn is_vowel(ch: char) -> bool {
68    matches!(ch.to_ascii_uppercase(), 'A' | 'E' | 'I' | 'O' | 'U')
69}

71/// Calculates Full Pythagorean Numerology Analysis
72pub fn calculate_numerology(
73    year: u32,
74    month: u32,
75    day: u32,
76    name_latin: Option<&str>,
77    target_year: u32,
78) -> NumerologyResult {
79    // 1. Life Path Number
80    let m_red = reduce_number(month, true) as u32;
81    let d_red = reduce_number(day, true) as u32;
82    let y_red = reduce_number(year, true) as u32;
83    let life_path = reduce_number(m_red + d_red + y_red, true);
84    let is_master_life_path = matches!(life_path, 11 | 22 | 33);
85
86    // 2. Birthday Number
87    let birthday = reduce_number(day, true);
88
89    // 3. Name Based Numbers (Expression, Soul Urge, Personality)
90    let (expression, soul_urge, personality) = if let Some(name) = name_latin {
91        let mut expr_sum = 0u32;
92        let mut soul_sum = 0u32;
93        let mut pers_sum = 0u32;
94
95        for ch in name.chars() {
96            let val = letter_value(ch);
97            if val > 0 {
98                expr_sum += val as u32;
99                if is_vowel(ch) {
100                    soul_sum += val as u32;
101                } else {
102                    pers_sum += val as u32;
103                }
104            }
105        }
106        (
107            reduce_number(expr_sum, true),
108            reduce_number(soul_sum, true),
109            reduce_number(pers_sum, true),
110        )
111    } else {
112        // Fallback proxy from birth values if latin name not provided
113        (
114            reduce_number(d_red + y_red, true),
115            reduce_number(m_red, true),
116            reduce_number(y_red, true),
117        )
118    };
119
120    // 4. 4 Pinnacles & Challenges
121    // Pinnacle 1: Month + Day
122    let p1 = reduce_number(month + day, true);
123    let c1 = reduce_number((month as i32 - day as i32).unsigned_abs(), false);
124
125    // Pinnacle 2: Day + Year
126    let p2 = reduce_number(day + year, true);
127    let c2 = reduce_number((day as i32 - year as i32).unsigned_abs(), false);
128
129    // Pinnacle 3: P1 + P2
130    let p3 = reduce_number(p1 as u32 + p2 as u32, true);
131    let c3 = reduce_number((c1 as i32 - c2 as i32).unsigned_abs(), false);
132
133    // Pinnacle 4: Month + Year
134    let p4 = reduce_number(month + year, true);
135    let c4 = reduce_number((month as i32 - year as i32).unsigned_abs(), false);
136
137    // Base pinnacle transition age: 36 - Life Path (single digit base)
138    let lp_base = reduce_number(life_path as u32, false) as u32;
139    let age1 = 36 - lp_base;
140    let age2 = age1 + 9;
141    let age3 = age2 + 9;
142
143    let pinnacles = vec![
144        PinnaclePeriod {
145            stage: 1,
146            pinnacle_number: p1,
147            challenge_number: c1,
148            start_age: 0,
149            end_age: age1,
150            description_ko: format!("제1정점: 기반 형성기 (정점수 {}, 과제수 {})", p1, c1),
151        },
152        PinnaclePeriod {
153            stage: 2,
154            pinnacle_number: p2,
155            challenge_number: c2,
156            start_age: age1,
157            end_age: age2,
158            description_ko: format!("제2정점: 개인적 도약기 (정점수 {}, 과제수 {})", p2, c2),
159        },
160        PinnaclePeriod {
161            stage: 3,
162            pinnacle_number: p3,
163            challenge_number: c3,
164            start_age: age2,
165            end_age: age3,
166            description_ko: format!(
167                "제3정점: 성숙과 사회적 결실기 (정점수 {}, 과제수 {})",
168                p3, c3
169            ),
170        },
171        PinnaclePeriod {
172            stage: 4,
173            pinnacle_number: p4,
174            challenge_number: c4,
175            start_age: age3,
176            end_age: 100,
177            description_ko: format!(
178                "제4정점: 완성과 지혜의 통달기 (정점수 {}, 과제수 {})",
179                p4, c4
180            ),
181        },
182    ];
183
184    // 5. Personal Year Number
185    let personal_year = reduce_number(month + day + target_year, false);
186
187    let summary_ko = format!(
188        "생애여정수: {}, 표현수: {}, 영혼수: {}, 개인 연운수: {} (피타고라스 수비 진동 분석 완료)",
189        life_path, expression, soul_urge, personal_year
190    );
191
192    NumerologyResult {
193        core: NumerologyCoreNumbers {
194            life_path,
195            expression,
196            soul_urge,
197            personality,
198            birthday,
199        },
200        pinnacles,
201        personal_year,
202        is_master_life_path,
203        summary_ko,
204    }
205}
