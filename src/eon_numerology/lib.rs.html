9/// Letter-to-number table used for the name based numbers.
10#[derive(Debug, Clone, Copy, PartialEq, Eq, Default, Serialize, Deserialize)]
11#[serde(rename_all = "lowercase")]
12pub enum NumerologySystem {
13    /// A=1 .. I=9, J=1 .. (default).
14    #[default]
15    Pythagorean,
16    /// Chaldean table: values 1..=8 only (9 is sacred and never assigned to a letter).
17    Chaldean,
18}
19
20impl NumerologySystem {
21    pub fn label(self) -> &'static str {
22        match self {
23            NumerologySystem::Pythagorean => "Pythagorean",
24            NumerologySystem::Chaldean => "Chaldean",
25        }
26    }
27
28    /// Highest number a letter can have in this system.
29    pub fn max_letter_value(self) -> u8 {
30        match self {
31            NumerologySystem::Pythagorean => 9,
32            NumerologySystem::Chaldean => 8,
33        }
34    }
35
36    /// Value of a Latin letter in this system (0 for anything that is not A-Z).
37    pub fn letter_value(self, ch: char) -> u8 {
38        match self {
39            NumerologySystem::Pythagorean => letter_value(ch),
40            NumerologySystem::Chaldean => chaldean_letter_value(ch),
41        }
42    }
43}
44
45impl fmt::Display for NumerologySystem {
46    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
47        f.write_str(match self {
48            NumerologySystem::Pythagorean => "pythagorean",
49            NumerologySystem::Chaldean => "chaldean",
50        })
51    }
52}
53
54impl FromStr for NumerologySystem {
55    type Err = String;
56
57    fn from_str(s: &str) -> Result<Self, Self::Err> {
58        match s.trim().to_ascii_lowercase().as_str() {
59            "pythagorean" => Ok(NumerologySystem::Pythagorean),
60            "chaldean" => Ok(NumerologySystem::Chaldean),
61            other => Err(format!(
62                "Invalid numerology system '{other}'. Expected pythagorean or chaldean"
63            )),
64        }
65    }
66}

222/// Reduces a number down to a single digit (1..=9) or Master Number (11, 22, 33)
223pub fn reduce_number(mut num: u32, preserve_master: bool) -> u8 {
224    while num > 9 {
225        if preserve_master && (num == 11 || num == 22 || num == 33) {
226            return num as u8;
227        }
228        let sum: u32 = num.to_string().chars().filter_map(|c| c.to_digit(10)).sum();
229        num = sum;
230    }
231    num as u8
232}

234/// Converts a letter to Pythagorean Numerology number (1..=9)
235pub fn letter_value(ch: char) -> u8 {
236    match ch.to_ascii_uppercase() {
237        'A' | 'J' | 'S' => 1,
238        'B' | 'K' | 'T' => 2,
239        'C' | 'L' | 'U' => 3,
240        'D' | 'M' | 'V' => 4,
241        'E' | 'N' | 'W' => 5,
242        'F' | 'O' | 'X' => 6,
243        'G' | 'P' | 'Y' => 7,
244        'H' | 'Q' | 'Z' => 8,
245        'I' | 'R' => 9,
246        _ => 0,
247    }
248}

250/// Converts a letter to its Chaldean number (1..=8; 9 is never assigned).
251pub fn chaldean_letter_value(ch: char) -> u8 {
252    match ch.to_ascii_uppercase() {
253        'A' | 'I' | 'J' | 'Q' | 'Y' => 1,
254        'B' | 'K' | 'R' => 2,
255        'C' | 'G' | 'L' | 'S' => 3,
256        'D' | 'M' | 'T' => 4,
257        'E' | 'H' | 'N' | 'X' => 5,
258        'U'..='W' => 6,
259        'O' | 'Z' => 7,
260        'F' | 'P' => 8,
261        _ => 0,
262    }
263}

265pub fn is_vowel(ch: char) -> bool {
266    matches!(ch.to_ascii_uppercase(), 'A' | 'E' | 'I' | 'O' | 'U')
267}

269const KARMIC_DEBT_NUMBERS: [u32; 4] = [13, 14, 16, 19];

271/// Looks for a karmic debt number (13, 14, 16, 19) in `num` and in every intermediate sum
272/// produced while reducing it (e.g. 19 -> 10 -> 1 contains the debt 19).
273fn find_karmic_debt(mut num: u32, preserve_master: bool) -> Option<u8> {
274    loop {
275        if KARMIC_DEBT_NUMBERS.contains(&num) {
276            return Some(num as u8);
277        }
278        if num <= 9 || (preserve_master && matches!(num, 11 | 22 | 33)) {
279            return None;
280        }
281        num = num.to_string().chars().filter_map(|c| c.to_digit(10)).sum();
282    }
283}

337/// Full numerology analysis for a letter system and a target date.
338pub fn calculate_numerology_ext(
339    year: u32,
340    month: u32,
341    day: u32,
342    name_latin: Option<&str>,
343    target: TargetDate,
344    system: NumerologySystem,
345) -> NumerologyResult {
346    let target_year = target.year;
347    let mut karmic_debts: Vec<KarmicDebt> = Vec::new();
348
349    // 1. Life Path Number
350    let m_red = reduce_number(month, true) as u32;
351    let d_red = reduce_number(day, true) as u32;
352    let y_red = reduce_number(year, true) as u32;
353    let life_path_sum = m_red + d_red + y_red;
354    let life_path = reduce_number(life_path_sum, true);
355    let is_master_life_path = matches!(life_path, 11 | 22 | 33);
356    if let Some(d) = find_karmic_debt(life_path_sum, true) {
357        karmic_debts.push(karmic_debt(d, "lifePath"));
358    }
359
360    // 2. Birthday Number
361    let birthday = reduce_number(day, true);
362    if let Some(d) = find_karmic_debt(day, true) {
363        karmic_debts.push(karmic_debt(d, "birthday"));
364    }
365
366    // 3. Name Based Numbers (Expression, Soul Urge, Personality, Maturity, ...)
367    let mut expr_sum = 0u32;
368    let mut soul_sum = 0u32;
369    let mut pers_sum = 0u32;
370    let mut counts = [0u32; 9];
371    if let Some(name) = name_latin {
372        for ch in name.chars() {
373            let val = system.letter_value(ch);
374            if val > 0 {
375                expr_sum += val as u32;
376                counts[val as usize - 1] += 1;
377                if is_vowel(ch) {
378                    soul_sum += val as u32;
379                } else {
380                    pers_sum += val as u32;
381                }
382            }
383        }
384    }
385    let has_name = expr_sum > 0;
386    let reduce_opt = |sum: u32| (sum > 0).then(|| reduce_number(sum, true));
387    let expression = reduce_opt(expr_sum);
388    let soul_urge = reduce_opt(soul_sum);
389    let personality = reduce_opt(pers_sum);
390    for (sum, source) in [
391        (expr_sum, "expression"),
392        (soul_sum, "soulUrge"),
393        (pers_sum, "personality"),
394    ] {
395        if let Some(d) = find_karmic_debt(sum, true) {
396            karmic_debts.push(karmic_debt(d, source));
397        }
398    }
399    let maturity = expression.map(|e| reduce_number(life_path as u32 + e as u32, true));
400    let name_insights = if has_name {
401        name_latin.and_then(|n| name_insights(n, system, counts))
402    } else {
403        None
404    };
405
406    // 4. 4 Pinnacles & Challenges
407    // The cycle calculations use the reduced birth-date components. This is
408    // the convention used by the Pythagorean pinnacle/challenge references;
409    // using raw 28/1990 values produces different, non-canonical challenges.
410    let m_base = reduce_number(month, false) as u32;
411    let d_base = reduce_number(day, false) as u32;
412    let y_base = reduce_number(year, false) as u32;
413
414    // Pinnacle 1: reduced month + reduced day
415    let p1 = reduce_number(m_base + d_base, true);
416    let c1 = reduce_number(m_base.abs_diff(d_base), false);
417
418    // Pinnacle 2: reduced day + reduced year
419    let p2 = reduce_number(d_base + y_base, true);
420    let c2 = reduce_number(d_base.abs_diff(y_base), false);
421
422    // Pinnacle 3: P1 + P2
423    let p3 = reduce_number(p1 as u32 + p2 as u32, true);
424    let c3 = reduce_number((c1 as i32 - c2 as i32).unsigned_abs(), false);
425
426    // Pinnacle 4: reduced month + reduced year
427    let p4 = reduce_number(m_base + y_base, true);
428    let c4 = reduce_number(m_base.abs_diff(y_base), false);
429
430    // Base pinnacle transition age: 36 - Life Path (single digit base)
431    let lp_base = reduce_number(life_path as u32, false) as u32;
432    let age1 = 36 - lp_base;
433    let age2 = age1 + 9;
434    let age3 = age2 + 9;
435
436    let pinnacles = vec![
437        PinnaclePeriod {
438            stage: 1,
439            pinnacle_number: p1,
440            challenge_number: c1,
441            start_age: 0,
442            end_age: age1,
443            description_ko: format!(
444                "Pinnacle 1: Foundation Era (Pinnacle {}, Challenge {})",
445                p1, c1
446            ),
447            description_en: format!(
448                "Pinnacle 1: Foundation Era (Pinnacle {}, Challenge {})",
449                p1, c1
450            ),
451        },
452        PinnaclePeriod {
453            stage: 2,
454            pinnacle_number: p2,
455            challenge_number: c2,
456            start_age: age1,
457            end_age: age2,
458            description_ko: format!(
459                "Pinnacle 2: Expansion Era (Pinnacle {}, Challenge {})",
460                p2, c2
461            ),
462            description_en: format!(
463                "Pinnacle 2: Expansion Era (Pinnacle {}, Challenge {})",
464                p2, c2
465            ),
466        },
467        PinnaclePeriod {
468            stage: 3,
469            pinnacle_number: p3,
470            challenge_number: c3,
471            start_age: age2,
472            end_age: age3,
473            description_ko: format!(
474                "Pinnacle 3: Maturation & Achievement Era (Pinnacle {}, Challenge {})",
475                p3, c3
476            ),
477            description_en: format!(
478                "Pinnacle 3: Maturation & Achievement Era (Pinnacle {}, Challenge {})",
479                p3, c3
480            ),
481        },
482        PinnaclePeriod {
483            stage: 4,
484            pinnacle_number: p4,
485            challenge_number: c4,
486            start_age: age3,
487            end_age: 100,
488            description_ko: format!(
489                "Pinnacle 4: Completion & Wisdom Era (Pinnacle {}, Challenge {})",
490                p4, c4
491            ),
492            description_en: format!(
493                "Pinnacle 4: Completion & Wisdom Era (Pinnacle {}, Challenge {})",
494                p4, c4
495            ),
496        },
497    ];
498
499    // 5. Personal Year / Month / Day Number
500    // Cycle numbers are reduced to a single digit (master numbers are not preserved).
501    let personal_year = reduce_number(month + day + target_year, false);
502    let personal_month = target
503        .month
504        .map(|m| reduce_number(personal_year as u32 + m, false));
505    let personal_day = match (personal_month, target.month, target.day) {
506        (Some(pm), Some(_), Some(d)) => Some(reduce_number(pm as u32 + d, false)),
507        _ => None,
508    };
509
510    let opt = |v: Option<u8>| {
511        v.map(|v| v.to_string())
512            .unwrap_or_else(|| "N/A".to_string())
513    };
514    let debts_en = if karmic_debts.is_empty() {
515        String::new()
516    } else {
517        format!(
518            ", Karmic Debt: {}",
519            karmic_debts
520                .iter()
521                .map(|k| format!("{}/{} ({})", k.debt_number, k.reduces_to, k.source))
522                .collect::<Vec<_>>()
523                .join(" ")
524        )
525    };
526    let cycle_en = match (personal_month, personal_day) {
527        (Some(m), Some(d)) => format!(", Personal Month: {m}, Personal Day: {d}"),
528        (Some(m), None) => format!(", Personal Month: {m}"),
529        _ => String::new(),
530    };
531    let summary_en = format!(
532        "Life Path: {}, Expression: {}, Soul Urge: {}, Maturity: {}, Personal Year: {}{}{} ({} Numerology analysis complete)",
533        life_path,
534        opt(expression),
535        opt(soul_urge),
536        opt(maturity),
537        personal_year,
538        cycle_en,
539        debts_en,
540        system.label()
541    );
542    let system_ko = match system {
543        NumerologySystem::Pythagorean => "피타고라스",
544        NumerologySystem::Chaldean => "칼데안",
545    };
546    let cycle_ko = match (personal_month, personal_day) {
547        (Some(m), Some(d)) => format!(", 개인 월수: {m}, 개인 일수: {d}"),
548        (Some(m), None) => format!(", 개인 월수: {m}"),
549        _ => String::new(),
550    };
551    let debts_ko = if karmic_debts.is_empty() {
552        String::new()
553    } else {
554        format!(
555            ", 카르마 부채: {}",
556            karmic_debts
557                .iter()
558                .map(|k| format!("{}/{}({})", k.debt_number, k.reduces_to, k.source))
559                .collect::<Vec<_>>()
560                .join(" ")
561        )
562    };
563    let summary_ko = format!(
564        "생명수(Life Path): {}, 표현수: {}, 영혼수: {}, 성숙수: {}, 개인 연수: {}{}{} ({} 수비학 분석 완료)",
565        life_path,
566        opt(expression),
567        opt(soul_urge),
568        opt(maturity),
569        personal_year,
570        cycle_ko,
571        debts_ko,
572        system_ko
573    );
574
575    NumerologyResult {
576        system,
577        core: NumerologyCoreNumbers {
578            life_path,
579            expression,
580            soul_urge,
581            personality,
582            birthday,
583            maturity,
584        },
585        pinnacles,
586        personal_year,
587        personal_month,
588        personal_day,
589        is_master_life_path,
590        karmic_debts,
591        name_insights,
592        summary_ko,
593        summary_en,
594    }
595}

597/// Hidden passion, balance, cornerstone/capstone and missing numbers of a name.
598/// `counts[i]` is how often the number `i + 1` occurs among the letters.
599fn name_insights(name: &str, system: NumerologySystem, counts: [u32; 9]) -> Option<NameInsights> {
600    // Words are separated by whitespace; a word without any A-Z letter is skipped.
601    let words: Vec<Vec<(char, u8)>> = name
602        .split_whitespace()
603        .map(|w| {
604            w.chars()
605                .filter_map(|c| {
606                    let v = system.letter_value(c);
607                    (v > 0).then(|| (c.to_ascii_uppercase(), v))
608                })
609                .collect::<Vec<_>>()
610        })
611        .filter(|w| !w.is_empty())
612        .collect();
613    let first = words.first()?;
614
615    let max_count = counts.iter().copied().max().unwrap_or(0);
616    let hidden_passions = if max_count == 0 {
617        Vec::new()
618    } else {
619        (1..=9u8)
620            .filter(|n| counts[*n as usize - 1] == max_count)
621            .collect()
622    };
623    let missing_numbers = (1..=system.max_letter_value())
624        .filter(|n| counts[*n as usize - 1] == 0)
625        .collect();
626    let initials: u32 = words.iter().map(|w| w[0].1 as u32).sum();
627    let (cl, cv) = first[0];
628    let (pl, pv) = first[first.len() - 1];
629
630    Some(NameInsights {
631        number_counts: counts,
632        hidden_passions,
633        balance: reduce_number(initials, false),
634        cornerstone: NameLetter {
635            letter: cl,
636            value: cv,
637        },
638        capstone: NameLetter {
639            letter: pl,
640            value: pv,
641        },
642        missing_numbers,
643    })
644}

646/// Weights of the pair scores in [`compare_numerology`].
647const LIFE_PATH_WEIGHT: u32 = 50;
648const EXPRESSION_WEIGHT: u32 = 30;
649const SOUL_URGE_WEIGHT: u32 = 20;
650
651/// Temperament group of a number: 0 = {1,5,7} independent seekers, 1 = {2,4,8} builders,
652/// 2 = {3,6,9} expressive humanitarians. Master numbers use their base digit (11 -> 2, ...).
653fn number_group(n: u8) -> u8 {
654    match reduce_number(n as u32, false) {
655        1 | 5 | 7 => 0,
656        2 | 4 | 8 => 1,
657        _ => 2,
658    }
659}
660
661fn pair_score(a: u8, b: u8) -> NumberPairScore {
662    let (ba, bb) = (
663        reduce_number(a as u32, false),
664        reduce_number(b as u32, false),
665    );
666    let (ga, gb) = (number_group(a), number_group(b));
667    let (score, relation, en, ko) = if ba == bb {
668        (
669            85,
670            "same-number",
671            "Same vibration: deep mutual understanding, but shared blind spots too.",
672            "같은 진동수: 서로를 깊이 이해하지만 약점도 같이 공유합니다.",
673        )
674    } else if ga == gb {
675        (
676            75,
677            "same-group",
678            "Same temperament group (1-5-7, 2-4-8 or 3-6-9): natural rapport and similar rhythm.",
679            "같은 기질 그룹(1-5-7, 2-4-8, 3-6-9): 자연스러운 공감대와 비슷한 리듬.",
680        )
681    } else if (ga.min(gb), ga.max(gb)) == (0, 2) {
682        (
683            65,
684            "complementary-groups",
685            "Seekers (1-5-7) and expressives (3-6-9): lively, stimulating pairing that needs shared direction.",
686            "탐구형(1-5-7)과 표현형(3-6-9): 활기와 자극이 있으나 공통의 방향이 필요합니다.",
687        )
688    } else if (ga.min(gb), ga.max(gb)) == (1, 2) {
689        (
690            60,
691            "complementary-groups",
692            "Builders (2-4-8) and expressives (3-6-9): warmth meets stability, with differing priorities.",
693            "구축형(2-4-8)과 표현형(3-6-9): 따뜻함과 안정감이 만나지만 우선순위가 다릅니다.",
694        )
695    } else {
696        (
697            50,
698            "contrasting-groups",
699            "Seekers (1-5-7) and builders (2-4-8): freedom versus security; needs deliberate compromise.",
700            "탐구형(1-5-7)과 구축형(2-4-8): 자유와 안정의 대비로 의식적인 타협이 필요합니다.",
701        )
702    };
703    NumberPairScore {
704        number_a: a,
705        number_b: b,
706        score,
707        relation: relation.to_string(),
708        description_en: en.to_string(),
709        description_ko: ko.to_string(),
710    }
711}

713/// Two-person numerology compatibility.
714///
715/// **This is a heuristic, not a canonical table.** Each pair of core numbers (Life Path,
716/// Expression, Soul Urge) is scored from the temperament groups {1,5,7}, {2,4,8}, {3,6,9}
717/// (master numbers use their base digit): identical number 85, same group 75, groups
718/// {1,5,7}+{3,6,9} 65, {2,4,8}+{3,6,9} 60, {1,5,7}+{2,4,8} 50. The total is the weighted
719/// average Life Path 50%, Expression 30%, Soul Urge 20%; when a name is missing for either
720/// person, only the available pairs are used and the weights are renormalised.
721pub fn compare_numerology(a: &NumerologyResult, b: &NumerologyResult) -> NumerologyCompatibility {
722    let life_path = pair_score(a.core.life_path, b.core.life_path);
723    let both = |x: Option<u8>, y: Option<u8>| match (x, y) {
724        (Some(x), Some(y)) => Some(pair_score(x, y)),
725        _ => None,
726    };
727    let expression = both(a.core.expression, b.core.expression);
728    let soul_urge = both(a.core.soul_urge, b.core.soul_urge);
729
730    let mut weighted = LIFE_PATH_WEIGHT * life_path.score as u32;
731    let mut weights = LIFE_PATH_WEIGHT;
732    if let Some(p) = &expression {
733        weighted += EXPRESSION_WEIGHT * p.score as u32;
734        weights += EXPRESSION_WEIGHT;
735    }
736    if let Some(p) = &soul_urge {
737        weighted += SOUL_URGE_WEIGHT * p.score as u32;
738        weights += SOUL_URGE_WEIGHT;
739    }
740    let total_score = (weighted as f64 * 10.0 / weights as f64).round() / 10.0;
741
742    let methodology = "Heuristic: pair scores from temperament groups {1,5,7} {2,4,8} {3,6,9} \
743(same number 85, same group 75, 1-5-7 + 3-6-9 65, 2-4-8 + 3-6-9 60, 1-5-7 + 2-4-8 50); \
744weights Life Path 50 / Expression 30 / Soul Urge 20, renormalised when names are missing."
745        .to_string();
746    let used = 1 + expression.is_some() as u8 + soul_urge.is_some() as u8;
747    let summary_en = format!(
748        "Numerology compatibility {:.1}/100 (Life Path {} & {}{}{}; {} pair(s) compared, heuristic score)",
749        total_score,
750        a.core.life_path,
751        b.core.life_path,
752        expression
753            .as_ref()
754            .map(|p| format!(", Expression {} & {}", p.number_a, p.number_b))
755            .unwrap_or_default(),
756        soul_urge
757            .as_ref()
758            .map(|p| format!(", Soul Urge {} & {}", p.number_a, p.number_b))
759            .unwrap_or_default(),
760        used
761    );
762    let summary_ko = format!(
763        "수비학 궁합 {:.1}/100 (생명수 {} & {}{}{}; {}개 쌍 비교, 휴리스틱 점수)",
764        total_score,
765        a.core.life_path,
766        b.core.life_path,
767        expression
768            .as_ref()
769            .map(|p| format!(", 표현수 {} & {}", p.number_a, p.number_b))
770            .unwrap_or_default(),
771        soul_urge
772            .as_ref()
773            .map(|p| format!(", 영혼수 {} & {}", p.number_a, p.number_b))
774            .unwrap_or_default(),
775        used
776    );
777
778    NumerologyCompatibility {
779        system: a.system,
780        life_path,
781        expression,
782        soul_urge,
783        total_score,
784        methodology,
785        summary_en,
786        summary_ko,
787    }
788}
