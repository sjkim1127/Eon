1//! Timezone utilities
2//!
3//! Calculates standard meridian longitude from IANA timezone strings.
4
5use chrono::TimeZone;
6use chrono_tz::Tz;
7
8/// Calculates the standard meridian (longitude in degrees) from an IANA timezone string
9/// at an optional historical reference datetime.
10///
11/// Example:
12/// - "Asia/Seoul" at 2024 -> 135.0 (UTC+9)
13/// - "Asia/Seoul" at 1955 -> 127.5 (UTC+8:30, used historically between 1954-03-21 and 1961-08-09)
14pub fn standard_meridian_from_tz_at(timezone: &str, dt_opt: Option<chrono::NaiveDateTime>) -> f64 {
15    if let Ok(tz) = timezone.parse::<Tz>() {
16        let ref_dt = match dt_opt {
17            Some(dt) => tz
18                .from_local_datetime(&dt)
19                .earliest()
20                .or_else(|| tz.from_local_datetime(&dt).latest()),
21            None => tz.with_ymd_and_hms(2024, 1, 15, 12, 0, 0).single(),
22        };
23        if let Some(dt) = ref_dt {
24            use chrono_tz::OffsetComponents;
25            let base_offset_secs = dt.offset().base_utc_offset().num_seconds() as f64;
26            return (base_offset_secs / 3600.0) * 15.0;
27        }
28    }
29    135.0 // Fallback: Standard Meridian
30}
31
32/// Calculates the standard meridian (longitude in degrees) from an IANA timezone string (defaulting to modern 2024 reference).
33pub fn standard_meridian_from_tz(timezone: &str) -> f64 {
34    standard_meridian_from_tz_at(timezone, None)
35}

37#[cfg(test)]
38mod tests {
39    use super::*;
40
41    #[test]
42    fn test_asia_seoul() {
43        let m = standard_meridian_from_tz("Asia/Seoul");
44        assert!((m - 135.0).abs() < 0.01);
45    }
46
47    #[test]
48    fn test_asia_seoul_historical_1955() {
49        use chrono::NaiveDate;
50        let dt = NaiveDate::from_ymd_opt(1955, 6, 15)
51            .unwrap()
52            .and_hms_opt(12, 0, 0)
53            .unwrap();
54        let m = standard_meridian_from_tz_at("Asia/Seoul", Some(dt));
55        assert!(
56            (m - 127.5).abs() < 0.01,
57            "1955 Korea must use UTC+8:30 (127.5 deg meridian), got {}",
58            m
59        );
60    }
61
62    #[test]
63    fn test_america_new_york() {
64        let m = standard_meridian_from_tz("America/New_York");
65        assert!((m - (-75.0)).abs() < 0.01);
66    }
67
68    #[test]
69    fn test_invalid_fallback() {
70        let m = standard_meridian_from_tz("invalid/tz");
71        assert!((m - 135.0).abs() < 0.01);
72    }
