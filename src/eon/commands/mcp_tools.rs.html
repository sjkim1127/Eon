24/// Typed access to the `arguments` object of a `tools/call` request.
25struct Args<'a>(&'a Value);
26
27impl Args<'_> {
28    fn str(&self, key: &str) -> Option<String> {
29        self.0
30            .get(key)
31            .and_then(Value::as_str)
32            .map(str::trim)
33            .filter(|s| !s.is_empty())
34            .map(str::to_string)
35    }
36
37    fn require(&self, key: &str) -> Result<String> {
38        self.str(key)
39            .ok_or_else(|| anyhow!("missing required argument '{key}'"))
40    }
41
42    fn bool(&self, key: &str) -> bool {
43        self.0.get(key).and_then(Value::as_bool).unwrap_or(false)
44    }
45
46    fn f64(&self, key: &str) -> Option<f64> {
47        self.0.get(key).and_then(Value::as_f64)
48    }
49
50    fn i32(&self, key: &str) -> Result<Option<i32>> {
51        match self.0.get(key).and_then(Value::as_i64) {
52            Some(v) => Ok(Some(
53                i32::try_from(v).map_err(|_| anyhow!("'{key}' is out of range"))?,
54            )),
55            None => Ok(None),
56        }
57    }
58
59    /// Birth fields of one person; `prefix` is "" or "partner_".
60    fn birth(&self, prefix: &str, default_gender: &str) -> Result<BirthArgs> {
61        let k = |name: &str| format!("{prefix}{name}");
62        Ok(BirthArgs {
63            birth: self.require(&k("birth"))?,
64            gender: self
65                .str(&k("gender"))
66                .unwrap_or_else(|| default_gender.to_string()),
67            city: self.str(&k("city")),
68            lat: self.f64(&k("lat")),
69            lon: self.f64(&k("lon")),
70            tz: self.str(&k("timezone")).or_else(|| self.str(&k("tz"))),
71            lunar: self.bool(&k("lunar")),
72            leap: self.bool(&k("leap")),
73            name: if prefix.is_empty() {
74                self.str("name")
75            } else {
76                None
77            },
78        })
79    }
80
81    fn partner(&self) -> Result<PartnerArgs> {
82        let b = self.birth("partner_", "f")?;
83        Ok(PartnerArgs {
84            birth: b.birth,
85            gender: b.gender,
86            city: b.city,
87            lat: b.lat,
88            lon: b.lon,
89            tz: b.tz,
90            lunar: b.lunar,
91            leap: b.leap,
92        })
93    }
94
95    fn format(&self) -> Result<EngineFormat> {
96        match self.str("format").as_deref() {
97            None | Some("text") => Ok(EngineFormat::Text),
98            Some("json") => Ok(EngineFormat::Json),
99            Some(other) => Err(anyhow!(
100                "Invalid format '{other}'. Expected 'text' or 'json'"
101            )),
102        }
103    }
104}

108fn birth_props(prefix: &str, who: &str, with_name: bool) -> Map<String, Value> {
109    let mut m = Map::new();
110    let mut put = |name: &str, schema: Value| {
111        m.insert(format!("{prefix}{name}"), schema);
112    };
113    put(
114        "birth",
115        json!({
116            "type": "string",
117            "description": format!("{who} birth date and time as 'YYYY-MM-DD HH:MM' (e.g. '1990-05-15 10:30'). A date alone means the birth time is unknown, which the time-dependent engines (Vedic, Western, Zi Wei Dou Shu, Human Design, QiMen) reject.")
118        }),
119    );
120    put(
121        "gender",
122        json!({ "type": "string", "enum": ["m", "f"], "description": format!("{who} gender") }),
123    );
124    put(
125        "lunar",
126        json!({ "type": "boolean", "description": "True if the birth date is a lunar calendar date" }),
127    );
128    put(
129        "leap",
130        json!({ "type": "boolean", "description": "True if the lunar birth month is a leap month" }),
131    );
132    put(
133        "city",
134        json!({ "type": "string", "description": "Birth city (resolves coordinates and timezone), e.g. 'Seoul', 'Tokyo', 'New York'" }),
135    );
136    put(
137        "lat",
138        json!({ "type": "number", "description": "Latitude (-90 to 90)" }),
139    );
140    put(
141        "lon",
142        json!({ "type": "number", "description": "Longitude (-180 to 180)" }),
143    );
144    put(
145        "timezone",
146        json!({ "type": "string", "description": "IANA timezone, e.g. 'Asia/Seoul'" }),
147    );
148    if with_name {
149        m.insert(
150            "name".to_string(),
151            json!({ "type": "string", "description": "Latin-letter name (Numerology Expression/Soul Urge, report title)" }),
152        );
153    }
154    m
155}

157fn format_prop() -> Value {
158    json!({
159        "type": "string",
160        "enum": ["text", "json"],
161        "default": "text",
162        "description": "'text' for a compact summary, 'json' for the full structured result (large)"
163    })
164}
165
166fn tool(name: &str, description: &str, props: Map<String, Value>, required: &[&str]) -> Value {
167    let mut props = props;
168    props.insert("format".to_string(), format_prop());
169    json!({
170        "name": name,
171        "description": description,
172        "inputSchema": { "type": "object", "properties": props, "required": required }
173    })
174}
175
176fn with(mut base: Map<String, Value>, extra: Vec<(&str, Value)>) -> Map<String, Value> {
177    for (k, v) in extra {
178        base.insert(k.to_string(), v);
179    }
180    base
181}

304/// Runs a tool. `None` means the tool name is not one of these tools.
305pub fn call(name: &str, args: &Value) -> Option<Result<String, String>> {
306    let a = Args(args);
307    let result: Result<String> = match name {
308        "eon_saju" => (|| {
309            let format = a.format()?;
310            let cmd = SajuCommand {
311                birth: a.birth("", "m")?,
312                night_rat: a.bool("night_rat"),
313                format,
314            };
315            render(super::saju::execute(&cmd)?, format)
316        })(),
317        "eon_saju_luck" => (|| {
318            let format = a.format()?;
319            let cmd = TransitCommand {
320                birth: a.birth("", "m")?,
321                at: a.str("at"),
322                night_rat: a.bool("night_rat"),
323                format,
324            };
325            render(super::transit::execute(&cmd)?, format)
326        })(),
327        "eon_report" => (|| {
328            let format = a.format()?;
329            let theme = match a.require("theme")?.as_str() {
330                "wealth" => ThemeArg::Wealth,
331                "love" => ThemeArg::Love,
332                "health" => ThemeArg::Health,
333                other => {
334                    return Err(anyhow!(
335                        "Invalid theme '{other}'. Expected wealth, love or health"
336                    ))
337                }
338            };
339            let cmd = ReportCommand {
340                birth: a.birth("", "m")?,
341                theme,
342                format,
343            };
344            render(super::report::execute(&cmd)?, format)
345        })(),
346        "eon_vedic" => (|| {
347            let format = a.format()?;
348            let cmd = VedicCommand {
349                birth: a.birth("", "m")?,
350                ayanamsa: a.str("ayanamsa"),
351                target_year: a.i32("target_year")?,
352                year_type: a.str("year_type"),
353                format,
354            };
355            render(super::vedic::execute(&cmd)?, format)
356        })(),
357        "eon_western" => (|| {
358            let format = a.format()?;
359            let cmd = WesternCommand {
360                birth: a.birth("", "m")?,
361                house_system: a
362                    .str("house_system")
363                    .unwrap_or_else(|| "Placidus".to_string()),
364                progress_to: a.str("progress_to"),
365                transit_at: a.str("transit_at"),
366                solar_arc_to: a.str("solar_arc_to"),
367                profection_at: a.str("profection_at"),
368                solar_return: a.i32("solar_return")?,
369                lunar_return_after: a.str("lunar_return_after"),
370                return_city: a.str("return_city"),
371                format,
372            };
373            let chosen = [
374                cmd.progress_to.is_some(),
375                cmd.transit_at.is_some(),
376                cmd.solar_arc_to.is_some(),
377                cmd.profection_at.is_some(),
378                cmd.solar_return.is_some(),
379                cmd.lunar_return_after.is_some(),
380            ]
381            .iter()
382            .filter(|set| **set)
383            .count();
384            if chosen > 1 {
385                return Err(anyhow!(
386                    "Use only one of progress_to, transit_at, solar_arc_to, profection_at, solar_return, lunar_return_after"
387                ));
388            }
389            render(super::western::execute(&cmd)?, format)
390        })(),
391        "eon_zwds" => (|| {
392            let format = a.format()?;
393            let cmd = ZwdsCommand {
394                birth: a.birth("", "m")?,
395                target_year: a.i32("target_year")?,
396                format,
397            };
398            render(super::zwds::execute(&cmd)?, format)
399        })(),
400        "eon_human_design" => (|| {
401            let format = a.format()?;
402            let return_type = match a.str("return").as_deref() {
403                None => None,
404                Some("solar") => Some(HdReturn::Solar),
405                Some("saturn") => Some(HdReturn::Saturn),
406                Some("chiron") => Some(HdReturn::Chiron),
407                Some(other) => {
408                    return Err(anyhow!(
409                        "Invalid return '{other}'. Expected solar, saturn or chiron"
410                    ))
411                }
412            };
413            let cmd = HdCommand {
414                birth: a.birth("", "m")?,
415                transit_at: a.str("transit_at"),
416                return_type,
417                year: a.i32("year")?,
418                format,
419            };
420            if cmd.transit_at.is_some() && cmd.return_type.is_some() {
421                return Err(anyhow!("Use either 'transit_at' or 'return', not both"));
422            }
423            render(super::hd::execute(&cmd)?, format)
424        })(),
425        "eon_human_design_penta" => (|| {
426            let format = a.format()?;
427            let members = args
428                .get("members")
429                .and_then(Value::as_array)
430                .ok_or_else(|| anyhow!("missing required argument 'members'"))?
431                .iter()
432                .map(|v| {
433                    v.as_str()
434                        .map(str::to_string)
435                        .ok_or_else(|| anyhow!("'members' must be an array of strings"))
436                })
437                .collect::<Result<Vec<_>>>()?;
438            let cmd = PentaCommand {
439                members,
440                city: a.str("city"),
441                format,
442            };
443            render(super::penta::execute(&cmd)?, format)
444        })(),
445        "eon_qimen" => (|| {
446            let format = a.format()?;
447            let cmd = QimenCommand {
448                birth: a.birth("", "m")?,
449                format,
450            };
451            render(super::qimen::execute(&cmd)?, format)
452        })(),
453        "eon_numerology" => (|| {
454            let format = a.format()?;
455            let target_year = match a.i32("target_year")? {
456                Some(y) => {
457                    Some(u32::try_from(y).map_err(|_| anyhow!("'target_year' must be positive"))?)
458                }
459                None => None,
460            };
461            let cmd = NumerologyCommand {
462                birth: a.birth("", "m")?,
463                target_year,
464                format,
465            };
466            render(super::numerology::execute(&cmd)?, format)
467        })(),
468        "eon_iching" => (|| {
469            let format = a.format()?;
470            let cmd = IchingCommand {
471                birth: a.birth("", "m")?,
472                format,
473            };
474            render(super::iching::execute(&cmd)?, format)
475        })(),
476        "eon_compatibility" => (|| {
477            let format = a.format()?;
478            let system = match a.str("system").as_deref() {
479                None | Some("all") => CompatSystem::All,
480                Some("saju") => CompatSystem::Saju,
481                Some("vedic") => CompatSystem::Vedic,
482                Some("western") => CompatSystem::Western,
483                Some("hd") => CompatSystem::Hd,
484                Some(other) => {
485                    return Err(anyhow!(
486                        "Invalid system '{other}'. Expected all, saju, vedic, western or hd"
487                    ))
488                }
489            };
490            let cmd = CompatCommand {
491                birth: a.birth("", "m")?,
492                partner: a.partner()?,
493                system,
494                house_system: a
495                    .str("house_system")
496                    .unwrap_or_else(|| "Placidus".to_string()),
497                format,
498            };
499            render(super::compat::execute(&cmd)?, format)
500        })(),
501        _ => return None,
502    };
503
504    Some(result.map_err(|e| format!("{e}")))
