26#[derive(Args, Debug, Clone)]
27pub struct BirthArgs {
28    /// Birth datetime: 'YYYY-MM-DD HH:MM' or 'YYYY-MM-DD' (e.g. '1990-05-15 10:30')
29    #[arg(short, long)]
30    pub birth: String,
31
32    /// Gender: 'm', 'male', 'f', 'female' (default: male)
33    #[arg(short, long, default_value = "m")]
34    pub gender: String,
35
36    /// Birth city or region (e.g. 'seoul', 'busan', 'tokyo', 'new york', 'la', 'london')
37    /// Automatically resolves latitude, longitude, and timezone.
38    #[arg(short, long)]
39    pub city: Option<String>,
40
41    /// Latitude (default: 37.5665 for Seoul, or automatically resolved from --city)
42    #[arg(long)]
43    pub lat: Option<f64>,
44
45    /// Longitude (default: 126.9780 for Seoul, or automatically resolved from --city)
46    #[arg(long)]
47    pub lon: Option<f64>,
48
49    /// Timezone (default: Asia/Seoul, or automatically resolved from --city)
50    #[arg(long)]
51    pub tz: Option<String>,
52
53    /// Whether the birth date is Lunar calendar
54    #[arg(long)]
55    pub lunar: bool,
56
57    /// Whether the lunar birth month is a leap month
58    #[arg(long)]
59    pub leap: bool,
60
61    /// Full name in English/Latin letters for Numerology
62    #[arg(long)]
63    pub name: Option<String>,
64}

66impl BirthArgs {
67    pub fn is_male(&self) -> bool {
68        let g = self.gender.trim().to_lowercase();
69        !matches!(g.as_str(), "f" | "female" | "2" | "4")
70    }
71
72    pub fn to_analysis_input(&self) -> Result<AnalysisInput> {
73        let raw = self.birth.trim();
74        let (date_part, time_part) = if let Some((d, t)) = raw.split_once(' ') {
75            (d, t)
76        } else if let Some((d, t)) = raw.split_once('T') {
77            (d, t)
78        } else {
79            (raw, "12:00")
80        };
81
82        let date_seps = ['-', '.', '/'];
83        let parts: Vec<&str> = date_part
84            .split(|c| date_seps.contains(&c))
85            .filter(|s| !s.is_empty())
86            .collect();
87
88        if parts.len() != 3 {
89            return Err(anyhow!(
90                "Invalid date format: '{}'. Expected 'YYYY-MM-DD [HH:MM]'",
91                self.birth
92            ));
93        }
94
95        let year: i32 = parts[0]
96            .parse()
97            .map_err(|_| anyhow!("Invalid year: {}", parts[0]))?;
98        let month: u32 = parts[1]
99            .parse()
100            .map_err(|_| anyhow!("Invalid month: {}", parts[1]))?;
101        let day: u32 = parts[2]
102            .parse()
103            .map_err(|_| anyhow!("Invalid day: {}", parts[2]))?;
104
105        let time_parts: Vec<&str> = time_part.split(':').collect();
106        let hour: u32 = time_parts.first().unwrap_or(&"12").parse().unwrap_or(12);
107        let minute: u32 = time_parts.get(1).unwrap_or(&"0").parse().unwrap_or(0);
108
109        let (lat, lon, tz, resolved_city) = if let Some(ref city_query) = self.city {
110            match eon_core::lookup_city(city_query) {
111                Some(info) => {
112                    let lat = self.lat.unwrap_or(info.latitude);
113                    let lon = self.lon.unwrap_or(info.longitude);
114                    let tz = self.tz.clone().unwrap_or_else(|| info.timezone.to_string());
115                    (lat, lon, tz, Some(info.name.to_string()))
116                }
117                None => {
118                    if let (Some(lat), Some(lon)) = (self.lat, self.lon) {
119                        let tz = self.tz.clone().unwrap_or_else(|| "Asia/Seoul".to_string());
120                        (lat, lon, tz, Some(city_query.clone()))
121                    } else {
122                        return Err(anyhow!(
123                            "Unknown city '{}'. Please provide a recognized city (e.g. 'Seoul', 'Busan', 'Tokyo', 'New York', 'London') or specify explicit coordinates with --lat and --lon.",
124                            city_query
125                        ));
126                    }
127                }
128            }
129        } else {
130            let lat = self.lat.unwrap_or(37.5665);
131            let lon = self.lon.unwrap_or(126.9780);
132            let tz = self.tz.clone().unwrap_or_else(|| "Asia/Seoul".to_string());
133            (lat, lon, tz, None)
134        };
135
136        Ok(AnalysisInput {
137            year,
138            month,
139            day,
140            hour,
141            minute,
142            is_lunar: self.lunar,
143            is_leap_month: self.leap,
144            lat,
145            lon,
146            timezone: tz,
147            city: resolved_city,
148        })
149    }
150}
