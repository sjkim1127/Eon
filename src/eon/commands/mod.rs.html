27#[derive(Args, Debug, Clone)]
28pub struct BirthArgs {
29    /// Birth datetime: 'YYYY-MM-DD HH:MM' or 'YYYY-MM-DD' (e.g. '1990-05-15 10:30')
30    #[arg(short, long)]
31    pub birth: String,
32
33    /// Gender: 'm', 'male', 'f', 'female' (default: male)
34    #[arg(short, long, default_value = "m")]
35    pub gender: String,
36
37    /// Birth city or region (e.g. 'seoul', 'busan', 'tokyo', 'new york', 'la', 'london')
38    /// Automatically resolves latitude, longitude, and timezone.
39    #[arg(short, long)]
40    pub city: Option<String>,
41
42    /// Latitude (default: 37.5665 for Seoul, or automatically resolved from --city)
43    #[arg(long)]
44    pub lat: Option<f64>,
45
46    /// Longitude (default: 126.9780 for Seoul, or automatically resolved from --city)
47    #[arg(long)]
48    pub lon: Option<f64>,
49
50    /// Timezone (default: Asia/Seoul, or automatically resolved from --city)
51    #[arg(long)]
52    pub tz: Option<String>,
53
54    /// Whether the birth date is Lunar calendar
55    #[arg(long)]
56    pub lunar: bool,
57
58    /// Whether the lunar birth month is a leap month
59    #[arg(long)]
60    pub leap: bool,
61
62    /// Full name in English/Latin letters for Numerology
63    #[arg(long)]
64    pub name: Option<String>,
65}

67impl BirthArgs {
68    pub fn parse_gender(&self) -> Result<eon_core::Gender> {
69        let g = self.gender.trim().to_lowercase();
70        match g.as_str() {
71            "m" | "male" | "1" | "3" => Ok(eon_core::Gender::Male),
72            "f" | "female" | "2" | "4" => Ok(eon_core::Gender::Female),
73            _ => Err(anyhow!(
74                "Invalid gender: '{}'. Expected 'm', 'male', 'f', or 'female'",
75                self.gender
76            )),
77        }
78    }
79
80    pub fn is_male(&self) -> Result<bool> {
81        self.parse_gender().map(|g| g == eon_core::Gender::Male)
82    }
83
84    pub fn to_analysis_input(&self) -> Result<AnalysisInput> {
85        // Validate gender strictly
86        let _ = self.parse_gender()?;
87
88        let raw = self.birth.trim();
89        let (date_part, time_part) = if let Some((d, t)) = raw.split_once(' ') {
90            (d, t)
91        } else if let Some((d, t)) = raw.split_once('T') {
92            (d, t)
93        } else {
94            (raw, "12:00")
95        };
96
97        let date_seps = ['-', '.', '/'];
98        let parts: Vec<&str> = date_part
99            .split(|c| date_seps.contains(&c))
100            .filter(|s| !s.is_empty())
101            .collect();
102
103        if parts.len() != 3 {
104            return Err(anyhow!(
105                "Invalid date format: '{}'. Expected 'YYYY-MM-DD [HH:MM]'",
106                self.birth
107            ));
108        }
109
110        let year: i32 = parts[0]
111            .parse()
112            .map_err(|_| anyhow!("Invalid year: {}", parts[0]))?;
113        let month: u32 = parts[1]
114            .parse()
115            .map_err(|_| anyhow!("Invalid month: {}", parts[1]))?;
116        let day: u32 = parts[2]
117            .parse()
118            .map_err(|_| anyhow!("Invalid day: {}", parts[2]))?;
119
120        if !(1..=12).contains(&month) {
121            return Err(anyhow!("Invalid month: {}. Month must be 1-12", month));
122        }
123        if !(1..=31).contains(&day) {
124            return Err(anyhow!("Invalid day: {}. Day must be 1-31", day));
125        }
126
127        let time_parts: Vec<&str> = time_part.split(':').collect();
128        if time_parts.is_empty() || time_parts.len() > 3 {
129            return Err(anyhow!(
130                "Invalid time format: '{}'. Expected 'HH:MM'",
131                time_part
132            ));
133        }
134        let hour: u32 = time_parts[0].parse().map_err(|_| {
135            anyhow!(
136                "Invalid hour '{}' in birth time '{}'",
137                time_parts[0],
138                self.birth
139            )
140        })?;
141        let minute: u32 = if time_parts.len() >= 2 {
142            time_parts[1].parse().map_err(|_| {
143                anyhow!(
144                    "Invalid minute '{}' in birth time '{}'",
145                    time_parts[1],
146                    self.birth
147                )
148            })?
149        } else {
150            0
151        };
152
153        if hour > 23 || minute > 59 {
154            return Err(anyhow!(
155                "Birth time out of range: '{:02}:{:02}'. Hour must be 0-23, minute 0-59",
156                hour,
157                minute
158            ));
159        }
160
161        let (lat, lon, tz, resolved_city) = if let Some(ref city_query) = self.city {
162            match eon_core::lookup_city(city_query) {
163                Some(info) => {
164                    let lat = self.lat.unwrap_or(info.latitude);
165                    let lon = self.lon.unwrap_or(info.longitude);
166                    let tz = self.tz.clone().unwrap_or_else(|| info.timezone.to_string());
167                    (lat, lon, tz, Some(info.name.to_string()))
168                }
169                None => {
170                    if let (Some(lat), Some(lon)) = (self.lat, self.lon) {
171                        let tz = self.tz.clone().unwrap_or_else(|| "Asia/Seoul".to_string());
172                        (lat, lon, tz, Some(city_query.clone()))
173                    } else {
174                        return Err(anyhow!(
175                            "Unknown city '{}'. Please provide a recognized city (e.g. 'Seoul', 'Busan', 'Tokyo', 'New York', 'London') or specify explicit coordinates with --lat and --lon.",
176                            city_query
177                        ));
178                    }
179                }
180            }
181        } else {
182            let lat = self.lat.unwrap_or(37.5665);
183            let lon = self.lon.unwrap_or(126.9780);
184            let tz = self.tz.clone().unwrap_or_else(|| "Asia/Seoul".to_string());
185            (lat, lon, tz, None)
186        };
187
188        Ok(AnalysisInput {
189            year,
190            month,
191            day,
192            hour,
193            minute,
194            is_lunar: self.lunar,
195            is_leap_month: self.leap,
196            lat,
197            lon,
198            timezone: tz,
199            city: resolved_city,
200        })
201    }
202}

208    #[test]
209    fn test_valid_birth_args() {
210        let args = BirthArgs {
211            birth: "1990-05-15 10:30".to_string(),
212            gender: "m".to_string(),
213            city: Some("Seoul".to_string()),
214            lat: None,
215            lon: None,
216            tz: None,
217            lunar: false,
218            leap: false,
219            name: None,
220        };
221        let input = args.to_analysis_input().unwrap();
222        assert_eq!(input.year, 1990);
223        assert_eq!(input.month, 5);
224        assert_eq!(input.day, 15);
225        assert_eq!(input.hour, 10);
226        assert_eq!(input.minute, 30);
227        assert!(args.is_male().unwrap());
228    }
