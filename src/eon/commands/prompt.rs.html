1use anyhow::Result;
2use clap::Args;
3use eon_service::dto::*;
4use eon_service::facade;
5
6use super::BirthArgs;
7use crate::format::agent::format_agent_context;
8
9#[derive(Args, Debug)]
10pub struct PromptCommand {
11    #[command(flatten)]
12    pub birth: BirthArgs,
13
14    /// Add system prompt instruction prefix for LLMs
15    #[arg(short, long)]
16    pub system: bool,
17}
18
19pub fn run(cmd: PromptCommand) -> Result<()> {
20    let input = cmd.birth.to_analysis_input()?;
21    let is_male = cmd.birth.is_male();
22
23    let saju_in = SajuAnalysisInput::new(input.clone(), is_male, false, Some(false));
24    let vedic_in = VedicAnalysisInput::new(input.clone(), Some(false), None);
25    let western_in = WesternAnalysisInput::new(input.clone(), "Placidus".to_string());
26    let zwds_in = ZwdsAnalysisInput::new(input.clone(), is_male, None);
27    let hd_in = HumanDesignAnalysisInput::new(input.clone());
28    let qimen_in = QimenAnalysisInput::new(input.clone(), is_male);
29    let num_in = NumerologyAnalysisInput::new(input.clone(), cmd.birth.name.clone(), None);
30
31    let saju_res = facade::analyze_saju(saju_in)?;
32    let vedic_res = facade::analyze_vedic(vedic_in)?;
33    let western_res = facade::analyze_western(western_in).ok();
34    let zwds_res = facade::analyze_zwds(zwds_in).ok();
35    let hd_res = facade::analyze_human_design(hd_in).ok();
36    let qimen_res = facade::analyze_qimen(qimen_in).ok();
37    let num_res = facade::analyze_numerology(num_in).ok();
38
39    let omni_input = OmniDestinyTierInput {
40        saju: saju_res.clone(),
41        vedic: vedic_res.clone(),
42        western: western_res.clone(),
43        zwds: zwds_res.clone(),
44        human_design: hd_res.clone(),
45        qimen: qimen_res.clone(),
46        numerology: num_res.clone(),
47        transit: None,
48    };
49
50    let tier_res = facade::analyze_destiny_tier_omni(omni_input)?;
51
52    if cmd.system {
53        println!("You are an expert Destiny Intelligence AI Counselor powered by the Eon 8-Engine Framework.");
54        println!("Below is the verified multi-disciplinary astrological and metaphysical calculation context for the client.");
55        println!("Synthesize these cross-engine insights to provide deep, actionable, compassionate, and precise counseling.\n");
56    }
57
58    let text = format_agent_context(
59        &input,
60        Some(&saju_res),
61        Some(&vedic_res),
62        western_res.as_ref(),
63        zwds_res.as_ref(),
64        hd_res.as_ref(),
65        qimen_res.as_ref(),
66        num_res.as_ref(),
67        Some(&tier_res),
68    );
69
70    println!("{}", text);
71    Ok(())
72}
