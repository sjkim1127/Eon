10/// Letter-to-number table for the name numbers.
11#[derive(ValueEnum, Clone, Copy, Debug, PartialEq, Eq, Default)]
12pub enum NumerologySystemArg {

14    Pythagorean,

18impl From<NumerologySystemArg> for NumerologySystem {
19    fn from(arg: NumerologySystemArg) -> Self {
20        match arg {
21            NumerologySystemArg::Pythagorean => NumerologySystem::Pythagorean,
22            NumerologySystemArg::Chaldean => NumerologySystem::Chaldean,
23        }
24    }
25}

49/// Parses 'YYYY-MM-DD' (the calendar validity is checked by the service).
50pub fn parse_target_date(raw: &str) -> Result<(u32, u32, u32)> {
51    let invalid = || anyhow!("Invalid target date '{raw}'. Expected 'YYYY-MM-DD'");
52    let mut parts = raw.trim().splitn(3, '-');
53    let mut next = || -> Result<u32> {
54        parts
55            .next()
56            .ok_or_else(invalid)?
57            .parse()
58            .map_err(|_| invalid())
59    };
60    Ok((next()?, next()?, next()?))
61}

63pub fn execute(cmd: &NumerologyCommand) -> Result<Rendered> {
64    let input = cmd.birth.to_analysis_input()?;
65    let mut num_in = NumerologyAnalysisInput::new(input, cmd.birth.name.clone(), cmd.target_year)
66        .with_system(cmd.system.into());
67    if let Some(raw) = &cmd.target_date {
68        let (y, m, d) = parse_target_date(raw)?;
69        num_in = num_in.with_target_date(y, m, d);
70    }
71    let res = facade::analyze_numerology(num_in)?;
72
73    let needs_name = "N/A (pass --name with a Latin name)";
74    let core = &res.result.core;
75    let opt = |v: Option<u8>| {
76        v.map(|v| v.to_string())
77            .unwrap_or_else(|| needs_name.to_string())
78    };
79    let mut t = String::new();
80    writeln!(t, "== {} Numerology ==", res.result.system.label())?;
81    writeln!(
82        t,
83        "Life Path Number: {}{}",
84        core.life_path,
85        if res.result.is_master_life_path {
86            " (Master Number)"
87        } else {
88            ""
89        }
90    )?;
91    writeln!(t, "Expression Number: {}", opt(core.expression))?;
92    writeln!(t, "Soul Urge Number: {}", opt(core.soul_urge))?;
93    writeln!(t, "Personality Number: {}", opt(core.personality))?;
94    writeln!(t, "Birthday Number: {}", core.birthday)?;
95    writeln!(t, "Maturity Number: {}", opt(core.maturity))?;
96    writeln!(t, "Personal Year Vibration: {}", res.result.personal_year)?;
97    if let Some(m) = res.result.personal_month {
98        writeln!(t, "Personal Month Vibration: {m}")?;
99    }
100    if let Some(d) = res.result.personal_day {
101        writeln!(t, "Personal Day Vibration: {d}")?;
102    }
103    if res.result.karmic_debts.is_empty() {
104        writeln!(t, "Karmic Debt: none")?;
105    } else {
106        writeln!(t, "Karmic Debt:")?;
107        for k in &res.result.karmic_debts {
108            writeln!(
109                t,
110                "- {}/{} in {}: {}",
111                k.debt_number, k.reduces_to, k.source, k.meaning_en
112            )?;
113        }
114    }
115    if let Some(ni) = &res.result.name_insights {
116        let list = |v: &[u8]| {
117            if v.is_empty() {
118                "none".to_string()
119            } else {
120                v.iter().map(u8::to_string).collect::<Vec<_>>().join(", ")
121            }
122        };
123        writeln!(t, "Hidden Passion: {}", list(&ni.hidden_passions))?;
124        writeln!(t, "Balance Number: {}", ni.balance)?;
125        writeln!(
126            t,
127            "Cornerstone: {} ({}), Capstone: {} ({})",
128            ni.cornerstone.letter, ni.cornerstone.value, ni.capstone.letter, ni.capstone.value
129        )?;
130        writeln!(
131            t,
132            "Missing Numbers (Karmic Lessons): {}",
133            list(&ni.missing_numbers)
134        )?;
135    }
136    writeln!(t, "Four Pinnacle Cycles:")?;
137    for p in &res.result.pinnacles {
138        writeln!(t, "- {}", p.description_en)?;
139    }
140
141    Rendered::new(&res, t)
142}
