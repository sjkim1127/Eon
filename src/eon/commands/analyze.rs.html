1use anyhow::Result;
2use clap::Args;
3use eon_service::dto::*;
4use eon_service::facade;
5use serde::Serialize;
6
7use super::{BirthArgs, OutputFormat};
8use crate::format::agent::format_agent_context;
9use crate::format::markdown::format_markdown_report;
10
11#[derive(Args, Debug)]
12pub struct AnalyzeCommand {
13    #[command(flatten)]
14    pub birth: BirthArgs,
15
16    /// Output format: agent (LLM Context), json (raw structured), markdown, text
17    #[arg(short, long, value_enum, default_value = "agent")]
18    pub format: OutputFormat,
19}
20
21#[derive(Serialize)]
22pub struct FullOmniResult {
23    pub input: AnalysisInput,
24    pub tier: TierResult,
25    pub saju: SajuAnalysisOutput,
26    pub vedic: VedicAnalysisOutput,
27    pub western: Option<WesternAnalysisOutput>,
28    pub zwds: Option<ZwdsAnalysisOutput>,
29    pub human_design: Option<HumanDesignAnalysisOutput>,
30    pub qimen: Option<QimenAnalysisOutput>,
31    pub numerology: Option<NumerologyAnalysisOutput>,
32}
33
34pub fn run(cmd: AnalyzeCommand) -> Result<()> {
35    let input = cmd.birth.to_analysis_input()?;
36    let is_male = cmd.birth.is_male();
37
38    // 1. Individual engine analyses
39    let saju_in = SajuAnalysisInput::new(input.clone(), is_male, false, Some(false));
40    let vedic_in = VedicAnalysisInput::new(input.clone(), Some(false), None);
41    let western_in = WesternAnalysisInput::new(input.clone(), "Placidus".to_string());
42    let zwds_in = ZwdsAnalysisInput::new(input.clone(), is_male, None);
43    let hd_in = HumanDesignAnalysisInput::new(input.clone());
44    let qimen_in = QimenAnalysisInput::new(input.clone(), is_male);
45    let num_in = NumerologyAnalysisInput::new(input.clone(), cmd.birth.name.clone(), None);
46
47    let saju_res = facade::analyze_saju(saju_in)?;
48    let vedic_res = facade::analyze_vedic(vedic_in)?;
49    let western_res = facade::analyze_western(western_in).ok();
50    let zwds_res = facade::analyze_zwds(zwds_in).ok();
51    let hd_res = facade::analyze_human_design(hd_in).ok();
52    let qimen_res = facade::analyze_qimen(qimen_in).ok();
53    let num_res = facade::analyze_numerology(num_in).ok();
54
55    // 2. Omni Destiny Tier 7.0 Synthesis
56    let omni_input = OmniDestinyTierInput {
57        saju: saju_res.clone(),
58        vedic: vedic_res.clone(),
59        western: western_res.clone(),
60        zwds: zwds_res.clone(),
61        human_design: hd_res.clone(),
62        qimen: qimen_res.clone(),
63        numerology: num_res.clone(),
64        transit: None,
65    };
66
67    let tier_res = facade::analyze_destiny_tier_omni(omni_input)?;
68
69    // 3. Output by requested format
70    match cmd.format {
71        OutputFormat::Json => {
72            let full = FullOmniResult {
73                input,
74                tier: tier_res,
75                saju: saju_res,
76                vedic: vedic_res,
77                western: western_res,
78                zwds: zwds_res,
79                human_design: hd_res,
80                qimen: qimen_res,
81                numerology: num_res,
82            };
83            println!("{}", serde_json::to_string_pretty(&full)?);
84        }
85        OutputFormat::Agent | OutputFormat::Text => {
86            let text = format_agent_context(
87                &input,
88                Some(&saju_res),
89                Some(&vedic_res),
90                western_res.as_ref(),
91                zwds_res.as_ref(),
92                hd_res.as_ref(),
93                qimen_res.as_ref(),
94                num_res.as_ref(),
95                Some(&tier_res),
96            );
97            println!("{}", text);
98        }
99        OutputFormat::Markdown => {
100            let md = format_markdown_report(
101                &input,
102                Some(&saju_res),
103                Some(&vedic_res),
104                western_res.as_ref(),
105                zwds_res.as_ref(),
106                hd_res.as_ref(),
107                qimen_res.as_ref(),
108                num_res.as_ref(),
109                Some(&tier_res),
110            );
111            println!("{}", md);
112        }
113    }
114
115    Ok(())
116}
