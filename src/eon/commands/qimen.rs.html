1use anyhow::Result;
2use clap::Args;
3use eon_service::dto::QimenAnalysisInput;
4use eon_service::facade;
5
6use super::{BirthArgs, OutputFormat};
7
8#[derive(Args, Debug)]
9pub struct QimenCommand {
10    #[command(flatten)]
11    pub birth: BirthArgs,
12
13    /// Output format: json (default), text
14    #[arg(short, long, value_enum, default_value = "json")]
15    pub format: OutputFormat,
16}
17
18pub fn run(cmd: QimenCommand) -> Result<()> {
19    let input = cmd.birth.to_analysis_input()?;
20    let is_male = cmd.birth.is_male()?;
21    let qimen_in = QimenAnalysisInput::new(input, is_male);
22    let res = facade::analyze_qimen(qimen_in)?;
23
24    match cmd.format {
25        OutputFormat::Json => {
26            println!("{}", serde_json::to_string_pretty(&res)?);
27        }
28        _ => {
29            println!("== QiMen DunJia ==");
30            println!(
31                "Ju (Structure): {} Ju {}",
32                if res.report.pan.is_yin_ju {
33                    "Yin Dun (陰遁)"
34                } else {
35                    "Yang Dun (陽遁)"
36                },
37                res.report.pan.ju_number
38            );
39            if let Some(chief) = &res.report.pan.value_chief_star {
40                println!("Chief Star (Zhi Fu): {:?}", chief);
41            }
42            if let Some(envoy) = &res.report.pan.value_envoy_door {
43                println!("Envoy Door (Zhi Shi): {:?}", envoy);
44            }
45            println!("Palaces Deployed: {} palaces", res.report.pan.palaces.len());
46        }
47    }
48
49    Ok(())
50}
