1use eon_service::dto::*;
2
3#[allow(clippy::too_many_arguments)]
4pub fn format_markdown_report(
5    input: &AnalysisInput,
6    saju: Option<&SajuAnalysisOutput>,
7    vedic: Option<&VedicAnalysisOutput>,
8    western: Option<&WesternAnalysisOutput>,
9    zwds: Option<&ZwdsAnalysisOutput>,
10    hd: Option<&HumanDesignAnalysisOutput>,
11    qimen: Option<&QimenAnalysisOutput>,
12    num: Option<&NumerologyAnalysisOutput>,
13    tier: Option<&TierResult>,
14    handbook: Option<&InterpretationHandbook>,
15) -> String {
16    let mut md = String::new();
17
18    md.push_str("# Eon: 8-Engine Destiny Intelligence Report\n\n");
19    let loc_str = if let Some(ref c) = input.city {
20        format!(
21            "**Location**: {c} (Lat: {:.4}, Lon: {:.4})",
22            input.lat, input.lon
23        )
24    } else {
25        format!("**Coordinates**: ({:.4}, {:.4})", input.lat, input.lon)
26    };
27
28    let std_meridian = eon_core::standard_meridian_from_tz(&input.timezone);
29    let offset_mins = ((input.lon - std_meridian) * 4.0).round() as i32;
30    let sign = if offset_mins >= 0 { "+" } else { "" };
31
32    let solar_str = if let Some(s) = saju {
33        let dst_str = if s.meta.is_dst { ", DST: Active" } else { "" };
34        format!(
35            "**True Solar Time**: {} ({}{}m{})",
36            s.meta.corrected_time, sign, offset_mins, dst_str
37        )
38    } else {
39        format!("**Solar Time Offset**: {}{}m", sign, offset_mins)
40    };
41
42    md.push_str(&format!(
43        "**Birth**: {:04}-{:02}-{:02} {:02}:{:02} | **Timezone**: {} | {} | {}\n\n",
44        input.year,
45        input.month,
46        input.day,
47        input.hour,
48        input.minute,
49        input.timezone,
50        loc_str,
51        solar_str
52    ));
53
54    if let Some(t) = tier {
55        md.push_str("## Omni Destiny Tier 7.0 Summary\n");
56        md.push_str("| Metric | Score / Grade |\n|---|---|\n");
57        md.push_str(&format!(
58            "| **Overall Destiny Tier** | **{}** ({:.1} pts) |\n",
59            t.destiny_tier.grade, t.destiny_tier_score
60        ));
61        md.push_str(&format!(
62            "| **Birth Chart Potential** | **{}** ({:.1} pts) |\n",
63            t.potential_tier.grade, t.potential_score
64        ));
65        md.push_str(&format!(
66            "| **Current Transit Score** | {:.1} pts |\n",
67            t.current_score
68        ));
69        md.push_str(&format!("| **Growth Gap** | +{:.1} pts |\n", t.growth_gap));
70        md.push_str(&format!(
71            "| **Risk / Stability Level** | {} |\n\n",
72            t.risk_level
73        ));
74
75        if !t.detailed_components.is_empty() {
76            md.push_str("### 15 Detailed Component Weights\n");
77            md.push_str(
78                "| Component | Weight | Score | Primary Analysis Factor |\n|---|---|---|---|\n",
79            );
80            for c in &t.detailed_components {
81                let reason = c.reasons.first().cloned().unwrap_or_default();
82                md.push_str(&format!(
83                    "| {} | {:.0}% | {:.1} | {} |\n",
84                    c.label,
85                    c.weight * 100.0,
86                    c.score,
87                    reason
88                ));
89            }
90            md.push('\n');
91        }
92
93        if !t.quantum_synergies.is_empty() {
94            md.push_str("### 8D Quantum Cross Synergies\n");
95            for s in &t.quantum_synergies {
96                let badge = if s.score_delta >= 0.0 { "[+]" } else { "[-]" };
97                md.push_str(&format!(
98                    "- {} **{}** ({:+.0}pt): {}\n",
99                    badge, s.title, s.score_delta, s.description
100                ));
101            }
102            md.push('\n');
103        }
104    }
105
106    if let Some(s) = saju {
107        md.push_str("## 1. Four Pillars of Destiny (Saju / BaZi)\n");
108        let p = &s.report.pillars;
109        md.push_str(&format!(
110            "- **Chart**: Year: {} ({:?}-{:?}) / Month: {} ({:?}-{:?}) / Day: {} ({:?}-{:?}) / Hour: {} ({:?}-{:?})\n",
111            p.year,
112            p.year.stem,
113            p.year.branch,
114            p.month,
115            p.month.stem,
116            p.month.branch,
117            p.day,
118            p.day.stem,
119            p.day.branch,
120            p.hour,
121            p.hour.stem,
122            p.hour.branch
123        ));
124        md.push_str(&format!(
125            "- **Strength**: {:?} (Score: {:.1})\n",
126            s.report.strength.strength_type, s.report.strength.strength_score
127        ));
128        md.push_str(&format!(
129            "- **Structure**: {:?} | **Primary Yongshin**: {:?}\n\n",
130            s.report.structure.structure, s.report.yongshin.primary
131        ));
132    }
133
134    if let Some(v) = vedic {
135        md.push_str("## 2. Vedic Astrology (Jyotish)\n");
136        md.push_str(&format!(
137            "- **Ascendant (Lagna)**: {:?}\n",
138            v.chart.ascendant
139        ));
140        md.push_str(&format!(
141            "- **Moon Nakshatra**: {}\n",
142            v.report.nakshatra_info
143        ));
144        md.push_str(&format!(
145            "- **Detected Yogas**: {}\n\n",
146            v.report.yogas.len()
147        ));
148    }
149
150    if let Some(w) = western {
151        md.push_str("## 3. Western Astrology\n");
152        md.push_str(&format!(
153            "- **Planets**: {} | **Aspects**: {}\n",
154            w.result.planets.len(),
155            w.result.aspects.len()
156        ));
157        md.push_str(&format!(
158            "- **Aspect Configurations**: {}\n\n",
159            w.result.aspect_patterns.len()
160        ));
161    }
162
163    if let Some(z) = zwds {
164        md.push_str("## 4. Zi Wei Dou Shu\n");
165        let soul = &z.chart.palaces[z.chart.soul_idx];
166        md.push_str(&format!("- **Soul Palace**: {}\n", soul.name.english()));
167        md.push_str(&format!(
168            "- **Stars in Soul Palace**: {}\n\n",
169            soul.stars.len()
170        ));
171    }
172
173    if let Some(h) = hd {
174        md.push_str("## 5. Human Design\n");
175        md.push_str(&format!(
176            "- **Chart Type**: {} | **Profile**: {}\n",
177            h.result.chart_type, h.result.profile
178        ));
179        md.push_str(&format!(
180            "- **Defined Centers**: {}\n\n",
181            h.result.defined_centers.len()
182        ));
183    }
184
185    if let Some(q) = qimen {
186        md.push_str("## 6. QiMen DunJia\n");
187        md.push_str(&format!(
188            "- **Chart**: {} Ju {} (9 Palaces Deployed)\n\n",
189            if q.report.pan.is_yin_ju {
190                "Yin Dun"
191            } else {
192                "Yang Dun"
193            },
194            q.report.pan.ju_number
195        ));
196    }
197
198    if let Some(n) = num {
199        md.push_str("## 7. Pythagorean Numerology\n");
200        md.push_str(&format!(
201            "- **Life Path Number**: **{}**{}\n",
202            n.result.core.life_path,
203            if n.result.is_master_life_path {
204                " (Master Number)"
205            } else {
206                ""
207            }
208        ));
209        md.push_str(&format!(
210            "- **Expression**: {} | **Soul Urge**: {} | **Personal Year**: {}\n\n",
211            n.result.core.expression, n.result.core.soul_urge, n.result.personal_year
212        ));
213    }
214
215    if let Some(hb) = handbook {
216        md.push_str("## 8. Domain Interpretation Handbook & Archetypal Decoder\n");
217        md.push_str(&format!("> **Persona**: {}\n\n", hb.archetypal_persona));
218        md.push_str(&format!("{}\n\n", hb.core_essence_summary));
219
220        if !hb.symbols.is_empty() {
221            md.push_str("### Pinpoint Classical Symbol Decoders\n\n");
222            for s in &hb.symbols {
223                md.push_str(&format!("#### {} ({})\n", s.symbol, s.engine));
224                md.push_str(&format!(
225                    "- **Classical Grounding**: {}\n",
226                    s.classical_grounding
227                ));
228                md.push_str(&format!(
229                    "- **Modern Translation**: {}\n",
230                    s.modern_translation
231                ));
232                md.push_str(&format!("- **Shadow & Risk**: {}\n", s.shadow_and_caution));
233                md.push_str(&format!("- **Tactical Remedy**: {}\n\n", s.tactical_remedy));
234            }
235        }
236
237        if !hb.tactical_dos.is_empty() || !hb.tactical_donts.is_empty() {
238            md.push_str("### Tactical Directives\n\n");
239            md.push_str("**DO**:\n");
240            for d in &hb.tactical_dos {
241                md.push_str(&format!("- [x] {}\n", d));
242            }
243            md.push_str("\n**DON'T**:\n");
244            for d in &hb.tactical_donts {
245                md.push_str(&format!("- [ ] {}\n", d));
246            }
247            md.push('\n');
248        }
249    }
250
251    md
252}
