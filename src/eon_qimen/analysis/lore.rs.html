1//! Classical lookup tables of 奇門遁甲: palace branches and elements, the auspice of doors, stars
2//! and deities, and the stem-over-stem formations (格局).
3//!
4//! Sources are the common classical texts (《煙波釣叟歌》, 《奇門遁甲統宗》) as summarised in modern
5//! manuals. Schools disagree on the details, so every choice below is documented in
6//! `docs/CALCULATION.md` section 7a and the scores are heuristics, not verdicts.
7
8use crate::core::{Deity, Door, Palace, Star};
9use eon_saju::core::branch::EarthlyBranch;
10use eon_saju::core::element::Element;
11use eon_saju::core::stem::HeavenlyStem;
12use serde::{Deserialize, Serialize};

14/// Earthly branches that sit in a palace (後天八卦 / Later Heaven arrangement).
15pub fn palace_branches(palace: Palace) -> &'static [EarthlyBranch] {
16    use EarthlyBranch::*;
17    match palace {
18        Palace::Kan1 => &[Zi],
19        Palace::Gen8 => &[Chou, Yin],
20        Palace::Zhen3 => &[Mao],
21        Palace::Xun4 => &[Chen, Si],
22        Palace::Li9 => &[Wu],
23        Palace::Kun2 => &[Wei, Shen],
24        Palace::Dui7 => &[You],
25        Palace::Qian6 => &[Xu, Hai],
26        Palace::Center5 => &[],
27    }
28}
29
30/// The palace a branch belongs to (never the centre).
31pub fn palace_of_branch(branch: EarthlyBranch) -> Palace {
32    use EarthlyBranch::*;
33    match branch {
34        Zi => Palace::Kan1,
35        Chou | Yin => Palace::Gen8,
36        Mao => Palace::Zhen3,
37        Chen | Si => Palace::Xun4,
38        Wu => Palace::Li9,
39        Wei | Shen => Palace::Kun2,
40        You => Palace::Dui7,
41        Xu | Hai => Palace::Qian6,
42    }
43}
44
45pub fn palace_element(palace: Palace) -> Element {
46    match palace {
47        Palace::Kan1 => Element::Water,
48        Palace::Kun2 | Palace::Center5 | Palace::Gen8 => Element::Earth,
49        Palace::Zhen3 | Palace::Xun4 => Element::Wood,
50        Palace::Qian6 | Palace::Dui7 => Element::Metal,
51        Palace::Li9 => Element::Fire,
52    }
53}
54
55pub fn door_element(door: Door) -> Element {
56    match door {
57        Door::Xiu => Element::Water,
58        Door::Sheng | Door::Si => Element::Earth,
59        Door::Shang | Door::Du => Element::Wood,
60        Door::Jing => Element::Fire,
61        Door::Jing2 | Door::Kai => Element::Metal,
62    }
63}

65/// Auspice of a door: the three good doors 開休生 (+2), the three bad doors 傷驚死 (-2, death -3);
66/// 景 and 杜 are "middling" (0).
67pub fn door_score(door: Door) -> i32 {
68    match door {
69        Door::Kai | Door::Xiu | Door::Sheng => 2,
70        Door::Jing | Door::Du => 0,
71        Door::Shang | Door::Jing2 => -2,
72        Door::Si => -3,
73    }
74}
75
76/// Auspice of a star: 輔 心 (+2), 禽 任 沖 (+1), 英 柱 (-1), 蓬 芮 (-2).
77pub fn star_score(star: Star) -> i32 {
78    match star {
79        Star::Fu | Star::Xin => 2,
80        Star::Qin | Star::Ren | Star::Chong => 1,
81        Star::Ying | Star::Zhu => -1,
82        Star::Peng | Star::Rui => -2,
83    }
84}
85
86/// Auspice of a deity: 值符 (+2); 九天 九地 太陰 六合 (+1); 螣蛇 玄武 (-1); 白虎 (-2).
87pub fn deity_score(deity: Deity) -> i32 {
88    match deity {
89        Deity::ZhiFu => 2,
90        Deity::JiuTian | Deity::JiuDi | Deity::TaiYin | Deity::LiuHe => 1,
91        Deity::TengShe | Deity::XuanWu => -1,
92        Deity::BaiHu => -2,
93    }
94}

96/// Palace where a heaven-plate stem sits "in its own temple" (三奇升殿): 乙 in 震, 丙 in 離, 丁 in 兌.
97pub fn temple_palace(stem: HeavenlyStem) -> Option<Palace> {
98    match stem {
99        HeavenlyStem::Yi => Some(Palace::Zhen3),
100        HeavenlyStem::Bing => Some(Palace::Li9),
101        HeavenlyStem::Ding => Some(Palace::Dui7),
102        _ => None,
103    }
104}
105
106/// Palace where a heaven-plate stem is "punished" (六儀擊刑): 戊 in 震, 己 in 坤, 庚 in 艮,
107/// 辛 in 離, 壬 and 癸 in 巽.
108pub fn punishment_palace(stem: HeavenlyStem) -> Option<Palace> {
109    match stem {
110        HeavenlyStem::Wu => Some(Palace::Zhen3),
111        HeavenlyStem::Ji => Some(Palace::Kun2),
112        HeavenlyStem::Geng => Some(Palace::Gen8),
113        HeavenlyStem::Xin => Some(Palace::Li9),
114        HeavenlyStem::Ren | HeavenlyStem::Gui => Some(Palace::Xun4),
115        _ => None,
116    }
117}
118
119/// Palace where a three-wonder stem enters its tomb (三奇入墓): 乙 in 坤, 丙 in 乾, 丁 in 艮.
120pub fn tomb_palace(stem: HeavenlyStem) -> Option<Palace> {
121    match stem {
122        HeavenlyStem::Yi => Some(Palace::Kun2),
123        HeavenlyStem::Bing => Some(Palace::Qian6),
124        HeavenlyStem::Ding => Some(Palace::Gen8),
125        _ => None,
126    }
127}

129#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
130pub enum PatternKind {
131    Auspicious,
132    Inauspicious,
133}
134
135/// A named formation. `id` is stable for programmatic use.
136#[derive(Debug, Clone, Copy)]
137pub struct PatternInfo {
138    pub id: &'static str,
139    pub hanja: &'static str,
140    pub english: &'static str,
141    pub korean: &'static str,
142    pub kind: PatternKind,
143}
144
145const fn pat(
146    id: &'static str,
147    hanja: &'static str,
148    english: &'static str,
149    korean: &'static str,
150    kind: PatternKind,
151) -> PatternInfo {
152    PatternInfo {
153        id,
154        hanja,
155        english,
156        korean,
157        kind,
158    }
159}
