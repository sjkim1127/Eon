1//! Choosing the 用神 (the palaces that stand for the matter asked about) and reading them.
2//!
3//! The choice of 用神 follows common manual practice; schools differ, so treat the result as one
4//! reasonable reading. Scores are the heuristic palace scores of [`crate::analysis::insights`].
5
6use crate::analysis::insights::{QimenInsights, Verdict};
7use crate::core::{Deity, Door, Palace, QimenPan, Star};
8use eon_saju::core::stem::HeavenlyStem;
9use serde::{Deserialize, Serialize};
10
11#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
12#[serde(rename_all = "camelCase")]
13pub enum QimenQuestion {
14    /// General reading: the querent and the matter of the hour
15    General,
16    Wealth,
17    Career,
18    Marriage,
19    Health,
20    Travel,
21    /// Contest or lawsuit: the querent against the opponent
22    Lawsuit,
23    Study,
24}
25
26#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
27#[serde(rename_all = "camelCase")]
28pub struct YongShenReading {
29    pub label: String,
30    pub palace: Palace,
31    pub score: i32,
32    pub verdict: Verdict,
33}
34
35#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
36#[serde(rename_all = "camelCase")]
37pub struct QuestionReading {
38    pub question: QimenQuestion,
39    /// Palace of the day stem (the querent) and its score
40    pub self_palace: Option<Palace>,
41    pub self_score: i32,
42    pub yongshen: Vec<YongShenReading>,
43    /// Aggregate questions: half the querent's score plus half the mean 用神 score.
44    /// Contests: the querent's palace score minus the opponent's.
45    pub overall_score: f64,
46    pub verdict: Verdict,
47    pub advice_en: String,
48    pub advice_ko: String,
49}

51fn palace_of_door(pan: &QimenPan, door: Door) -> Option<Palace> {
52    pan.palaces
53        .iter()
54        .find(|p| p.door == Some(door))
55        .map(|p| p.palace)
56}
57
58fn palace_of_star(pan: &QimenPan, star: Star) -> Option<Palace> {
59    pan.palaces
60        .iter()
61        .find(|p| p.palace != Palace::Center5 && p.star == Some(star))
62        .map(|p| p.palace)
63}
64
65fn palace_of_deity(pan: &QimenPan, deity: Deity) -> Option<Palace> {
66    pan.palaces
67        .iter()
68        .find(|p| p.deity == Some(deity))
69        .map(|p| p.palace)
70}
71
72fn palace_of_heaven_stem(pan: &QimenPan, stem: HeavenlyStem) -> Option<Palace> {
73    pan.palaces
74        .iter()
75        .find(|p| p.palace != Palace::Center5 && p.heaven_stem == Some(stem))
76        .map(|p| p.palace)
77}

79/// Read the plate for a question. `is_male` only matters for marriage (the spouse stem differs).
80pub fn read_question(
81    pan: &QimenPan,
82    insights: &QimenInsights,
83    question: QimenQuestion,
84    is_male: bool,
85) -> QuestionReading {
86    let score_of = |palace: Palace| insights.palace(palace).map(|p| p.score).unwrap_or(0);
87
88    let mut yongshen: Vec<(String, Option<Palace>)> = Vec::new();
89    let mut contest = false;
90    match question {
91        QimenQuestion::General => {
92            yongshen.push(("值使門 (the matter in hand)".into(), insights.envoy_palace));
93            yongshen.push(("時干宮 (the hour)".into(), insights.event_palace));
94        }
95        QimenQuestion::Wealth => {
96            yongshen.push(("生門 (income)".into(), palace_of_door(pan, Door::Sheng)));
97            yongshen.push((
98                "戊 (capital)".into(),
99                palace_of_heaven_stem(pan, HeavenlyStem::Wu),
100            ));
101        }
102        QimenQuestion::Career => {
103            yongshen.push(("開門 (office)".into(), palace_of_door(pan, Door::Kai)));
104            yongshen.push((
105                "值符 (authority)".into(),
106                palace_of_deity(pan, Deity::ZhiFu),
107            ));
108        }
109        QimenQuestion::Marriage => {
110            let (label, stem) = if is_male {
111                ("乙 (wife)", HeavenlyStem::Yi)
112            } else {
113                ("庚 (husband)", HeavenlyStem::Geng)
114            };
115            yongshen.push((label.into(), palace_of_heaven_stem(pan, stem)));
116            yongshen.push(("六合 (union)".into(), palace_of_deity(pan, Deity::LiuHe)));
117        }
118        QimenQuestion::Health => {
119            yongshen.push(("天心 (physician)".into(), palace_of_star(pan, Star::Xin)));
120        }
121        QimenQuestion::Travel => {
122            yongshen.push(("值使門 (the journey)".into(), insights.envoy_palace));
123            yongshen.push(("馬星 (movement)".into(), insights.horse_palace));
124        }
125        QimenQuestion::Lawsuit => {
126            contest = true;
127            yongshen.push(("時干宮 (opponent)".into(), insights.event_palace));
128        }
129        QimenQuestion::Study => {
130            yongshen.push(("景門 (documents)".into(), palace_of_door(pan, Door::Jing)));
131            yongshen.push(("天輔 (learning)".into(), palace_of_star(pan, Star::Fu)));
132            yongshen.push((
133                "丁 (scholar's wonder)".into(),
134                palace_of_heaven_stem(pan, HeavenlyStem::Ding),
135            ));
136        }
137    }
138
139    let yongshen: Vec<YongShenReading> = yongshen
140        .into_iter()
141        .filter_map(|(label, palace)| {
142            palace.map(|palace| {
143                let score = score_of(palace);
144                YongShenReading {
145                    label,
146                    palace,
147                    score,
148                    verdict: Verdict::from_score(f64::from(score)),
149                }
150            })
151        })
152        .collect();
153
154    let self_score = insights.self_palace.map(score_of).unwrap_or(0);
155    let mean = if yongshen.is_empty() {
156        0.0
157    } else {
158        yongshen.iter().map(|y| f64::from(y.score)).sum::<f64>() / yongshen.len() as f64
159    };
160    let overall_score = if contest {
161        f64::from(self_score) - mean
162    } else {
163        0.5 * f64::from(self_score) + 0.5 * mean
164    };
165    let verdict = Verdict::from_score(overall_score);
166
167    let (advice_en, advice_ko) = match (verdict, contest) {
168        (Verdict::VeryAuspicious | Verdict::Auspicious, false) => (
169            "The palaces that stand for you and for the matter are supportive: a favourable moment to act.",
170            "나와 사안을 뜻하는 궁이 모두 받쳐 주는 흐름입니다. 움직이기 좋은 때입니다.",
171        ),
172        (Verdict::Neutral, false) => (
173            "Mixed or quiet signals: proceed carefully and keep options open.",
174            "길흉이 섞이거나 잠잠한 신호입니다. 신중히 진행하고 선택지를 열어 두세요.",
175        ),
176        (Verdict::Inauspicious | Verdict::VeryInauspicious, false) => (
177            "The palaces are under strain: postpone or reduce commitments if you can.",
178            "관련 궁이 눌려 있습니다. 가능하면 미루거나 규모를 줄이세요.",
179        ),
180        (Verdict::VeryAuspicious | Verdict::Auspicious, true) => (
181            "Your palace outweighs the opponent's: the position favours you.",
182            "내 궁이 상대 궁보다 강합니다. 형세가 유리합니다.",
183        ),
184        (Verdict::Neutral, true) => (
185            "The two sides are evenly matched: negotiation beats confrontation.",
186            "양쪽이 비슷합니다. 대결보다 협상이 낫습니다.",
187        ),
188        (Verdict::Inauspicious | Verdict::VeryInauspicious, true) => (
189            "The opponent's palace outweighs yours: avoid a head-on contest.",
190            "상대 궁이 더 강합니다. 정면 승부는 피하세요.",
191        ),
192    };
193
194    QuestionReading {
195        question,
196        self_palace: insights.self_palace,
197        self_score,
198        yongshen,
199        overall_score,
200        verdict,
201        advice_en: advice_en.to_string(),
202        advice_ko: advice_ko.to_string(),
203    }
204}
