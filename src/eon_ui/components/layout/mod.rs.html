325/// 사주와 베딕 데이터가 둘 다 성공 상태일 때 통합 운명 티어 분석을 연쇄 수행합니다.
326fn check_and_run_tier(
327    mut state: AnalysisState,
328    saju_res_opt: Option<eon_service::dto::SajuAnalysisOutput>,
329    vedic_res_opt: Option<eon_service::dto::VedicAnalysisOutput>,
330) {
331    let saju_res = saju_res_opt.or_else(|| {
332        if matches!(state.saju.read().status, TaskStatus::Success) {
333            state.saju.read().data.clone()
334        } else {
335            None
336        }
337    });
338
339    let vedic_res = vedic_res_opt.or_else(|| {
340        if matches!(state.vedic.read().status, TaskStatus::Success) {
341            state.vedic.read().data.clone()
342        } else {
343            None
344        }
345    });
346
347    if let (Some(s), Some(v)) = (saju_res, vedic_res) {
348        state.tier.write().status = TaskStatus::Loading;
349        match facade::analyze_destiny_tier(s, v, None) {
350            Ok(res) => {
351                state.tier.write().data = Some(res);
352                state.tier.write().status = TaskStatus::Success;
353            }
354            Err(e) => {
355                state.tier.write().error = Some(e.to_string());
356                state.tier.write().status = TaskStatus::Error(e.to_string());
357            }
358        }
359    }
360}

428/// Write locale code to localStorage (best-effort — no-op on error).
429fn web_sys_set_locale(code: &str) -> Result<(), ()> {
430    use wasm_bindgen::prelude::*;
431    #[wasm_bindgen]
432    extern "C" {
433        #[wasm_bindgen(js_namespace = localStorage, js_name = setItem)]
434        fn ls_set(key: &str, value: &str);
435    }
436    ls_set("eon_locale", code);
437    Ok(())
438}
