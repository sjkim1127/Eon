29impl DestinyEntropy {
30    pub fn analyze(pillars: &FourPillars) -> EntropyAnalysis {
31        // 1. Collect all elements (stems + branches + hidden stems for higher data density)
32        let mut element_counts = [0.0f32; 5];
33        let mut total_elements = 0.0f32;
34
35        // Base 8 characters
36        for (el, count) in pillars.element_counts() {
37            element_counts[el.index() as usize] += count as f32;
38            total_elements += count as f32;
39        }
40
41        // Add hidden stems (Jijanggan) data
42        for branch in [
43            &pillars.year.branch,
44            &pillars.month.branch,
45            &pillars.day.branch,
46            &pillars.hour.branch,
47        ] {
48            for stem in branch.jijanggan() {
49                element_counts[stem.element().index() as usize] += 0.3;
50                total_elements += 0.3;
51            }
52        }
53
54        // 2. Compute Shannon Entropy: H = -sum(pi * log2(pi))
55        let mut entropy = 0.0f32;
56        for count in element_counts {
57            if count > 0.0 {
58                let p = count / total_elements;
59                entropy -= p * p.log2();
60            }
61        }
62
63        // 3. Classify level
64        // Max entropy is log2(5) ≈ 2.32
65        let (level, description) = if entropy < 1.0 {
66            (ObfuscationLevel::Plaintext, "Five elements are concentrated in one domain, providing a clear path. Goals are distinct, though flexibility may be limited.".to_string())
67        } else if entropy < 1.8 {
68            (
69                ObfuscationLevel::Standard,
70                "Five elements are evenly distributed, conferring balanced life momentum."
71                    .to_string(),
72            )
73        } else if entropy < 2.1 {
74            (
75                ObfuscationLevel::Packed,
76                "Diverse energies are intricately bound, revealing their true character at critical junctures.".to_string(),
77            )
78        } else {
79            (ObfuscationLevel::Encrypted, "Five elements are highly entangled, presenting complex life dynamics requiring in-depth analysis.".to_string())
80        };
81
82        // 4. Find unpacker element: primary yongshin resolving complexity
83        let unpacker = pillars.yongshin().primary;
84
85        EntropyAnalysis {
86            score: entropy,
87            level,
88            description,
89            unpacker_element: Some(unpacker),
90        }
91    }
92}
