1//! Hardware-accelerated batch simulator
2//!
3//! Simulates hundreds of thousands of life frames in milliseconds via multi-core parallel processing.
4//! Automatically falls back to sequential execution in WASM environments.
5
6#[cfg(feature = "parallel")]
7use rayon::prelude::*;
8
9use crate::analysis::major_luck::MajorLuckAnalysis;
10use crate::core::ganzi::GanZi;
11use crate::core::pillars::FourPillars;
12use crate::engine::vm::{LifeFrame, SajuVM};
13
14pub struct BatchSimulator {
15    vm: SajuVM,
16}
17
18impl BatchSimulator {
19    pub fn new(pillars: FourPillars) -> Self {
20        Self {
21            vm: SajuVM::new(pillars),
22        }
23    }
24
25    /// Simulate 1,000 years exhaustive life frames
26    pub fn simulate_1000_years(
27        &self,
28        start_year: i32,
29        major_luck: &MajorLuckAnalysis,
30    ) -> Vec<LifeFrame> {
31        let years: Vec<i32> = (start_year..start_year + 1000).collect();
32
33        let mapper = |&year: &i32| {
34            let age = (year - start_year).unsigned_abs();
35            let yearly_ganzi = GanZi::from_year(year);
36            let major_ganzi = major_luck
37                .cycle_for_year(year)
38                .map(|l| l.ganzi)
39                .unwrap_or(yearly_ganzi);
40            self.vm
41                .step(age, major_ganzi, yearly_ganzi, None, None, None)
42        };
43
44        #[cfg(feature = "parallel")]
45        {
46            years.par_iter().map(mapper).collect()
47        }
48
49        #[cfg(not(feature = "parallel"))]
50        {
51            years.iter().map(mapper).collect()
52        }
53    }
54
55    /// Simulate specified range of years
56    pub fn simulate_range(
57        &self,
58        start_year: i32,
59        end_year: i32,
60        major_luck: &MajorLuckAnalysis,
61    ) -> Vec<LifeFrame> {
62        let years: Vec<i32> = (start_year..=end_year).collect();
63
64        let mapper = |&year: &i32| {
65            let age = (year - start_year).unsigned_abs();
66            let yearly_ganzi = GanZi::from_year(year);
67            let major_ganzi = major_luck
68                .cycle_for_year(year)
69                .map(|l| l.ganzi)
70                .unwrap_or(yearly_ganzi);
71            self.vm
72                .step(age, major_ganzi, yearly_ganzi, None, None, None)
73        };
74
75        #[cfg(feature = "parallel")]
76        {
77            years.par_iter().map(mapper).collect()
78        }
79
80        #[cfg(not(feature = "parallel"))]
81        {
82            years.iter().map(mapper).collect()
83        }
84    }
85}

87#[cfg(test)]
88mod tests {
89    use super::*;
90    use crate::core::branch::EarthlyBranch as B;
91    use crate::core::pillars::SajuInput;
92    use crate::core::stem::HeavenlyStem as S;
93    use eon_core::Gender;
94
95    #[test]
96    fn batch_major_luck_follows_cycle_start_date_not_age() {
97        let input = SajuInput::new_solar(2004, 11, 27, 22, 0);
98        let pillars = FourPillars::calculate(&input).unwrap();
99        let major_luck = pillars
100            .major_luck(Gender::Male, 2004, 11, 27, 22, 0)
101            .unwrap();
102        let sim = BatchSimulator::new(pillars);
103        let frames = sim.simulate_range(2004, 2030, &major_luck);
104        let major_at = |year: i32| frames[(year - 2004) as usize].major_ganzi;
105
106        // 丁丑 runs 2017-12-27..2027-12-27: 2027 is still 丁丑, 2028 is 戊寅.
107        assert_eq!(major_at(2018), GanZi::new(S::Ding, B::Chou));
108        assert_eq!(major_at(2027), GanZi::new(S::Ding, B::Chou));
109        assert_eq!(major_at(2028), GanZi::new(S::Wu, B::Yin));
110    }
