1//! Dynamic GanZi Calculation Utilities
2//!
3//! Provides shared functions for dynamic calculation of month, day, and hour GanZi.
4//! Uses the identical core logic as `FourPillars::calculate_*` to guarantee consistency.
5//!
6//! ## Usage
7//! - Used during high-resolution timeline sweeps in the fuzzer
8//! - Used for rapid calculation of GanZi for a given date/time
9
10use crate::core::branch::EarthlyBranch;
11use crate::core::ganzi::GanZi;
12use crate::core::stem::HeavenlyStem;
13use chrono::{DateTime, NaiveDate, Utc};

15/// Calculates the month GanZi corresponding to a specific year and month.
16///
17/// # Arguments
18/// * `year` - Solar year
19/// * `month` - Solar month (1-12)
20///
21/// # Returns
22/// GanZi of the specified month
23///
24/// # Note
25/// Simplified calculation without considering precise solar term boundaries.
26/// If precision solar term month GanZi is required, use `FourPillars::calculate_month_pillar`.
27pub fn calculate_month_ganzi(year: i32, month: i32) -> GanZi {
28    let year_ganzi = GanZi::from_year(year);
29
30    // Five Tigers Seeking Month (五虎遁月法): Determine Yin-month stem from Year stem
31    // Jia/Ji years -> Bing-Yin month, Yi/Geng years -> Wu-Yin month, Bing/Xin years -> Geng-Yin month,
32    // Ding/Ren years -> Ren-Yin month, Wu/Gui years -> Jia-Yin month
33    let first_month_stem_idx = match year_ganzi.stem.index() % 5 {
34        0 => 2, // Jia, Ji -> Bing
35        1 => 4, // Yi, Geng -> Wu
36        2 => 6, // Bing, Xin -> Geng
37        3 => 8, // Ding, Ren -> Ren
38        4 => 0, // Wu, Gui -> Jia
39        _ => 0,
40    };
41
42    // Month branch: 1st month = Yin(2), 2nd month = Mao(3), etc.
43    // Approximate mapping from solar month to lunar month (ignoring exact solar term cutoffs)
44    let month_branch_idx = (month + 1) % 12;
45    let month_stem_idx = (first_month_stem_idx + (month - 1)) % 10;
46
47    GanZi::new(
48        HeavenlyStem::from_index(month_stem_idx),
49        EarthlyBranch::from_index(month_branch_idx),
50    )
51}

53/// Calculates the day GanZi corresponding to a specific date.
54///
55/// # Arguments
56/// * `year` - Solar year
57/// * `month` - Solar month (1-12)
58/// * `day` - Solar day (1-31)
59///
60/// # Returns
61/// GanZi of the specified day
62///
63/// # Note
64/// Internally uses `eon_data::manseryuk::get_day_ganzi_index` to guarantee
65/// consistency with Manseryuk ephemeris calendar data.
66pub fn calculate_day_ganzi(year: i32, month: u32, day: u32) -> GanZi {
67    if let Some(date) = NaiveDate::from_ymd_opt(year, month, day) {
68        let idx = eon_data::manseryuk::get_day_ganzi_index(date);
69        GanZi::from_index(idx as i32)
70    } else {
71        // Fallback for invalid date (should not occur)
72        GanZi::from_index(0)
73    }
74}

76/// Day GanZi calculation using Julian Day Number (without Manseryuk calendar cache)
77///
78/// # Arguments
79/// * `year` - Solar year
80/// * `month` - Solar month (1-12)
81/// * `day` - Solar day (1-31)
82///
83/// # Returns
84/// GanZi of the specified day
85///
86/// # Note
87/// Computes day GanZi purely algorithmically without Manseryuk tables.
88/// When highest precision is required, use `calculate_day_ganzi`.
89pub fn calculate_day_ganzi_jdn(year: i32, month: u32, day: u32) -> GanZi {
90    // Gregorian calendar -> Julian Day Number conversion (simplified formula)
91    let a = (14 - month as i32) / 12;
92    let y = year + 4800 - a;
93    let m = month as i32 + 12 * a - 3;
94    let jdn = day as i64 + (153 * m + 2) as i64 / 5 + 365 * y as i64 + y as i64 / 4
95        - y as i64 / 100
96        + y as i64 / 400
97        - 32045;
98
99    // JDN + 49 -> 60 JiaZi index (JiaZi = 0 baseline alignment)
100    let idx = (jdn + 49).rem_euclid(60);
101    GanZi::from_index(idx as i32)
102}

104/// Calculates the hour GanZi corresponding to a specific hour.
105///
106/// # Arguments
107/// * `daily_ganzi` - Day GanZi (required for determining hour stem)
108/// * `hour` - Hour (0-23)
109///
110/// # Returns
111/// GanZi of the specified hour
112pub fn calculate_hour_ganzi(daily_ganzi: GanZi, hour: u32) -> GanZi {
113    let branch = EarthlyBranch::from_hour(hour as u8);
114
115    // Five Rats Seeking Hour (五子遁時法): Determine Zi-hour stem from Day stem
116    // Jia/Ji day -> Jia-Zi, Yi/Geng day -> Bing-Zi, Bing/Xin day -> Wu-Zi,
117    // Ding/Ren day -> Geng-Zi, Wu/Gui day -> Ren-Zi
118    let zi_stem_idx = (daily_ganzi.stem.index() % 5) * 2;
119    let stem = HeavenlyStem::from_index((zi_stem_idx + branch.index()) as i32);
120
121    GanZi::new(stem, branch)
122}

124/// Calculates the full GanZi pillar set from a given DateTime<Utc>.
125///
126/// # Arguments
127/// * `dt` - UTC timestamp
128///
129/// # Returns
130/// (Year GanZi, Month GanZi, Day GanZi, Hour GanZi) tuple
131///
132/// # Note
133/// Simplified calculation without considering precise solar term boundaries.
134pub fn calculate_all_ganzi_from_datetime(dt: DateTime<Utc>) -> (GanZi, GanZi, GanZi, GanZi) {
135    use chrono::Datelike;
136    use chrono::Timelike;
137
138    let year = dt.year();
139    let month = dt.month() as i32;
140    let day = dt.day();
141    let hour = dt.hour();
142
143    let year_ganzi = GanZi::from_year(year);
144    let month_ganzi = calculate_month_ganzi(year, month);
145    let day_ganzi = calculate_day_ganzi(year, month as u32, day);
146    let hour_ganzi = calculate_hour_ganzi(day_ganzi, hour);
147
148    (year_ganzi, month_ganzi, day_ganzi, hour_ganzi)
149}

151#[cfg(test)]
152mod tests {
153    use super::*;
154
155    #[test]
156    fn test_month_ganzi_basic() {
157        // January 2024 (Yin month of Jia-Chen year) -> Bing-Yin month
158        let mg = calculate_month_ganzi(2024, 1);
159        assert_eq!(mg.stem, HeavenlyStem::Bing); // 丙
160        assert_eq!(mg.branch, EarthlyBranch::Yin); // 寅
161    }
162
163    #[test]
164    fn test_day_ganzi_consistency() {
165        // Verify consistency between Manseryuk-based and JDN-based calculation
166        let day1 = calculate_day_ganzi(2024, 1, 1);
167        let day1_jdn = calculate_day_ganzi_jdn(2024, 1, 1);
168
169        // Both methods should produce identical branch outcomes
170        assert_eq!(day1.branch, day1_jdn.branch);
171    }
172
173    #[test]
174    fn test_hour_ganzi() {
175        // Jia-Zi day Zi hour (0:00) -> Jia-Zi hour
176        let day = GanZi::new(HeavenlyStem::Jia, EarthlyBranch::Zi);
177        let hour = calculate_hour_ganzi(day, 0);
178        assert_eq!(hour.stem, HeavenlyStem::Jia);
179        assert_eq!(hour.branch, EarthlyBranch::Zi);
180
181        // Jia-Zi day Wu hour (12:00) -> Geng-Wu hour
182        let hour_noon = calculate_hour_ganzi(day, 12);
183        assert_eq!(hour_noon.stem, HeavenlyStem::Geng);
184        assert_eq!(hour_noon.branch, EarthlyBranch::Wu);
185    }
