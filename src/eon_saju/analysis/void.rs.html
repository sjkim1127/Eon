1//! 공망(空亡, Void/Emptiness) 분석
2//!
3//! 일주(日柱)를 기준으로 천간 10자와 지지 12자의 짝을 맞출 때
4//! 남게 되는 두 개의 지지를 분석합니다.

12/// 공망 분석 결과
13#[derive(Debug, Clone, Serialize, Deserialize)]
14#[serde(rename_all = "camelCase")]
15pub struct VoidAnalysis {
16    /// 해당 사주의 공망 지지 (2개)
17    pub void_branches: [EarthlyBranch; 2],
18    /// 공망이 발생한 위치 (년, 월, 시)
19    pub void_positions: Vec<String>,
20    /// 공망된 십성 목록
21    pub void_ten_gods: Vec<TenGod>,
22    /// 공망 그룹 이름 (예: 갑자순)
23    pub xun_group: String,
24    /// 상세 해석 (Explainable DTO)
25    pub mapped_voids: Vec<VoidDetail>,
26}
27
28#[derive(Debug, Clone, Serialize, Deserialize)]
29#[serde(rename_all = "camelCase")]
30pub struct VoidDetail {
31    pub branch: EarthlyBranch,
32    pub position: String,
33    pub ten_god: crate::core::ten_gods::TenGod,
34    pub summary: String,
35    pub description: String,
36    pub reasons: Vec<String>,
37    pub level: crate::analysis::supplementary_pillars::InterpretationLevel,
38    #[serde(default)]
39    pub is_dissolved: bool,
40    #[serde(default, skip_serializing_if = "Option::is_none")]
41    pub dissolution_reason: Option<String>,
42}

44impl VoidAnalysis {
45    /// 사주 팔자로부터 공망 분석
46    pub fn from_pillars(pillars: &FourPillars) -> Self {
47        let day_pillar = pillars.day;
48        let (void_branches, xun_group) = calculate_void_branches(day_pillar);
49
50        let mut void_positions = Vec::new();
51        let mut void_ten_gods = Vec::new();
52        let mut mapped_voids = Vec::new();
53
54        let checks = [
55            ("년주", pillars.year),
56            ("월주", pillars.month),
57            ("시주", pillars.hour),
58        ];
59
60        let dm = pillars.day_master();
61
62        for (pos, ganzi) in checks {
63            if void_branches.contains(&ganzi.branch) {
64                let tg = TenGod::from_stem_and_branch(dm, ganzi.branch);
65                void_positions.push(pos.to_string());
66                void_ten_gods.push(tg);
67
68                // 공망해충/공망해합 (해소) 여부 검사
69                let dissolution = check_void_dissolution(ganzi.branch, pos, pillars);
70                let (is_dissolved, dissolution_reason) = match dissolution {
71                    Some((diss, msg)) => (diss, Some(msg)),
72                    None => (false, None),
73                };
74
75                let mut reasons = vec![
76                    format!("일주 기준 공망: {}", ganzi.branch.hangul()),
77                    format!("{} 위치 중복", pos),
78                ];
79
80                let (summary, level, desc_extra) = if is_dissolved {
81                    let reason_str = dissolution_reason.clone().unwrap_or_default();
82                    reasons.push(reason_str.clone());
83                    (
84                        format!("{}에 위치한 {} 공망 (해충/해합 해소)", pos, tg.hangul()),
85                        crate::analysis::supplementary_pillars::InterpretationLevel::Neutral,
86                        format!(" (단, {}에 의해 공망이 해소/구원되었습니다.)", reason_str),
87                    )
88                } else {
89                    (
90                        format!("{}에 위치한 {} 공망", pos, tg.hangul()),
91                        crate::analysis::supplementary_pillars::InterpretationLevel::Caution,
92                        "".to_string(),
93                    )
94                };
95
96                let description = match pos {
97                    "년주" => "선조나 국가적 혜택이 약하거나, 어린 시절의 근간이 흔들릴 수 있음을 의미합니다.",
98                    "월주" => "부모/형제운이 약하거나 직업적 정착에 더 많은 노력이 필요할 수 있습니다.",
99                    "시주" => "자녀나 말년의 결실이 예상보다 늦게 나타나거나 허망함이 있을 수 있습니다.",
100                    _ => "",
101                };
102
103                mapped_voids.push(VoidDetail {
104                    branch: ganzi.branch,
105                    position: pos.to_string(),
106                    ten_god: tg,
107                    summary,
108                    description: format!("{}{}", description, desc_extra),
109                    reasons,
110                    level,
111                    is_dissolved,
112                    dissolution_reason,
113                });
114            }
115        }
116
117        Self {
118            void_branches,
119            void_positions,
120            void_ten_gods,
121            xun_group,
122            mapped_voids,
123        }
124    }
125}

127/// 공망 해충 / 공망 해합 여부 판정
128fn check_void_dissolution(
129    target_branch: EarthlyBranch,
130    target_pos: &str,
131    pillars: &FourPillars,
132) -> Option<(bool, String)> {
133    let other_branches = [
134        ("년주", pillars.year.branch),
135        ("월주", pillars.month.branch),
136        ("일주", pillars.day.branch),
137        ("시주", pillars.hour.branch),
138    ];
139    let all_branches = [
140        pillars.year.branch,
141        pillars.month.branch,
142        pillars.day.branch,
143        pillars.hour.branch,
144    ];
145
146    for (pos, b) in other_branches {
147        if pos == target_pos {
148            continue;
149        }
150        if crate::analysis::relationships::BranchClash::check(target_branch, b).is_some() {
151            return Some((
152                true,
153                format!("{} {}와 충(沖)하여 공망 해소 (공망해충)", pos, b.hangul()),
154            ));
155        }
156        if crate::analysis::relationships::SixCombination::check(target_branch, b).is_some() {
157            return Some((
158                true,
159                format!(
160                    "{} {}와 육합(六合)하여 공망 해소 (공망해합)",
161                    pos,
162                    b.hangul()
163                ),
164            ));
165        }
166    }
167
168    let triples = crate::analysis::relationships::TripleCombination::check(&all_branches);
169    if triples
170        .iter()
171        .any(|c| c.branches().contains(&target_branch))
172    {
173        return Some((
174            true,
175            "원국 삼합(三合) 성국으로 공망 해소 (공망해합)".to_string(),
176        ));
177    }
178    let seasonals = crate::analysis::relationships::SeasonalCombination::check(&all_branches);
179    if seasonals
180        .iter()
181        .any(|c| c.branches().contains(&target_branch))
182    {
183        return Some((
184            true,
185            "원국 방합(方合) 성국으로 공망 해소 (공망해합)".to_string(),
186        ));
187    }
188
189    None
190}

192/// 특정 간지의 공망 지지와 순(旬) 그룹 산출
193pub fn calculate_void_branches(ganzi: GanZi) -> ([EarthlyBranch; 2], String) {
194    let s_idx = ganzi.stem.index() as i32;
195    let b_idx = ganzi.branch.index() as i32;
196
197    // 순(旬)의 시작점 계산
198    let xun_start_idx = (b_idx - s_idx).rem_euclid(12);
199
200    let xun_name = match xun_start_idx {
201        0 => "갑자순(甲子旬)",
202        10 => "갑술순(甲戌旬)",
203        8 => "갑신순(甲申旬)",
204        6 => "갑오순(甲午旬)",
205        4 => "갑진순(甲辰旬)",
206        2 => "갑인순(甲寅旬)",
207        _ => "기타",
208    };
209
210    // 공망은 순의 시작점에서 2개 앞 (역순)
211    let v1_idx = (xun_start_idx - 2).rem_euclid(12);
212    let v2_idx = (xun_start_idx - 1).rem_euclid(12);
213
214    let v1 = EarthlyBranch::from_index(v1_idx);
215    let v2 = EarthlyBranch::from_index(v2_idx);
216
217    ([v1, v2], xun_name.to_string())
218}

220impl std::fmt::Display for VoidAnalysis {
221    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
222        writeln!(f, "【공망(空亡) 분석】")?;
223        writeln!(f, "─────────────────────────────────")?;
224        writeln!(f, "일주 기준: {}", self.xun_group)?;
225        writeln!(
226            f,
227            "공망 지지: {} ({}), {} ({})",
228            self.void_branches[0].hangul(),
229            self.void_branches[0].hanja(),
230            self.void_branches[1].hangul(),
231            self.void_branches[1].hanja()
232        )?;
233
234        if self.void_positions.is_empty() {
235            writeln!(f, "▶ 원국(사주) 내에 공망이 없습니다.")?;
236        } else {
237            for (idx, pos) in self.void_positions.iter().enumerate() {
238                writeln!(
239                    f,
240                    "▶ {}에 공망 발생 (십성: {})",
241                    pos,
242                    self.void_ten_gods[idx].hangul()
243                )?;
244            }
245            writeln!(
246                f,
247                "  * 해당 육친이나 사회적 기운의 실효성이 낮아질 수 있습니다."
248            )?;
249        }
250        Ok(())
251    }
252}
