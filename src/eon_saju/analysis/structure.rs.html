174/// 격국 분석 결과
175#[derive(Debug, Clone, Serialize, Deserialize)]
176#[serde(rename_all = "camelCase")]
177pub struct StructureAnalysis {
178    /// 결정된 격국
179    pub structure: StructureType,
180    /// 투출된 천간 (있을 경우)
181    pub projected_stem: Option<HeavenlyStem>,
182    /// 투출 위치 (년간, 월간, 시간)
183    pub projection_path: Option<String>,
184    /// 격국 요약
185    pub summary: String,
186    /// 격국 상세 설명
187    pub description: String,
188    /// 격국 성립 이유 (근거 목록)
189    pub reasons: Vec<String>,
190}

192impl StructureAnalysis {
193    pub fn from_pillars(pillars: &FourPillars) -> Self {
194        Self::from_pillars_with_config(pillars, &AnalysisConfig::default())
195    }
196
197    pub fn from_pillars_with_config(pillars: &FourPillars, config: &AnalysisConfig) -> Self {
198        let dm = pillars.day_master();
199        let month_branch = pillars.month.branch;
200        let hidden_stems = month_branch.hidden_stems();
201
202        let other_stems = [
203            ("년간", pillars.year.stem),
204            ("월간", pillars.month.stem),
205            ("시간", pillars.hour.stem),
206        ];
207
208        // -1. 화기격(HwaGi) 우선 판정 (파성 破星 및 쟁합/투합 검증)
209        let mut hwagi_result = None;
210        let adjacent_stems = [("월간", pillars.month.stem), ("시간", pillars.hour.stem)];
211        for (path, stem_on_top) in &adjacent_stems {
212            if let Some(combo) =
213                crate::analysis::relationships::StemCombination::check(dm, *stem_on_top)
214            {
215                let transformed_elem = combo.transformed_element();
216                if month_branch.element() == transformed_elem {
217                    // 1) 합화 오행을 극하는 파성(破星) 존재 여부 검사
218                    let breaker_elem = transformed_elem.controlled_by();
219                    let all_stems = [
220                        pillars.year.stem,
221                        pillars.month.stem,
222                        pillars.day.stem,
223                        pillars.hour.stem,
224                    ];
225                    let has_break_star = all_stems.iter().any(|s| s.element() == breaker_elem);
226
227                    // 2) 쟁합/투합 검사
228                    let same_dm_count = all_stems.iter().filter(|&&s| s == dm).count();
229                    let same_partner_count =
230                        all_stems.iter().filter(|&&s| s == *stem_on_top).count();
231                    let is_competing = same_dm_count > 1 || same_partner_count > 1;
232
233                    if !has_break_star && !is_competing {
234                        hwagi_result = Some((path, combo, transformed_elem, *stem_on_top));
235                        break;
236                    }
237                }
238            }
239        }
240
241        if let Some((path, combo, transformed_elem, matched_stem)) = hwagi_result {
242            return Self {
243                structure: StructureType::HwaGi,
244                projected_stem: Some(matched_stem),
245                projection_path: Some(path.to_string()),
246                summary: format!("일간이 {}과 {}을 이루어 {}로 변하는 화기격", path, combo.hangul(), transformed_elem.hangul()),
247                description: "일간이 인접한 천간과 합을 이루고, 태어난 월(계절)이 그 합화된 기운을 강하게 뒷받침하여 완전히 새로운 기운으로 변화(化)한 특별한 사주입니다.".to_string(),
248                reasons: vec![
249                    format!("일간 {}와 {} {}이 {}", dm.hanja(), path, matched_stem.hanja(), combo.hanja()),
250                    format!("합화 오행({})이 월지 {}과 일치", transformed_elem.hanja(), month_branch.hanja()),
251                    "파성(破星) 없음 및 쟁합/투합 없음 검증 완료".to_string(),
252                ],
253            };
254        }
255
256        // 0. 특수 격국(종격/전왕격) 우선 판정
257        let strength = pillars.strength_with_config(config);
258        let is_polarized = strength.deuk_se.support_ratio >= config.strength.polarized_high
259            || strength.deuk_se.support_ratio <= config.strength.polarized_low;
260
261        if is_polarized {
262            let all_branches = [
263                pillars.year.branch,
264                pillars.month.branch,
265                pillars.day.branch,
266                pillars.hour.branch,
267            ];
268
269            // 일간 및 인성 통근(뿌리) 여부 검사 (진종 眞從 vs 가종 假從 판정용)
270            let dm_el = dm.element();
271            let yin_el = dm_el.generated_by();
272            let has_dm_root = all_branches.iter().any(|b| {
273                b.hidden_stems()
274                    .iter()
275                    .any(|hs| hs.element() == dm_el || hs.element() == yin_el)
276            });
277
278            if strength.deuk_se.support_ratio >= config.strength.polarized_high {
279                // 삼합/방합 전왕 외격 5종 (곡직, 염상, 가색, 종혁, 윤하) 및 전왕격 판정
280                use crate::core::branch::EarthlyBranch::*;
281                use crate::core::element::Element;
282
283                let (structure, name, desc) = match dm_el {
284                    Element::Wood if matches!(month_branch, Yin | Mao | Chen | Hai) => (
285                        StructureType::GokJik,
286                        "곡직격(曲直格 - 木전왕)",
287                        "목(木)의 청아하고 인자한 기운이 전왕을 이루어 성품이 어질고 학문과 예술에 탁월합니다.",
288                    ),
289                    Element::Fire if matches!(month_branch, Si | Wu | Wei | Yin) => (
290                        StructureType::YeomSang,
291                        "염상격(炎上格 - 火전왕)",
292                        "화(火)의 뜨겁고 예의 바른 기운이 전왕을 이루어 명예와 예의가 높고 기상이 화려합니다.",
293                    ),
294                    Element::Earth if matches!(month_branch, Chen | Xu | Chou | Wei) => (
295                        StructureType::GaSaek,
296                        "가색격(稼穡格 - 土전왕)",
297                        "토(土)의 두텁고 신의 있는 기운이 전왕을 이루어 포용력이 크고 재물이 넉넉합니다.",
298                    ),
299                    Element::Metal if matches!(month_branch, Shen | You | Xu | Si) => (
300                        StructureType::JongHyeok,
301                        "종혁격(從革格 - 金전왕)",
302                        "금(金)의 과단하고 숙살하는 기운이 전왕을 이루어 결단력이 강하고 권위를 쥡니다.",
303                    ),
304                    Element::Water if matches!(month_branch, Hai | Zi | Chou | Shen) => (
305                        StructureType::YoonHa,
306                        "윤하격(潤下格 - 水전왕)",
307                        "수(水)의 지혜롭고 흐르는 기운이 전왕을 이루어 지혜가 깊고 만물을 유통시킵니다.",
308                    ),
309                    _ => {
310                        let yinxing = strength.deuk_se.yinxing_count;
311                        let bijie = strength.deuk_se.bijie_count;
312                        if bijie >= yinxing {
313                            (
314                                StructureType::JongWang,
315                                "종왕격(從旺格)",
316                                "자신의 기운이 극도로 강하여 그 기세를 유지해야 하는 전왕 격국입니다.",
317                            )
318                        } else {
319                            (
320                                StructureType::JongGang,
321                                "종강격(從强格)",
322                                "자신을 돕는 인성의 기운이 극도로 강하여 그 기세를 따르는 전왕 격국입니다.",
323                            )
324                        }
325                    }
326                };
327
328                return Self {
329                    structure,
330                    projected_stem: None,
331                    projection_path: None,
332                    summary: format!("일간이 극강하여 기세를 따르는 {}", name),
333                    description: desc.to_string(),
334                    reasons: vec![
335                        format!("일간 세력비율: {:.1}%", strength.deuk_se.support_ratio),
336                        format!("전왕 임계치: {:.1}% 이상", config.strength.polarized_high),
337                    ],
338                };
339            } else {
340                use crate::analysis::power::{AnalysisOptions, IntegratedAnalysis};
341                let options = AnalysisOptions {
342                    apply_transform: false,
343                    apply_correction: true,
344                };
345                let integrated = IntegratedAnalysis::calculate(pillars, options, config);
346
347                let mut shishang_power = 0.0f32;
348                let mut cai_power = 0.0f32;
349                let mut guan_power = 0.0f32;
350
351                for (tg, pct, _) in integrated.ten_god_scores {
352                    match tg {
353                        TenGod::Shishen | TenGod::Shangguan => shishang_power += pct,
354                        TenGod::Zhengcai | TenGod::Piancai => cai_power += pct,
355                        TenGod::Zhengguan | TenGod::Pianguan => guan_power += pct,
356                        _ => {}
357                    }
358                }
359
360                let (structure, name, desc) = if shishang_power >= cai_power
361                    && shishang_power >= guan_power
362                {
363                    if has_dm_root {
364                        (
365                            StructureType::GaJongAh,
366                            "가종아격(假從兒格)",
367                            "일간이나 인성의 미약한 뿌리가 잔재하나 식상 세력을 따르는 가종격입니다.",
368                        )
369                    } else {
370                        (
371                            StructureType::JongAh,
372                            "진종아격(眞從兒格)",
373                            "일간/인성의 뿌리가 일절 없어 식상의 세력을 순수하게 따르는 진종격입니다.",
374                        )
375                    }
376                } else if cai_power >= shishang_power && cai_power >= guan_power {
377                    if has_dm_root {
378                        (
379                            StructureType::GaJongJae,
380                            "가종재격(假從財格)",
381                            "일간이나 인성의 미약한 뿌리가 잔재하나 재물 세력을 따르는 가종격입니다.",
382                        )
383                    } else {
384                        (
385                            StructureType::JongJae,
386                            "진종재격(眞從財格)",
387                            "일간/인성의 뿌리가 일절 없어 재물의 세력을 순수하게 따르는 진종격입니다.",
388                        )
389                    }
390                } else {
391                    if has_dm_root {
392                        (
393                            StructureType::GaJongSal,
394                            "가종살격(假從殺格)",
395                            "일간이나 인성의 미약한 뿌리가 잔재하나 관살 세력을 따르는 가종격입니다.",
396                        )
397                    } else {
398                        (
399                            StructureType::JongSal,
400                            "진종살격(眞從殺格)",
401                            "일간/인성의 뿌리가 일절 없어 관살의 세력을 순수하게 따르는 진종격입니다.",
402                        )
403                    }
404                };
405
406                return Self {
407                    structure,
408                    projected_stem: None,
409                    projection_path: None,
410                    summary: format!("일간이 극약하여 세력을 따르는 {}", name),
411                    description: desc.to_string(),
412                    reasons: vec![
413                        format!("일간 세력비율: {:.1}%", strength.deuk_se.support_ratio),
414                        format!("종격 임계치: {:.1}% 이하", config.strength.polarized_low),
415                        format!(
416                            "일간/인성 지지 통근: {}",
417                            if has_dm_root {
418                                "존재 (가종격 판정)"
419                            } else {
420                                "없음 (진종격 판정)"
421                            }
422                        ),
423                    ],
424                };
425            }
426        }
427
428        // 0.5. 관살혼잡격(官殺混雜格) 판정
429        // 년/월/시간에 정관과 편관이 동시에 투출하고 제화(합/극)되지 않은 경우
430        let exposed_stems = [pillars.year.stem, pillars.month.stem, pillars.hour.stem];
431        let exposed_tgs: Vec<TenGod> = exposed_stems
432            .iter()
433            .map(|&s| TenGod::from_stems(dm, s))
434            .collect();
435        let has_zhengguan = exposed_tgs.contains(&TenGod::Zhengguan);
436        let has_pianguan = exposed_tgs.contains(&TenGod::Pianguan);
437        if has_zhengguan && has_pianguan {
438            return Self {
439                structure: StructureType::GwanSalHonJab,
440                projected_stem: None,
441                projection_path: None,
442                summary: "정관과 편관이 함께 투출하여 혼란스러운 관살혼잡격".to_string(),
443                description: "원국 천간에 정관과 편관(칠살)이 동시에 드러나 있어 직업적 갈등이나 명예 선택의 난관이 따를 수 있으므로 정리가 필요합니다.".to_string(),
444                reasons: vec![
445                    "원국 천간에 정관(正官)과 편관(偏官/七殺)이 동시 투출됨".to_string(),
446                    "관살혼잡(官殺混雜) 상태 형성".to_string(),
447                ],
448            };
449        }
450
451        // 1. 건록격/양인격 우선 판정
452        let stage = crate::core::twelve_stages::calculate_twelve_stage(dm, month_branch);
453        if stage == crate::core::twelve_stages::TwelveStage::Jianlu {
454            return Self {
455                structure: StructureType::JianLu,
456                projected_stem: None,
457                projection_path: None,
458                summary: "일간이 월지에서 기운을 얻은 건록격".to_string(),
459                description: "일간이 가장 왕성한 기운을 가진 시기에 태어나 주관이 뚜렷하고 자수성가할 힘이 있습니다.".to_string(),
460                reasons: vec![format!("일간 {}가 월지 {}에서 12운성 건록(建祿)임", dm.hanja(), month_branch.hanja())],
461            };
462        }
463        if stage == crate::core::twelve_stages::TwelveStage::Diwang
464            && dm.polarity() == Polarity::Yang
465        {
466            return Self {
467                structure: StructureType::YangIn,
468                projected_stem: None,
469                projection_path: None,
470                summary: "가장 강렬한 기운을 품은 양인격".to_string(),
471                description:
472                    "기운이 너무 강하여 칼을 든 것과 같으니, 이를 잘 다스리면 큰 권위를 얻습니다."
473                        .to_string(),
474                reasons: vec![format!(
475                    "양간 {}가 월지 {}에서 12운성 제왕(帝旺)임",
476                    dm.hanja(),
477                    month_branch.hanja()
478                )],
479            };
480        }
481
482        // 2. 투출(透出) 분석 - 정기(본기)부터 역순으로 확인하여 가장 강한 것 선택
483        // 보통은 본기 투출이 가장 강력함
484        for stem_in_branch in hidden_stems.iter().rev() {
485            for (path, stem_on_top) in &other_stems {
486                if stem_in_branch == stem_on_top {
487                    let god = TenGod::from_stems(dm, *stem_on_top);
488                    if let Some(structure) = StructureType::from_ten_god(god) {
489                        return Self {
490                            structure,
491                            projected_stem: Some(*stem_on_top),
492                            projection_path: Some(path.to_string()),
493                            summary: format!("지장간의 기운이 {}에 투출된 {}", path, structure.hangul()),
494                            description: "월지의 숨은 기운이 천간으로 고개를 내밀어 사주의 핵심 성격이 되었습니다.".to_string(),
495                            reasons: vec![format!("월지 {}의 지장간 {}이 {} {}에 투출함", month_branch.hanja(), stem_in_branch.hanja(), path, stem_on_top.hanja())],
496                        };
497                    }
498                }
499            }
500        }
501
502        // 3. 투출된 것이 없으면 월지 본기(정기)로 판정 (월령 격국)
503        let primary_stem = month_branch.primary_stem();
504        let god = TenGod::from_stems(dm, primary_stem);
505        let structure = StructureType::from_ten_god(god).unwrap_or(StructureType::Special);
506
507        Self {
508            structure,
509            projected_stem: None,
510            projection_path: None,
511            summary: format!("월지의 본기를 격으로 삼은 {}", structure.hangul()),
512            description:
513                "천간에 드러난 기운은 없으나 태어난 계절의 기운이 가장 강력한 성격을 형성합니다."
514                    .to_string(),
515            reasons: vec![format!(
516                "투출된 기운이 없어 월지 본기 {}를 기준으로 판정함",
517                primary_stem.hanja()
518            )],
519        }
520    }
521}

523impl std::fmt::Display for StructureAnalysis {
524    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
525        writeln!(f, "【격국(格局) 분석】")?;
526        writeln!(f, "─────────────────────────────────")?;
527        writeln!(
528            f,
529            "▶ {} ({})",
530            self.structure.hangul(),
531            self.structure.hanja()
532        )?;
533        writeln!(f, "  요약: {}", self.summary)?;
534        writeln!(f, "  설명: {}", self.description)?;
535        if !self.reasons.is_empty() {
536            writeln!(f, "  근거: {}", self.reasons.join(", "))?;
537        }
538        Ok(())
539    }
